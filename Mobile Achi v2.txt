class VoicePipeline:
    def __init__(self, ws):
        self.ws = ws
        self.active = True

    async def handle_audio(self, audio_chunk):
        text_task = asyncio.create_task(self.asr(audio_chunk))

        partial_text = await text_task

        llm_stream = self.llm_stream(partial_text)

        async for token in llm_stream:
            await self.ws.send_json({
                "type": "text.stream",
                "chunk": token
            })

    async def asr(self, audio):
        return whisper_model.transcribe(audio)

    async def llm_stream(self, text):
        async for chunk in openai_stream(text):
            yield chunk

      "iceServers": [
        {"urls": "stun:stun.l.google.com:19302"}

    // 🎤 Get mic

    });

    // 🔊 receive audio
    pc!.onTrack = (event) {
      print("Receiving audio...");
    };

    RTCSessionDescription offer = await pc!.createOffer();

    // send to server
    final response = await http.post(
      Uri.parse("http://$serverIp:8000/offer"),
      headers: {"Content-Type": "application/json"},
      body: jsonEncode({
        "sdp": offer.sdp,
        "type": offer.type,
      }),
    );

    final data = jsonDecode(response.body);

    RTCSessionDescription answer = RTCSessionDescription(
      data["sdp"],
      data["type"],
    );

    await pc!.setRemoteDescription(answer);
  }
}

import asyncio
import numpy as np
from faster_whisper import WhisperModel
import edge_tts

# 🔥 Load once (CRITICAL for speed)
whisper = WhisperModel("base", compute_type="int8")


class AIVoicePipeline:

    def __init__(self):
        self.interrupt = False

    # 🎤 STREAMING ASR
    async def transcribe(self, audio_chunk):

        segments, _ = whisper.transcribe(
            audio_chunk,
            beam_size=1,
            language="en"
        )

        text = ""
        for seg in segments:
            text += seg.text

        return text


    # 🧠 STREAMING LLM (mock → replace with OpenAI streaming)
    async def llm_stream(self, text):

        words = text.split()

        for w in words:
            await asyncio.sleep(0.03)
            yield w + " "


    # 🔊 STREAMING TTS
    async def tts_stream(self, text_stream):

        async for chunk in text_stream:

            if self.interrupt:
                break

            communicate = edge_tts.Communicate(chunk, "en-US-AriaNeural")

            async for audio in communicate.stream():
                if audio["type"] == "audio":
                    yield audio["data"]


    # 🔥 FULL PIPELINE
    async def process(self, audio_chunk):

        self.interrupt = False

        text = await self.transcribe(audio_chunk)

        text_stream = self.llm_stream(text)

        async for audio in self.tts_stream(text_stream):
            yield audio


    def stop(self):
        self.interrupt = True

def run_workflow(task_plan, agents, memory, user_input):

    context = user_input
    result = None

    for agent_name in task_plan["agents"]:
        agent = agents[agent_name]

        result = agent.run(context, memory)
        context = result  # pass forward

    memory.add({
        "input": user_input,
        "output": result
    })

    return result

class ClientBrain {
  final EventBus bus;
  final StateMachine sm = StateMachine();
  final LatencyMask latency = LatencyMask();
  final PredictionEngine predictor = PredictionEngine();

  ClientBrain(this.bus);

  void onUserInput(String text) {
    sm.transition(InputReceived());

    latency.start(bus);

    final predicted = predictor.predict(text);
    bus.emit(PredictedEmotion(predicted));
  }

  void onServerStream(String chunk) {
    sm.transition(ResponseStreaming());
    latency.stop(bus);

    bus.emit(StreamChunk(chunk));
  }

  void onComplete() {
    sm.transition(TTSStart());
  }
}

class WSClient {
  final EventBus bus;
  late WebSocketChannel channel;
  final queue = WSQueue();

  WSClient(this.bus) {
    connect();
  }

  void connect() {
    channel = WebSocketChannel.connect(
      Uri.parse("ws://YOUR_IP:8000/ws/mobile"),
    );

    channel.stream.listen(_onMessage,
        onDone: _reconnect,
        onError: (_) => _reconnect());
  }

  void send(String text) {
    final msg = WSMessage.create(text);
    queue.add(msg);

    channel.sink.add(jsonEncode(msg.toJson()));
  }

  void _onMessage(dynamic data) {
    final json = jsonDecode(data);

    if (json["type"] == "ack") {
      queue.ack(json["id"]);
    }

    if (json["type"] == "stream") {
      bus.emit(StreamChunk(json["chunk"]));
    }

    if (json["type"] == "done") {
      bus.emit(StreamDone());
    }
  }

  void _reconnect() {
    Future.delayed(Duration(seconds: 2), connect);
  }
}

}
class WSQueue {
  final Map<String, WSMessage> pending = {};

  void add(WSMessage msg) {
    pending[msg.id] = msg;
  }

  void ack(String id) {
    pending.remove(id);
  }
}

async def websocket_endpoint(ws: WebSocket):
    await ws.accept()
    client_id = str(uuid.uuid4())
    clients[client_id] = ws

    try:
        while True:
            data = await ws.receive_text()
            msg = json.loads(data)

            # ACK
            if msg.get("requires_ack"):
                await ws.send_text(json.dumps({
                    "type": "ack",
                    "id": msg["id"]
                }))

            if msg["type"] == "user_input":
                asyncio.create_task(handle_stream(ws, msg))

    except:
        del clients[client_id]

class EmotionState {
  double valence;   // -1 → +1
  double arousal;   // 0 → 1

  EmotionState({required this.valence, required this.arousal});

  factory EmotionState.neutral() =>
      EmotionState(valence: 0.0, arousal: 0.2);

  EmotionState lerp(EmotionState target, double t) {
    return EmotionState(
      valence: valence + (target.valence - valence) * t,
      arousal: arousal + (target.arousal - arousal) * t,
    );
  }
}
