MILESTONE 2 — IDENTITY & SECURITY (Evaluation Pipeline Order)
================================================================================

  Request → JWT Verification → Credential Lifecycle (hash/expiry/revoke/rotate)

Step 5 — Permission Store

This stores approved permissions.

class PermissionStore:

    def __init__(self):

        self.permissions = {}

    def grant(self, capability, permission):

        self.permissions.setdefault(capability, set())

        self.permissions[capability].add(permission)

    def revoke(self, capability, permission):

        if capability in self.permissions:

            self.permissions[capability].discard(permission)

    def has_permission(self, capability, permission):

        return permission in self.permissions.get(capability, set())

from functools import wraps

def requires(permission):

    def wrapper(func):

        @wraps(func)

                permission

                raise PermissionError()

            return func(self, *args, **kwargs)

        return inner

    return wrapper

public boolean hasCycle(
        String node,
        Set<String> visiting,
        Set<String> visited
){

    if(visiting.contains(node))
        return true;


    if(visited.contains(node))
        return false;


    visiting.add(node);


    for(String child :
        graph.get(node))
    {

        if(hasCycle(
            child,
            visiting,
            visited))
            return true;
    }


    visiting.remove(node);

    visited.add(node);


    return false;
}

5. Authorization Engine

authorization/engine.py

class AuthorizationEngine:


    def check(
        self,
        user,
        permission,
        db
    ):


        result=db.execute(
        """

        SELECT COUNT(*)

        FROM role_permissions rp

        JOIN permissions p
        ON p.id=rp.permission_id

        JOIN user_roles ur
        ON ur.role_id=rp.role_id

        WHERE ur.user_id=:uid

        AND p.name=:permission

        """,
        {
        "uid":user.id,
        "permission":permission
        })


        count=result.scalar()


        return count > 0

from cryptography.hazmat.primitives.asymmetric.ed25519 import (
    Ed25519PrivateKey
)



private_key = (
    Ed25519PrivateKey.generate()
)



def sign(data):

    return private_key.sign(
        data.encode()
    )



def verify(
    public_key,
    data,
    signature
):

    try:

        public_key.verify(
            signature,
            data.encode()
        )

        return True

    except:

        return False

        ON p.id = rp.permission_id

            query,
            {
                "uid":user_id,
                "perm":permission
            }
        )


        return result.scalar() > 0

from datetime import datetime



class DelegationGraph:


    def __init__(self):

        self.edges=[]



    def add(
        self,
        issuer,
        receiver,
        permission,
        expiry
    ):

        self.edges.append({

            "issuer":issuer,

            "receiver":receiver,

            "permission":permission,

            "expiry":expiry

        })



    def find_permission(
        self,
        user,
        permission
    ):


        for edge in self.edges:


            if (
                edge["receiver"]==user
                and
                edge["permission"]==permission
                and
                datetime.utcnow()
                <
                edge["expiry"]
            ):

                return True



        return False

class TrustEngine:


    def __init__(self):

        self.scores={}



    def create_user(
        self,
        user_id
    ):

        self.scores[user_id]=50



    def update(
        self,
        user_id,
        event
    ):


        score=self.scores.get(
            user_id,
            50
        )


        if event=="successful_login":

            score+=5


        if event=="failed_login":

            score-=10


        if event=="policy_violation":

            score-=30


        if event=="verified_action":

            score+=10


        score=max(
            0,
            min(score,100)
        )


        self.scores[user_id]=score



    def get(
        self,
        user_id
    ):

        return self.scores[user_id]

import hashlib
import json
from datetime import datetime


class AuditLedger:


    def generate_hash(
        self,
        previous_hash,
        event
    ):

        payload = (
            previous_hash
            +
            json.dumps(event)
        )


        return hashlib.sha256(
            payload.encode()
        ).hexdigest()



    def create_entry(
        self,
        previous_hash,
        event
    ):


        return {

            "timestamp":
            datetime.utcnow(),

            "event":
            event,

            "previous_hash":
            previous_hash,

            "hash":
            self.generate_hash(
                previous_hash,
                event
            )
        }

        self.members=[]



    def add_member(
        self,
        member
    ):

        self.members.append(member)



    def vote(
        self,
        proposal
    ):

        approvals=0


        for member in self.members:

                approvals += 1


        return approvals > len(
