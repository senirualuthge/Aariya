import { useRef } from "react";
import { useFrame } from "@react-three/fiber";

export default function Planet({ data }) {
  const ref = useRef();

  const state = useRef({
    angle: 0,
    radius: data.orbit_radius,
    size: data.size,
    glow: data.glow
  });

  useFrame((_, delta) => {
    // Smooth transitions
    state.current.radius = lerp(
      state.current.radius,
      data.orbit_radius,
      delta * 2
    );

    state.current.size = lerp(
      state.current.size,
      data.size,
      delta * 2
    );

    state.current.glow = lerp(
      state.current.glow,
      data.glow,
      delta * 3
    );

    // Orbit progression
    state.current.angle += data.orbit_speed * delta;

    ref.current.position.x =
      Math.cos(state.current.angle) * state.current.radius;

    ref.current.position.z =
      Math.sin(state.current.angle) * state.current.radius;

    ref.current.scale.setScalar(state.current.size);
  });

  return (
    <mesh ref={ref}>
      <sphereGeometry args={[1, 32, 32]} />
      <meshStandardMaterial
        emissive={data.color}
        emissiveIntensity={state.current.glow * 2}
        color={data.color}
      />
    </mesh>
  );
}

class SynopticSmoother:
    def __init__(self, alpha=0.2):
        self.alpha = alpha
        self.prev = {}

    def smooth(self, domains: dict):
        smoothed = {}

        for k, v in domains.items():
            prev_v = self.prev.get(k, v)
            new_v = (1 - self.alpha) * prev_v + self.alpha * v
            smoothed[k] = new_v

        self.prev = smoothed
        return smoothed

function computeForces(planets) {
  const forces = planets.map(() => [0, 0, 0]);

  for (let i = 0; i < planets.length; i++) {
    for (let j = 0; j < planets.length; j++) {
      if (i === j) continue;

      const p1 = planets[i];
      const p2 = planets[j];

      const dx = p2.pos[0] - p1.pos[0];
      const dz = p2.pos[2] - p1.pos[2];

      const dist = Math.sqrt(dx * dx + dz * dz) + 0.01;

      // "Cognitive gravity"
      const force =
        (p1.mass * p2.mass) / (dist * dist);

      forces[i][0] += (dx / dist) * force;
      forces[i][2] += (dz / dist) * force;
    }
  }

  return forces;
}

    const data = JSON.parse(msg);

    if (client.readyState === 1) {
      client.send(payload);
    }
  }

from sklearn.cluster import KMeans

class IntentDetector:
    def __init__(self, k=5):
        self.kmeans = KMeans(n_clusters=k)

    def build_features(self, agents):
        return np.array([
            [a["vx"], a["vy"], a["x"], a["y"]]
            for a in agents
        ])

    def detect(self, agents):
        X = self.build_features(agents)

        labels = self.kmeans.fit_predict(X)

        intents = {}

import numpy as np

class ShockwaveEngine:
    def __init__(self, decay=0.85):
        self.decay = decay

    def propagate(self, adjacency, initial_impact, steps=5):
        impact = initial_impact

        history = [impact]

        for _ in range(steps):
            impact = self.decay * adjacency @ impact
            history.append(impact)

        return history

import numpy as np

class SwarmStabilizer:
    def __init__(self, lambda_=0.1):
        self.lambda_ = lambda_

    def compute_entropy(self, positions):
        var = np.var(positions, axis=0)
        return np.sum(var)

    def stabilize(self, agents):
        positions = np.array([[a["x"], a["y"]] for a in agents])

        entropy = self.compute_entropy(positions)

        correction = -self.lambda_ * entropy

        for a in agents:
            a["vx"] += correction * 0.01
            a["vy"] += correction * 0.01

        return agents
Effect
prevents collapse
smooths chaotic bursts
keeps swarm “alive but controlled”

from collections import deque

class NarrativeMemory:
    def __init__(self, size=100):
        self.history = deque(maxlen=size)

    def add_state(self, state):
        self.history.append(state)

    def drift_score(self):
        if len(self.history) < 2:
            return 0

        prev = self.history[-2]
        curr = self.history[-1]

        prev_mean = np.mean([a["x"] for a in prev])
        curr_mean = np.mean([a["x"] for a in curr])

        return abs(curr_mean - prev_mean)

import asyncio
import random
import json
import redis

r = redis.Redis()

AGENTS = [
    {"id": f"a{i}", "x": random.randint(0,100), "y": random.randint(0,100), "vx": 1, "vy": 1}
    for i in range(50)
]

def step():
    for a in AGENTS:
        a["x"] += a["vx"] + random.uniform(-0.5, 0.5)
        a["y"] += a["vy"] + random.uniform(-0.5, 0.5)

    return AGENTS

async def run():
    while True:
        state = step()

        r.set("swarm_state", json.dumps(state))
        r.publish("swarm_stream", json.dumps({
            "type": "state_update",
            "state": state
        }))

        await asyncio.sleep(0.1)

asyncio.run(run())

import redis
import json
import numpy as np

r = redis.Redis()

def predict(state):
    preds = []
    for a in state:
        preds.append({
            "id": a["id"],
            "x": a["x"] + np.random.randn(),
            "y": a["y"] + np.random.randn(),
            "confidence": float(np.random.rand())
        })
    return preds

pub = r.pubsub()
pub.subscribe("swarm_stream")

for msg in pub.listen():
    if msg["type"] != "message":
        continue

    state = json.loads(msg["data"])

    prediction = predict(state["state"])

    r.publish("swarm_stream", json.dumps({
        "type": "prediction_update",
        "prediction": prediction
    }))
