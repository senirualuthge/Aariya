import base64
from vad import vad_process
from asr import transcribe_audio
from tts import stream_tts
from context import ContextManager
from avatar_bridge import emit_avatar

async def voice_ws(ws):
    await ws.accept()
    ctx = ContextManager()
    audio_buffer = bytearray()
    tts_playing = False

    while True:
        msg = await ws.receive_json()

        if msg["type"] == "audio.chunk":
            pcm = base64.b64decode(msg["data"])

            event = vad_process(pcm)

            if event == "speech_start":
                if tts_playing:
                    await ws.send_json({
                        "type": "voice.event",
                        "event": "barge_in",
                        "action": "stop_tts"
                    })
                    tts_playing = False

                audio_buffer.clear()
                await ws.send_json({"type": "voice.event", "event": "speech_start"})

            elif event == "speech":
                audio_buffer.extend(pcm)

            elif event == "speech_end":
                await ws.send_json({"type": "voice.event", "event": "speech_end"})

                text, lang = transcribe_audio(bytes(audio_buffer))
                ctx.add_user(text, lang)

                reply = ctx.generate_reply()
                tts_playing = True

                await ws.send_json({
                    "type": "brain.response",
                    **reply
                })

                await stream_tts(ws, reply, lang)
                emit_avatar(reply)

                tts_playing = False

🧠 LipSyncController.cs
using UnityEngine;

[RequireComponent(typeof(AudioSource))]
public class LipSyncController : MonoBehaviour
{
    public SkinnedMeshRenderer faceMesh;
    public int mouthOpen = 0;
    public int mouthRound = 1;
    public int mouthWide = 2;
    public int mouthClosed = 3;

    private AudioSource audioSource;
    private float[] samples = new float[512];

    void Start()
    {
        audioSource = GetComponent<AudioSource>();
    }

    void Update()
    {
        if (!audioSource.isPlaying) {
            ResetMouth();
            return;
        }

        audioSource.GetOutputData(samples, 0);
        float rms = CalculateRMS(samples);

        float open = Mathf.Clamp01(rms * 15f);
        float round = Mathf.Clamp01(rms * 10f);
        float wide = Mathf.Clamp01(rms * 8f);

        faceMesh.SetBlendShapeWeight(mouthOpen, open * 100f);
        faceMesh.SetBlendShapeWeight(mouthRound, round * 100f);
        faceMesh.SetBlendShapeWeight(mouthWide, wide * 100f);
        faceMesh.SetBlendShapeWeight(mouthClosed, (1f - open) * 100f);
    }

    float CalculateRMS(float[] data)
    {
        float sum = 0f;
        foreach (float s in data) sum += s * s;
        return Mathf.Sqrt(sum / data.Length);
    }

    void ResetMouth()
    {
        faceMesh.SetBlendShapeWeight(mouthOpen, 0);
        faceMesh.SetBlendShapeWeight(mouthRound, 0);
        faceMesh.SetBlendShapeWeight(mouthWide, 0);
        faceMesh.SetBlendShapeWeight(mouthClosed, 100);
    }
}

Zero latency

Cheap CPU cost

using UnityEngine;
using System.Collections;

public class MicroExpressionController : MonoBehaviour
{
    public SkinnedMeshRenderer face;

    [Range(0f,1f)] public float trust = 0.5f;
    public string emotion = "neutral";

    float timer;

    void Update()
    {
        if (trust < 0.4f) return; // NO micro-expressions if trust is low

        timer += Time.deltaTime;

        if (timer > Random.Range(2.5f, 6f))
        {
            timer = 0f;
            TriggerMicroExpression();
        }
    }

    void TriggerMicroExpression()
    {
        float intensity = Mathf.Lerp(2f, 12f, trust);

        if (emotion == "happy")
            StartCoroutine(Blend("SmileL", intensity, 0.25f));
        else if (emotion == "concerned")
            StartCoroutine(Blend("BrowDown", intensity, 0.3f));
        else
            StartCoroutine(Blend("BlinkL", intensity, 0.15f));
    }

    IEnumerator Blend(string shape, float weight, float duration)
    {
        int idx = face.sharedMesh.GetBlendShapeIndex(shape);
        face.SetBlendShapeWeight(idx, weight);
        yield return new WaitForSeconds(duration);
        face.SetBlendShapeWeight(idx, 0);
    }
}

2️⃣ PHONEME TIMING OPTIMIZER (THIS IS RARELY DONE RIGHT)

Most systems:

Align phonemes to audio
Humans:

Move mouth before sound

We fix this.

🎧 Problem

If visemes lag audio by even 20–30ms, humans notice subconsciously.

✅ Solution: Predictive Phoneme Lead
🧠 Timing Model
Audio playback time = T
Viseme activation time = T - lead


Where:

lead = 30–50ms (language dependent)

🧠 Optimizer Formula
lead_ms = base_lead + (speech_rate * factor)

# Example
base_lead = 35
factor = 8
speech_rate = syllables_per_second

🧠 Example

Fast speech (6 syll/sec):

lead = 35 + (6 × 8) = 83ms


Slow speech:

lead = 35 + (2 × 8) = 51ms

🧠 Runtime Adjustment
{
  "phoneme": "EE",
  "audio_time_ms": 420,
  "viseme_time_ms": 350
}


This is why your avatar will feel alive.

🎮 Unity Rule (IMPORTANT)

Never snap visemes.
Always interpolate over 20–40ms.
