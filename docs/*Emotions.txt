function selectIntent(a: AffectSummary): SocialIntent {
  // Safety override
  if (a.safety < 0.3) return "WITHDRAWING";

  // Low trust defensive posture
  if (a.trust < 0.25) {
    if (a.valence < -0.2) return "DEFENSIVE";
    return "LISTENING";
  }

  // Comforting user when sad but bonded
  if (a.dominant === "sadness" && a.attachment > 0.4) return "COMFORTING";

  // High joy + bond → playful
  if (a.dominant === "joy" && a.attachment > 0.5) {
    return a.trust > 0.7 ? "FLIRTING" : "PLAYFUL";
  }

  // Anger → boundary if low safety
  if (a.dominant === "anger") {
    return a.safety < 0.5 ? "SETTING_BOUNDARY" : "DEFENSIVE";
  }

  // Fear → reassurance mode
  if (a.dominant === "fear") return "COMFORTING";

  // Default informational mode
  if (a.arousal < 0.3) return "INFORMING";

  return "LISTENING";
}

function updateTrust(state: AffectState, V: number, C: number, K: number, X: number) {
  const a = 0.4; // valence weight
  const b = 0.3; // contradiction penalty
  const c = 0.2; // continuity bonus
  const d = 0.5; // violations penalty

  const delta =
    a * V +
    b * (1 - C) +
    c * K -
    d * X;

  state.social.trustBaseline = clamp(
    state.social.trustBaseline + delta * 0.05,
    0,
    1
  );
}

function emotionalSafetyGate(memory: Memory, affect: AffectState): boolean {
  const isUserVulnerable =
    affect.current.sadness + affect.current.fear > 0.6;

  if (isUserVulnerable && memory.valence < -0.3) return false;

  return true;
}

Scoring Function
function scoreMemory(memory: Memory, ctx: RetrievalContext): number {
  const similarity = cosine(ctx.queryEmbedding, memory.embedding);

  const recency = Math.exp(-(now() - memory.timestamp) / HALF_LIFE);

  const affectBoost =
    ctx.userIntent === "emotional"
      ? memory.emotionalWeight
      : 0;

  const bondBoost =
    memory.socialTag === "bonding"
      ? ctx.affect.social.attachment
      : 0;

  return (
    similarity * 0.5 +
    recency * 0.2 +
    affectBoost * 0.2 +
    bondBoost * 0.1
  );
}

Full Retrieval Pipeline
function retrieveMemories(memories: Memory[], ctx: RetrievalContext) {
  return memories
    .filter(m => trustGate(m, ctx.affect))
    .filter(m => emotionalSafetyGate(m, ctx.affect))
    .map(m => ({ m, score: scoreMemory(m, ctx) }))
    .sort((a, b) => b.score - a.score)
    .slice(0, 5)
    .map(x => x.m);
}

type Expectation = {

Prediction Inputs

Use:

current tone of agent output

function predictValence(a: AffectState, dialogueAct: string): Expectation {
  const bond = a.social.attachment;
  const mood = a.mood.valence;

  let base = mood * 0.4 + bond * 0.6;

  if (dialogueAct === "correction") base -= 0.3;
  if (dialogueAct === "reassurance") base += 0.2;
  if (dialogueAct === "joke") base += 0.15;

  return {
    predictedValence: clamp(base, -1, 1),
    confidence: clamp(0.5 + bond * 0.4, 0, 1),
  };
}

Pre-Regulation Hook

If negative response expected → dampen joy/anticipation spike, increase caution.

function preregulateEmotion(a: AffectState, exp: Expectation) {
  if (exp.predictedValence < -0.2 && exp.confidence > 0.6) {
    a.current.anticipation *= 0.7;
    a.current.joy *= 0.8;
    a.current.fear += 0.1; // cautious stance
  }
}


This prevents over-enthusiastic mismatch before a likely negative reply.

1️⃣ Expectation Violation Signal

Goal: Compare predicted user response vs actual outcome to adjust surprise, trust, and emotional trajectory.

Implementation
// Compute violation
function computeExpectationViolation(exp: Expectation, actualValence: number) {
  const surprise = Math.abs(exp.predictedValence - actualValence); // 0–1
  const signedViolation = actualValence - exp.predictedValence; // direction of mismatch
  return { surprise, signedViolation };
}

// Update affect based on violation
function applyExpectationViolation(state: AffectState, violation: { surprise: number; signedViolation: number }) {
  // Surprise increases arousal slightly
  state.mood.arousal = clamp(state.mood.arousal + violation.surprise * 0.2, 0, 1);

  // Negative surprise reduces trust
  if (violation.signedViolation < -0.2) {
    state.social.trustBaseline = clamp(state.social.trustBaseline - violation.surprise * 0.15, 0, 1);
    state.current.fear += violation.surprise * 0.1;
  }

  // Positive surprise can boost joy slightly
  if (violation.signedViolation > 0.2) {
    state.current.joy += violation.surprise * 0.1;
    state.current.trust += violation.surprise * 0.05;
  }
}


Where it plugs in: After predicting valence and after receiving actual user input, before routing dialogue.

Behavioral effect:

Agent anticipates reactions but can be “surprised” naturally.

Trust dynamically adjusts to mismatch, creating more realistic relational tension or delight.
