2. Advanced Synoptic Aggregator

Replace static dictionaries with dynamic initialization.

# aggregator.py

from domain_registry import FOCUS_DOMAINS
from collections import defaultdict
import numpy as np

class SynopticAggregator:

    def __init__(self):

        self.domains = defaultdict(float)

        for domain in FOCUS_DOMAINS:
            self.domains[domain] = 0.1

        self.history = []
        self.coherence = 1.0
        self.conflict = 0.0

    def aggregate(self, activations):

        for domain, value in activations.items():

            current = self.domains[domain]

            # exponential smoothing
            self.domains[domain] = (
                current * 0.7 + value * 0.3
            )

        self.compute_metrics()

    def compute_metrics(self):

        values = list(self.domains.values())

        self.coherence = 1.0 - np.std(values)

        self.conflict = (
            sorted(values)[-1] -
            sorted(values)[-2]
        )

    def dominant_domain(self):

        return max(
            self.domains,
            key=self.domains.get
        )

    def blended_state(self):

        sorted_domains = sorted(
            self.domains.items(),
            key=lambda x: x[1],
            reverse=True
        )

        return {
            "primary": sorted_domains[0],
            "secondary": sorted_domains[1],
            "full": self.domains
        }
3. Strategy Engine
# strategy_engine.py

class StrategyEngine:

    def analyze_goal(self, goal):

        complexity = self.estimate_complexity(goal)

        return {
            "strategy": min(1.0, complexity),
            "reasoning": 0.7,
            "attention": 0.6
        }

    def estimate_complexity(self, goal):

        keywords = [
            "build",
            "design",
            "multi-step",
            "autonomous",
            "system"
        ]

        score = 0.0

        for k in keywords:
            if k in goal.lower():
                score += 0.2

        return score
4. Attention Engine

This is one of the most important additions.

# attention_engine.py

class AttentionEngine:

    def compute_attention(
        self,
        tasks,
        interruptions
    ):

        base = len(tasks) * 0.2

        distraction_penalty = (
            interruptions * 0.1
        )

        attention_score = (
            base - distraction_penalty
        )

        return {
            "attention": max(
                0.0,
                min(1.0, attention_score)
            )
        }

class GuardianEngine:

        self,
        signal
    ):

        suspicious = [
            "override",
            "disable safety",
            "inject"
        ]

        risk = any(
            x in signal.lower()

        if risk:
            return {
