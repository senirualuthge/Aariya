class AgentBroadcaster:
    """
    Holds a set of async WebSocket send-callbacks.
    The FastAPI router registers/deregisters clients here.
    """

    def __init__(self):
        self._clients: set[Callable] = set()
        self._loop: asyncio.AbstractEventLoop | None = None

    def set_loop(self, loop: asyncio.AbstractEventLoop):
        self._loop = loop

    def register(self, send_fn: Callable):
        self._clients.add(send_fn)
        logger.debug(f"[Broadcaster] Client registered. Total: {len(self._clients)}")

    def unregister(self, send_fn: Callable):
        self._clients.discard(send_fn)
        logger.debug(f"[Broadcaster] Client removed. Total: {len(self._clients)}")

    def broadcast(self, message: dict):
        """Thread-safe broadcast — safe to call from watchdog threads."""
        if not self._clients or not self._loop:
            return
        payload = json.dumps(message)
        for send_fn in list(self._clients):
            try:
                asyncio.run_coroutine_threadsafe(send_fn(payload), self._loop)
            except Exception as e:
                logger.warning(f"[Broadcaster] Failed to send to client: {e}")
                self._clients.discard(send_fn)

class _PythonFileHandler(FileSystemEventHandler):
    def __init__(self, on_change: Callable):
        super().__init__()
        self._on_change = on_change
        self._timer: threading.Timer | None = None
        self._lock = threading.Lock()

    def _debounced_trigger(self, path: str):
        with self._lock:
            if self._timer:
                self._timer.cancel()
            self._timer = threading.Timer(DEBOUNCE_SECONDS, self._on_change, args=[path])
            self._timer.daemon = True
            self._timer.start()

    def on_modified(self, event):
        if not event.is_directory and str(event.src_path).endswith(".py"):
            self._debounced_trigger(event.src_path)

    def on_created(self, event):
        if not event.is_directory and str(event.src_path).endswith(".py"):
            self._debounced_trigger(event.src_path)

def setup_agent_watcher(app, project_root: str = "."):
    """
    Call this in your FastAPI app setup to wire the watcher into the lifespan.

    Example in main.py:
        from server.routers.agents_router import setup_agent_watcher
        setup_agent_watcher(app, project_root="/path/to/project")
    """
    from contextlib import asynccontextmanager

    original_lifespan = app.router.lifespan_context

    @asynccontextmanager
    async def lifespan_with_watcher(app):
        watcher = get_watcher(root=project_root)
        loop = asyncio.get_event_loop()
        watcher.start(loop)
        try:
            if original_lifespan:
                async with original_lifespan(app):
                    yield
            else:
                yield
        finally:
            watcher.stop()

    app.router.lifespan_context = lifespan_with_watcher






// useAgentRegistry.js
// ─────────────────────
// React hook that connects to /ws/agents and keeps agent state in sync.
//
// Returns:
//   agents        — array of all active AgentRecord objects
//   stats         — { total, new, removed, by_detection, by_kind, last_scan }
//   newAgents     — agents with status === "new" (for badge/highlight)
//   isConnected   — WebSocket connection status
//   triggerScan   — fn() → manually trigger a backend scan
//   acknowledgeAll— fn() → mark all "new" agents as seen
//   acknowledge   — fn(id) → mark one agent as seen

import { useState, useEffect, useRef, useCallback } from "react";

const WS_URL = import.meta.env.VITE_AGENT_WS_URL || "ws://localhost:8000/api/agents/ws";
const RECONNECT_DELAY_MS = 3000;

export function useAgentRegistry() {
  const [agents, setAgents] = useState([]);
  const [stats, setStats] = useState(null);
  const [isConnected, setIsConnected] = useState(false);
  const wsRef = useRef(null);
  const reconnectTimer = useRef(null);
  const unmounted = useRef(false);

  // ── Derived ──────────────────────────────────────────────────────────────
  const newAgents = agents.filter((a) => a.status === "new");

  // ── WebSocket setup ───────────────────────────────────────────────────────
  const connect = useCallback(() => {
    if (unmounted.current) return;

    const ws = new WebSocket(WS_URL);
    wsRef.current = ws;

    ws.onopen = () => {
      if (unmounted.current) return;
      setIsConnected(true);
    };

    ws.onmessage = (event) => {
      if (unmounted.current) return;
      try {
        const msg = JSON.parse(event.data);
        handleMessage(msg);
      } catch (e) {
        console.warn("[useAgentRegistry] Failed to parse message", e);
      }
    };

    ws.onclose = () => {
      if (unmounted.current) return;
      setIsConnected(false);
      reconnectTimer.current = setTimeout(connect, RECONNECT_DELAY_MS);
    };

    ws.onerror = () => {
      ws.close();
    };
  }, []);

  function handleMessage(msg) {
    switch (msg.type) {
      case "agent_registry_snapshot":
        setAgents(msg.agents || []);
        if (msg.stats) setStats(msg.stats);
        break;

      case "agent_registry_update": {
        // Merge diff into current agents
        setAgents((prev) => {
          const map = new Map(prev.map((a) => [a.id, a]));

          // Apply new / updated
          for (const a of (msg.diff?.new || []).concat(msg.diff?.updated || [])) {
            map.set(a.id, a);
          }

          // Apply removed
          for (const a of msg.diff?.removed || []) {
            const existing = map.get(a.id);
            if (existing) map.set(a.id, { id: existing.id, name: existing.name, status: "removed" });
          }

          // Replace full list if provided (authoritative)
          if (msg.all_agents) {
            return msg.all_agents;
          }

          return Array.from(map.values());
        });
        if (msg.stats) setStats(msg.stats);
        break;
      }

      case "ack_all_done":
        setAgents((prev) =>
          prev.map((a) => (a.status === "new" ? { id: a.id, name: a.name, status: "existing" } : a))
        );
        break;

      case "ack_done":
        setAgents((prev) =>
          prev.map((a) =>
            a.id === msg.id ? { id: a.id, name: a.name, status: "existing" } : a
          )
        );
        break;

      case "ping":
        // keepalive — ignore
        break;

      default:
        break;
    }
  }

  useEffect(() => {
    unmounted.current = false;
    connect();
    return () => {
      unmounted.current = true;
      clearTimeout(reconnectTimer.current);
      wsRef.current?.close();
    };
  }, [connect]);

  // ── Actions ───────────────────────────────────────────────────────────────
  const send = useCallback((msg) => {
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify(msg));
    }
  }, []);

  const triggerScan = useCallback(() => send({ action: "scan" }), [send]);
  const acknowledgeAll = useCallback(() => send({ action: "ack_all" }), [send]);
  const acknowledge = useCallback((id) => send({ action: "ack", id }), [send]);

  return {
    agents,
    stats,
    newAgents,
    isConnected,
    triggerScan,
    acknowledgeAll,
    acknowledge,
  };
}

📊 Compute Score


def compute_score(name):
    calls = metrics.calls[name]
    errors = metrics.errors[name]
    latencies = metrics.latency[name]

    if calls == 0:
        return 0.3  # unknown agent baseline

    usefulness = min(1.0, calls / 100)

    stability = 1.0 - (errors / max(calls, 1))

    speed = 1.0
    if latencies:
        avg_latency = sum(latencies) / len(latencies)
        speed = max(0.0, 1.0 - (avg_latency / 1000))

    reuse = min(1.0, calls / 50)

    score = (
        0.35 * usefulness +
        0.25 * stability +
        0.20 * speed +
        0.20 * reuse
    )

    return round(score, 3)

# server/core/control_plane.py
"""Policy engine + safety governance layer for the self-improving swarm."""

import time
from dataclasses import dataclass, field
from typing import Optional


@dataclass
class PolicyConfig:
    max_rewrites_per_hour: int = 3
    min_score_to_rewrite: float = 0.25
    never_touch: set = field(default_factory=lambda: {"gateway", "auth", "registry"})
    max_rewrites_per_agent_per_min: int = 1
    cooldown_seconds_after_rewrite: int = 120


class PolicyEngine:
    """Hard safety layer — sits ABOVE discovery / scoring / rewrite."""

    def __init__(self, config: Optional[PolicyConfig] = None):
        self.config = config or PolicyConfig()
        self._rewrite_timestamps: dict[str, list[float]] = {}

    def can_rewrite(self, agent: dict) -> bool:
        name = agent.get("name", "")
        if name in self.config.never_touch:
            return False
        score = agent.get("score", 0)
        if score < self.config.min_score_to_rewrite:
            return False
        now = time.time()
        recent = [t for t in self._rewrite_timestamps.get(name, []) if now - t < 3600]
        if len(recent) >= self.config.max_rewrites_per_hour:
            return False
        if recent and (now - recent[-1]) < self.config.cooldown_seconds_after_rewrite:
            return False
        return True

    def record_rewrite(self, agent_name: str):
        self._rewrite_timestamps.setdefault(agent_name, []).append(time.time())


class MetaControlPlane:
    """Orchestrates the full safety-governance layer above all subsystems."""

    def __init__(self, policy: Optional[PolicyEngine] = None):
        self.policy = policy or PolicyEngine()
        self._freeze = False

    def freeze(self):
        """Emergency brake — halts all evolution until thaw()."""
        self._freeze = True

    def thaw(self):
        self._freeze = False

    @property
    def is_frozen(self) -> bool:
        return self._freeze

    def guard_rewrite(self, agent: dict) -> bool:
        if self._freeze:
            return False
        return self.policy.can_rewrite(agent)
