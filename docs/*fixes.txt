3.1 RuntimeLoop.js
class RuntimeLoop {
  constructor() {
    this.lastTime = performance.now();
    this.subscribers = [];
    this.running = false;
  }

  subscribe(fn, priority = 0) {
    this.subscribers.push({ fn, priority });
    this.subscribers.sort((a, b) => a.priority - b.priority);
  }

  start() {
    if (this.running) return;
    this.running = true;
    requestAnimationFrame(this.tick);
  }

  stop() {
    this.running = false;
  }

  tick = (now) => {
    if (!this.running) return;

    const deltaTime = Math.min((now - this.lastTime) / 1000, 0.05);
    this.lastTime = now;

    for (const sub of this.subscribers) {
      sub.fn(deltaTime, now);
    }

    requestAnimationFrame(this.tick);
  };
}

function resolveExpression(emotion, intent, personality) {
  const expressed = { calm: emotion.calm, happy: emotion.happy, sad: emotion.sad, angry: emotion.angry, nervous: emotion.nervous };

  // Intent-based suppression
  if (intent.intent === "COMFORTING") {
    expressed.angry *= 0.2;
    expressed.nervous *= 0.4;
    expressed.calm = Math.max(expressed.calm, 0.6);
  }

  if (intent.intent === "SETTING_BOUNDARY") {
    expressed.calm *= 0.5;
    expressed.angry = Math.max(expressed.angry, 0.5);
  }

  if (intent.intent === "WITHDRAWING") {
    for (let k in expressed) expressed[k] *= 0.3;
  }

  // Personality leak-through
  const leak = 1 - personality.formality;
  for (let k in expressed) {
    expressed[k] = lerp(expressed[k], emotion[k], leak);
  }

  return expressed;
}

class RuntimeLoop {
  constructor() {
    this.lastTime = performance.now();
    this.accumulator = 0;
    this.subscribers = [];
    this.running = false;
    this.debug = false;
  }

  subscribe(fn, priority = 50, label = "anonymous") {
    this.subscribers.push({ fn, priority, label });
    this.subscribers.sort((a, b) => a.priority - b.priority);
  }

  unsubscribe(fn) {
    this.subscribers = this.subscribers.filter(s => s.fn !== fn);
  }

  start() {
    if (this.running) return;
    this.running = true;
    this.lastTime = performance.now();
    requestAnimationFrame(this.tick);
  }

  stop() {
    this.running = false;
  }

  tick = (now) => {
    if (!this.running) return;

    let delta = (now - this.lastTime) / 1000;
    this.lastTime = now;

    // Clamp to prevent spiral-of-death
    delta = Math.min(delta, 0.05);

    for (const sub of this.subscribers) {
      try {
        sub.fn(delta, now);
      } catch (err) {
        console.error(`[RuntimeLoop:${sub.label}]`, err);
      }
    }

    requestAnimationFrame(this.tick);
  };
}

function resolveVoiceProsody(intent, socialCost) {
  const base = INTENT_VOICE_BASE[intent.intent] || INTENT_VOICE_BASE.LISTENING;

  const fatigue = socialCost.fatigue;
  const suppression = socialCost.suppression;

  return {
    rate: clamp(base.rate * (1 - fatigue * 0.3), 0.7, 1.4),
    pitch: clamp(

      0.8,
      1.3

    pauseBias: clamp(
      base.pauseBias + suppression * 0.4,
      0,
      1
    ),
    onsetDelay: 150 + fatigue * 400 + suppression * 200,
  };
}

runtime/RuntimeLoop.js
class RuntimeLoop {
  constructor() {
    this.subscribers = [];
    this.running = false;
    this.lastTime = performance.now();
  }

  subscribe(fn, priority, label = "anon") {
    this.subscribers.push({ fn, priority, label });
    this.subscribers.sort((a, b) => a.priority - b.priority);
  }

  start() {
    if (this.running) return;
    this.running = true;
    this.lastTime = performance.now();
    requestAnimationFrame(this.tick);
  }

  tick = (now) => {
    if (!this.running) return;

    let dt = (now - this.lastTime) / 1000;
    this.lastTime = now;
    dt = Math.min(dt, 0.05);

    for (const s of this.subscribers) {
      try {
        s.fn(dt);
      } catch (e) {
        console.error(`[${s.label}]`, e);
      }
    }

    requestAnimationFrame(this.tick);
  };
}

export const runtime = new RuntimeLoop();

2️⃣ Gaze System (this is where realism explodes)

const gazeBias = {
  LISTENING:   { focus: 0.9, wander: 0.1 },
  COMFORTING: { focus: 0.7, wander: 0.2 },
  ASSERTIVE:  { focus: 0.95, wander: 0.05 },

Gaze update math (critical)

We use critically damped spring motion, not lerp.

θ'' + 2ζωθ' + ω²θ = ω²(target)


Where:

ζ = 1 (critical damping)

ω = gazeSpeed (intent + fatigue dependent)

Runtime system
let gaze = { x: 0, y: 0, vx: 0, vy: 0 };
let target = { x: 0, y: 0 };

runtime.subscribe((dt) => {
  const s = useStore.getState();
  const bias = gazeBias[s.socialIntent.intent];

  if (Math.random() < bias.wander * dt) {
    target.x = (Math.random() - 0.5) * 0.6;
    target.y = (Math.random() - 0.5) * 0.4;
  }

  const omega = 8 * (1 - s.socialCost.fatigue);

  // critically damped spring
  gaze.vx += (omega * omega * (target.x - gaze.x) - 2 * omega * gaze.vx) * dt;
  gaze.vy += (omega * omega * (target.y - gaze.y) - 2 * omega * gaze.vy) * dt;

  gaze.x += gaze.vx * dt;
  gaze.y += gaze.vy * dt;

  s.setAnimationSignal("gaze", gaze);
}, PRIORITY.ANIMATION, "GazeSystem");

  const s = useStore.getState();
