# server/infrastructure/rag_tracer.py
import time
import uuid

class RAGTracer:
    def __init__(self, query: str):
        self.trace_id = str(uuid.uuid4())
        self.start_time = time.time()

        self.data = {
            "trace_id": self.trace_id,
            "query": query,
            "intent": None,
            "steps": [],
            "chunks": [],
            "context": {},
            "metrics": {
                "total_latency": 0,
                "retrieval_confidence": 0,
                "hallucination_risk": 0
            }
        }

    def set_intent(self, intent: str):
        self.data["intent"] = intent

    def step(self, name: str, start_time: float, meta=None):
        self.data["steps"].append({
            "name": name,
            "latency": round((time.time() - start_time) * 1000, 2),
            "meta": meta or {}
        })

    def add_chunk(self, text, score, source, tokens):
        self.data["chunks"].append({
            "text": text[:300],
            "score": score,
            "source": source,
            "tokens": tokens
        })

    def set_context_stats(self, total_tokens, sources_used):
        self.data["context"] = {
            "total_tokens": total_tokens,
            "sources_used": sources_used
        }

    def finalize(self):
        self.data["metrics"]["total_latency"] = round(
            (time.time() - self.start_time) * 1000, 2
        )

        # simple heuristic
        scores = [c["score"] for c in self.data["chunks"]]
        if scores:
            self.data["metrics"]["retrieval_confidence"] = sum(scores) / len(scores)

        if self.data["metrics"]["retrieval_confidence"] < 0.5:
            self.data["metrics"]["hallucination_risk"] = 0.8

        return self.data

from infrastructure.rag_tracer import RAGTracer

async def run_research(query: str):
    tracer = RAGTracer(query)

    # STEP 1 — intent
    tracer.set_intent("web_research")

    # STEP 2 — search
    t0 = time.time()
    results = await search_web(query)
    tracer.step("search", t0, {"results": len(results)})

    # STEP 3 — extract + chunk
    t0 = time.time()
    chunks = extract_chunks(results)
    tracer.step("extract", t0, {"chunks": len(chunks)})

    # STEP 4 — embeddings
    t0 = time.time()
    embedded = embed_chunks(chunks)
    tracer.step("embed", t0)

    # STEP 5 — retrieval
    t0 = time.time()
    top_k = retrieve_top_k(embedded, query)
    tracer.step("retrieve", t0, {"k": len(top_k)})

    for c in top_k:
        tracer.add_chunk(c.text, c.score, c.source, c.tokens)

    tracer.set_context_stats(
        total_tokens=sum(c.tokens for c in top_k),
        sources_used=len(set(c.source for c in top_k))
    )

    return top_k, tracer

async def rewrite_query(query: str, llm):
    prompts = {
        "expand": f"Expand this query with context: {query}",
        "narrow": f"Make this query more specific: {query}",
        "keyword": f"Convert to search keywords: {query}",
        "semantic": f"Rewrite for semantic retrieval: {query}",
        "contrarian": f"Alternative perspective: {query}"
    }

    rewrites = []

    for k, p in prompts.items():
        r = await llm.generate(p)
        rewrites.append({
            "type": k,
            "query": r.strip()
        })

    return rewrites

function applyForces(nodes) {
  nodes.forEach(n => {
    nodes.forEach(m => {
      if (n !== m) {
        let dx = n.x - m.x;
        let dist = Math.sqrt(dx*dx) + 0.01;
        let force = 1 / dist;

        n.vx += dx * force * 0.001;
      }
    });
  });
}

def diagnose(issues, metrics):
    root_causes = []

    for issue in issues:
        candidates = CAUSE_MAP.get(issue, [])

        for c in candidates:
            if validate(c, metrics):
                root_causes.append(c)

    return root_causes
