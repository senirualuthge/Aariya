import torch
import torch.nn as nn
import numpy as np

class PolicyNet(nn.Module):
    def __init__(self, input_dim=4, output_dim=2):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(input_dim, 64),
            nn.ReLU(),
            nn.Linear(64, 64),
            nn.ReLU(),
            nn.Linear(64, output_dim)
        )

    def forward(self, x):
        return self.net(x)


class RLAgent:
    def __init__(self):
        self.policy = PolicyNet()

    def act(self, state):
        state = torch.tensor(state).float()
        actions = self.policy(state).detach().numpy()
        return actions

import numpy as np

class AnomalyEngine:
    def detect(self, state, prediction):
        anomalies = []

        drift = np.mean(np.abs(state[:, :2] - prediction))

        if drift > 2.0:
            anomalies.append({
                "type": "prediction_divergence",
                "severity": float(drift)
            })

        velocity_spike = np.std(state[:, 2:])

        if velocity_spike > 1.5:
            anomalies.append({
                "type": "instability",
                "severity": float(velocity_spike)
            })

        return anomalies

    def __init__(self, state, value=0):
        self.state = state
        self.value = value
        self.children = []
        self.action = None


class SearchTree:
    def __init__(self, world_model, policy, depth=3):
        self.world_model = world_model
        self.policy = policy
        self.depth = depth

    def expand(self, node):
        actions = self.policy.sample_actions(node.state)

        for a in actions:
            next_state = self.world_model(node.state, a)

            child = Node(next_state)
            child.action = a

            node.children.append(child)

import { useEffect, useState } from "react";

export function useSynopticsStream() {
  const [data, setData] = useState(null);

  useEffect(() => {
    const ws = new WebSocket("ws://localhost:8080");

    ws.onmessage = (msg) => {
      const parsed = JSON.parse(msg.data);
      setData(parsed);
    };

    return () => ws.close();
  }, []);

  return data;
}

🧠 6. PPO CORE UPDATE LOGIC
ppo.py
import torch
import torch.nn.functional as F

class PPO:
    def __init__(self, actor, lr=3e-4):
        self.actor = actor
        self.optimizer = torch.optim.Adam(actor.parameters(), lr=lr)

    def update(self, obs, actions, advantages):
        pred = self.actor(obs)

        log_prob = -((pred - actions) ** 2)

        loss = -(log_prob * advantages).mean()

        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()

class RewardShaper:

        return shaped_reward

from collections import deque
import numpy as np

class TemporalMemory:
    def __init__(self, max_len=200):
        self.history = deque(maxlen=max_len)
        self.embeddings = deque(maxlen=max_len)

    def store(self, state, embedding):
        self.history.append(state)
        self.embeddings.append(embedding)

    def get_sequence(self):
        return list(self.history)

    def compute_drift(self):
        if len(self.embeddings) < 2:
            return 0

        diffs = []
        for i in range(1, len(self.embeddings)):
            diffs.append(
                np.linalg.norm(
                    self.embeddings[i] - self.embeddings[i - 1]
                )
            )

        return float(np.mean(diffs))

const wss = new WebSocketServer({ server });

const redisPub = new Redis();

const CHANNEL = "synoptics_stream";

// Subscribe to Python engine updates
redisSub.subscribe(CHANNEL);

redisSub.on("message", (channel, message) => {
  if (channel === CHANNEL) {
    broadcast(JSON.parse(message));
  }
});

// WebSocket clients
const clients = new Set();

wss.on("connection", (ws) => {
  clients.add(ws);

  ws.on("message", (msg) => {
    const data = JSON.parse(msg);

    // optional: forward commands to Python engine
    redisPub.publish("synoptics_commands", JSON.stringify(data));
  });

  ws.on("close", () => {
    clients.delete(ws);
  });
});

function broadcast(data) {
  const payload = JSON.stringify(data);
  for (const client of clients) {
    if (client.readyState === 1) {
      client.send(payload);
    }
  }
}

server.listen(8080, () => {
  console.log("Synoptics Gateway running on :8080");

from sklearn.cluster import KMeans

class IntentDetector:
    def __init__(self, k=5):
        self.kmeans = KMeans(n_clusters=k)

    def build_features(self, agents):
        return np.array([
            [a["vx"], a["vy"], a["x"], a["y"]]
            for a in agents
        ])

    def detect(self, agents):
        X = self.build_features(agents)

class ShockwaveEngine:
    def __init__(self, decay=0.85):
        self.decay = decay

    def propagate(self, adjacency, initial_impact, steps=5):
        impact = initial_impact

        history = [impact]

        for _ in range(steps):
            impact = self.decay * adjacency @ impact
            history.append(impact)

        return history

import numpy as np

class SwarmStabilizer:
    def __init__(self, lambda_=0.1):
        self.lambda_ = lambda_

    def compute_entropy(self, positions):
        var = np.var(positions, axis=0)
        return np.sum(var)

    def stabilize(self, agents):
        positions = np.array([[a["x"], a["y"]] for a in agents])

        entropy = self.compute_entropy(positions)

        correction = -self.lambda_ * entropy

        for a in agents:
            a["vx"] += correction * 0.01
            a["vy"] += correction * 0.01

        return agents

import numpy as np
from collections import deque

class NarrativeMemory:
    def __init__(self, size=100):
        self.history = deque(maxlen=size)

    def add_state(self, state):
        self.history.append(state)

    def drift_score(self):
        if len(self.history) < 2:
            return 0

        prev = self.history[-2]
        curr = self.history[-1]

        prev_mean = np.mean([a["x"] for a in prev])
        curr_mean = np.mean([a["x"] for a in curr])

        return abs(curr_mean - prev_mean)

import numpy as np

def reward(agent, state):
    survival_bonus = 1.0 - 0.1 * np.linalg.norm(agent)
    distance_to_goal = -np.linalg.norm(agent[:2] - np.array([0.0, 0.0]))
    collision_penalty = 0.0
    for other in state:
        if np.linalg.norm(agent[:2] - other[:2]) < 0.5:
            collision_penalty += 0.5
    return survival_bonus + distance_to_goal - collision_penalty

import torch
import torch.optim as optim

def ppo_update(policy, optimizer, states, actions, advantages, clip_eps=0.2):
    old_probs = policy(states).gather(1, actions)
    new_probs = policy(states).gather(1, actions)
    ratio = new_probs / (old_probs + 1e-8)
    clipped = torch.clamp(ratio, 1 - clip_eps, 1 + clip_eps)
    loss = -torch.min(ratio * advantages, clipped * advantages).mean()
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()
    return loss.item()

import numpy as np

class DecisionTrace:
    def compute_scores(self, futures):
        scores = []

        for f in futures:
            stability = -np.var(f)
            compactness = -np.mean(np.linalg.norm(f, axis=1))
            volatility = np.std(f)

            score = stability + compactness - volatility

            scores.append(score)

        return scores

import numpy as np

class CounterfactualAnalysis:
    def compare(self, factual, counterfactuals, chosen_idx):
        results = []

        factual_score = self.score(factual)

        for i, cf in enumerate(counterfactuals):
            cf_score = self.score(cf)

            results.append({
                "action_id": i,
                "counterfactual_score": float(cf_score),
                "delta_vs_factual": float(cf_score - factual_score),
                "regret": float(factual_score - cf_score)
            })

        return results

    def score(self, state):
        stability = -np.var(state)
        compactness = -np.mean(np.linalg.norm(state, axis=1))

import torch.nn.functional as F

class WorldModelTrainer:
    def __init__(self, model, lr=1e-3):
        self.model = model
        self.opt = torch.optim.Adam(model.parameters(), lr=lr)

    def train_step(self, state, action, real_next_state):
        pred_next = self.model(state, action)

        loss = F.mse_loss(pred_next, real_next_state)

        self.opt.zero_grad()
        loss.backward()
        self.opt.step()

        return float(loss.item())

import ray
import numpy as np

@ray.remote(num_gpus=0.25)
class SearchWorker:
    def __init__(self, world_model):
        self.world_model = world_model

    def rollout(self, state, policy, depth=4):
        s = state
        total_reward = 0

        for _ in range(depth):
            action = policy(s)
            s = self.world_model(s, action)
            total_reward += -np.var(s)

        return total_reward

import torch
import numpy as np

class WorldModelSelector:
    def evaluate(self, model, dataset):
        loss = 0

        for s, a, next_s in dataset:
            pred = model(s, a)
            loss += ((pred - next_s) ** 2).mean().item()

        return -loss  # higher is better

    def select_best(self, candidates, dataset):
        scores = [self.evaluate(m, dataset) for m in candidates]
        best_idx = int(np.argmax(scores))

        return candidates[best_idx], scores

class PolicySelector:
    def evaluate(self, policy, search_env):
        total_reward = 0

        for state in search_env:
            action_dist = policy(state)
            total_reward += action_dist.max().item()

        return total_reward

    def select(self, policies, env):
        scores = [self.evaluate(p, env) for p in policies]
        return policies[int(max(range(len(scores)), key=lambda i: scores[i]))]

import copy
import random
import torch.nn as nn

class ModelMutator:

    def mutate(self, model):
        new_model = copy.deepcopy(model)

        mutation_type = random.choice([
            "add_layer",
            "remove_layer",
            "widen_layer",
            "change_activation"
        ])

        if mutation_type == "add_layer":
            self._add_layer(new_model)

        elif mutation_type == "widen_layer":
            self._widen_layer(new_model)

        elif mutation_type == "change_activation":
            self._change_activation(new_model)

        return new_model

    def _add_layer(self, model):
        model.net = nn.Sequential(
            model.net,
            nn.ReLU(),
            nn.Linear(64, 64)
        )

    def _widen_layer(self, model):
        old_out = model.net[-1].out_features
        new_out = old_out * 2
        model.net[-1] = nn.Linear(model.net[-1].in_features, new_out)
        model.net.add_module("expand", nn.Linear(new_out, old_out))

    def _change_activation(self, model):
        for i, layer in enumerate(model.net):
            if isinstance(layer, nn.ReLU):
                model.net[i] = nn.GELU()

📦 gating.py
class PerformanceGate:

    def __init__(self, min_improvement=0.01, max_loss=1.0):
        self.min_improvement = min_improvement
        self.max_loss = max_loss

    def approve(self, base_score, new_score, stability_score):
        improvement = new_score - base_score

        if new_score < -self.max_loss:
            return False

        if stability_score < 0.5:
            return False

        if improvement < self.min_improvement:
            return False

        return True

import numpy as np

class ArchitectureEvaluator:

    def evaluate(self, model, dataset):
        loss = 0

        for state, action, target in dataset:
            pred = model(state, action)
            loss += ((pred - target) ** 2).mean().item()

        stability = 1.0 / (loss + 1e-6)

        return {
            "loss": loss,
            "stability": stability,
            "score": -loss + stability * 0.1
        }
