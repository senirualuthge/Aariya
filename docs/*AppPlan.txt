from fastapi import FastAPI, WebSocket
import json

app = FastAPI()

@app.websocket("/ws")
async def websocket_endpoint(ws: WebSocket):
    await ws.accept()
    print("Client connected")

    try:
        while True:
            data = await ws.receive_text()
            msg = json.loads(data)

            if msg["type"] == "user_text":
                response = {
                    "type": "ai_response",
                    "text": f"I heard: {msg['text']}",
                    "emotion": "calm",
                    "status": {
                        "listening": False,
                        "thinking": False,
                        "speaking": True
                    }
                }
                await ws.send_text(json.dumps(response))

    except Exception as e:
        print("Client disconnected", e)

import 'dart:convert';
import 'dart:typed_data';
import 'package:flutter_sound/flutter_sound.dart';
import 'websocket_service.dart';

class AudioService {
  final FlutterSoundRecorder _recorder = FlutterSoundRecorder();
  bool _isRecording = false;

  Future<void> init() async {
    await _recorder.openRecorder();
    await _recorder.setSubscriptionDuration(
      const Duration(milliseconds: 20),
    );
  }

  Future<void> startRecording() async {
    if (_isRecording) return;

    _isRecording = true;

    await _recorder.startRecorder(
      toStream: (buffer) {
        final audioBytes = Uint8List.fromList(buffer);
        WebSocketService.instance.sendBinary(
          audioBytes,
          type: "audio_chunk",
        );
      },
      codec: Codec.pcm16,
      sampleRate: 16000,
      numChannels: 1,
    );

    WebSocketService.instance.send({
      "type": "audio_start",
    });
  }

  Future<void> stopRecording() async {
    if (!_isRecording) return;

    await _recorder.stopRecorder();
    _isRecording = false;

    WebSocketService.instance.send({
      "type": "audio_end",
    });
  }
}

🖥️ Python WebSocket Server (FastAPI)
from fastapi import FastAPI, WebSocket
import numpy as np
import json

app = FastAPI()

audio_buffer = bytearray()

@app.websocket("/ws")
async def ws(ws: WebSocket):
    await ws.accept()

    try:
        while True:
            msg = await ws.receive()

            if "text" in msg:
                data = json.loads(msg["text"])

                if data["type"] == "audio_start":
                    audio_buffer.clear()

                if data["type"] == "audio_end":
                    pcm = np.frombuffer(audio_buffer, dtype=np.int16)

                    import whisper
                    model = whisper.load_model("base")
                    result = model.transcribe(pcm.astype(np.float32) / 32768.0, sample_rate=16000)
                    text = result["text"].strip()

                    await ws.send_text(json.dumps({
                        "type": "ai_text",
                        "text": text
                    }))

            if "bytes" in msg:
                audio_buffer.extend(msg["bytes"])

    except Exception as e:
        print("Disconnected:", e)

receives live audio

import 'dart:typed_data';
import 'package:flutter_sound/flutter_sound.dart';

class AudioPlayerService {
  final FlutterSoundPlayer _player = FlutterSoundPlayer();
  bool ready = false;

  Future<void> init() async {
    await _player.openPlayer();
    ready = true;
  }

  void playChunk(Uint8List bytes) {
    if (!ready) return;

    _player.startPlayerFromStream(
      codec: Codec.pcm16,
      sampleRate: 16000,
      numChannels: 1,

@app.post("/connect")
def create_session(device_id: str):
    session_id = str(uuid.uuid4())
    token = str(uuid.uuid4())
    session = Session(session_id=session_id, device_id=device_id, token=token, role="owner")
    sessions[session_id] = session

@app.post("/message")
def receive_message(payload: Dict[str, Any]):
    """
    Unified entrypoint: validates incoming JSON and routes to handler.
    """
    try:
        msg_type = payload.get("type")
        if msg_type == "user_input":
            user_input = UserInput(**payload)
            # Here we process input and generate AI response
            response_text = f"Received your input: {user_input.content}"
            ai_resp = create_ai_response(response_text)
            return ai_resp.dict()

        elif msg_type.startswith("avatar."):
            avatar_cmd = AvatarCommand(**payload)
            # Normally, trigger avatar animation/audio
            return {"status": "avatar command received", "type": avatar_cmd.type}

        else:
            raise HTTPException(status_code=400, detail="Unknown message type")

    except ValidationError as e:
        raise HTTPException(status_code=422, detail=e.errors())

@app.get("/trust/{session_id}")
def get_trust(session_id: str):
    # Example trust calculation
    trust = TrustStatus(trust_level=0.8, expressiveness=0.7, personality_mode="core")
    return trust.dict()

@app.post("/memory")
def add_memory(payload: Dict[str, Any]):
    try:
        memory = MemoryEntry(**payload)
        memories[memory.memory_id] = memory
        return {"status": "memory saved", "memory_id": memory.memory_id}
    except ValidationError as e:
        raise HTTPException(status_code=422, detail=e.errors())

async def send_ai_response(ws: WebSocket, text: str):
    """
    Sends incremental AI responses (simulate streaming)
    """
    words = text.split()
    partial = ""
    for i, word in enumerate(words):
        partial += word + " "
        ai_resp = AIResponse(
            type="ai_response",
            text=partial.strip(),
            status={"thinking": True, "listening": False},
            timestamp=int(asyncio.get_event_loop().time())
        )
        await ws.send_json(ai_resp.dict())
        await asyncio.sleep(0.2)  # simulate streaming

import httpx
from app.config import settings

async def generate_response(user_id: str, text: str):

    payload = {
        "model": settings.MODEL_NAME,
        "messages": [
            {"role": "system", "content": "You are an emotionally intelligent AI companion."},
            {"role": "user", "content": text}
        ]
    }

    async with httpx.AsyncClient() as client:
        r = await client.post(
            settings.LLM_ENDPOINT,
            headers={"Authorization": f"Bearer {settings.LLM_API_KEY}"},
            json=payload,
            timeout=30
        )

    data = r.json()

    return {
        "type": "ai_response",
        "message": data["choices"][0]["message"]["content"]
    }

🔹 train.py (QLoRA Setup)
import torch
from transformers import AutoModelForCausalLM, AutoTokenizer
from peft import LoraConfig, get_peft_model
from datasets import load_dataset
from transformers import Trainer, TrainingArguments

model_name = "mistralai/Mistral-7B-v0.1"

tokenizer = AutoTokenizer.from_pretrained(model_name)

model = AutoModelForCausalLM.from_pretrained(
    model_name,
    load_in_4bit=True,
    device_map="auto"
)

lora_config = LoraConfig(
    r=16,
    lora_alpha=32,
    target_modules=["q_proj","v_proj"],
    lora_dropout=0.05,
    bias="none",
    task_type="CAUSAL_LM"
)

model = get_peft_model(model, lora_config)

dataset = load_dataset("json", data_files="data/processed/train.json")

training_args = TrainingArguments(
    output_dir="./outputs",
    per_device_train_batch_size=4,
    gradient_accumulation_steps=4,
    num_train_epochs=3,
    fp16=True,
    logging_steps=20,
    save_strategy="epoch"
)

trainer = Trainer(
    model=model,
    args=training_args,
    train_dataset=dataset["train"]
)

trainer.train()
