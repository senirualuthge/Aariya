    Brier score for probabilities
    calibration error
    hit rate
    information coefficient
    maximum drawdown
    Sharpe ratio only when strategy simulation is appropriate
    turnover
    transaction-cost sensitivity

Compare against naive baselines.

A model is not promoted because it has a high in-sample score.

    calibration
    no material leakage
    acceptable drift

2. Source registry

Aariya needs to know where information is allowed to come from.

# server/learning/acquisition/registry.py

from threading import RLock
from .source import DataSource


class SourceRegistry:
    def __init__(self) -> None:
        self._sources: dict[str, DataSource] = {}
        self._lock = RLock()

    def register(self, source: DataSource) -> None:
        if not source.id:
            raise ValueError("Source ID is required")

        with self._lock:
            if source.id in self._sources:
                raise ValueError(
                    f"Source already registered: {source.id}"
                )

            self._sources[source.id] = source

    def get(self, source_id: str) -> DataSource:
        with self._lock:
            try:
                return self._sources[source_id]
            except KeyError:
                raise KeyError(
                    f"Unknown data source: {source_id}"
                )

    def enabled(self) -> list[DataSource]:
        with self._lock:
            return [
                source
                for source in self._sources.values()
                if source.enabled
            ]

No fake sources should be registered.

5. Validation

Don't let garbage reach training.

# server/learning/acquisition/validator.py

from dataclasses import dataclass


@dataclass(frozen=True)
class ValidationResult:
    valid: bool
    errors: tuple[str, ...]


class DataValidator:

    def validate_payload(
        self,
        payload: bytes,
    ) -> ValidationResult:

        errors: list[str] = []

        if not payload:
            errors.append("Empty payload")

        if len(payload) > 500 * 1024 * 1024:
            errors.append("Payload exceeds maximum size")

        return ValidationResult(
            valid=not errors,
            errors=tuple(errors),
        )

6. Retry handling

Retries should happen only for recoverable failures.

# server/learning/acquisition/retry.py

import asyncio
from collections.abc import Awaitable, Callable
from typing import TypeVar

T = TypeVar("T")


class RetryPolicy:

    def __init__(
        self,
        attempts: int = 3,
        initial_delay: float = 1.0,
        max_delay: float = 30.0,
    ):
        self.attempts = attempts
        self.initial_delay = initial_delay
        self.max_delay = max_delay

    async def execute(
        self,
        operation: Callable[[], Awaitable[T]],
    ) -> T:

        last_error: Exception | None = None
        delay = self.initial_delay

        for attempt in range(self.attempts):
            try:
                return await operation()

            except Exception as exc:
                last_error = exc

                if attempt == self.attempts - 1:
                    break

                await asyncio.sleep(delay)
                delay = min(delay * 2, self.max_delay)

        raise RuntimeError(
            "Operation failed after retries"
        ) from last_error
7. Raw data store

Raw data should never be overwritten.

# server/learning/datasets/raw_store.py

from dataclasses import dataclass
from datetime import datetime
from pathlib import Path


@dataclass(frozen=True)
class RawObject:
    object_id: str
    path: Path
    checksum: str
    created_at: datetime


class RawDataStore:

    def __init__(self, root: Path):
        self.root = root
        self.root.mkdir(
            parents=True,
            exist_ok=True,
        )

    def write(
        self,
        object_id: str,
        data: bytes,
    ) -> RawObject:

        path = self.root / f"{object_id}.bin"

        if path.exists():
            raise FileExistsError(
                f"Raw object already exists: {object_id}"
            )

        path.write_bytes(data)

        import hashlib

        checksum = hashlib.sha256(data).hexdigest()

        return RawObject(
            object_id=object_id,
            path=path,
            checksum=checksum,
            created_at=datetime.utcnow(),
        )

That gives us an immutable audit trail.

9. Train/validation/test splitting

For financial data, normal random splitting is dangerous because it can leak future information into training.

# server/learning/datasets/splitter.py

from dataclasses import dataclass


@dataclass(frozen=True)
class TimeSplit:
    train_start: int
    train_end: int
    validation_start: int
    validation_end: int
    test_start: int
    test_end: int


class TemporalSplitter:

    def split(
        self,
        total_rows: int,
        train_ratio: float = 0.70,
        validation_ratio: float = 0.15,
    ) -> TimeSplit:

        train_end = int(total_rows * train_ratio)

        validation_end = train_end + int(
            total_rows * validation_ratio
        )

        return TimeSplit(
            train_start=0,
            train_end=train_end,
            validation_start=train_end,
            validation_end=validation_end,
            test_start=validation_end,
            test_end=total_rows,
        )

14. Walk-forward evaluation

For your stock engine, this is mandatory.

Conceptually:

TRAIN
████████████

VALIDATE
            ███

TEST
               ███

Then move forward:

TRAIN
███████████████

VALIDATE
               ███

TEST
                  ███

And repeat.

# server/learning/evaluation/walk_forward.py

from dataclasses import dataclass


@dataclass(frozen=True)
class Window:
    train_start: int
    train_end: int
    test_start: int
    test_end: int


class WalkForwardGenerator:

    def generate(
        self,
        total_rows: int,
        train_size: int,
        test_size: int,
        step: int,
    ) -> list[Window]:

        windows = []

        start = 0

        while start + train_size + test_size <= total_rows:

            train_end = start + train_size
            test_end = train_end + test_size

            windows.append(
                Window(
                    train_start=start,
                    train_end=train_end,
                    test_start=train_end,
                    test_end=test_end,
                )
            )

            start += step

        return windows
15. Promotion engine

A candidate model should have to beat the current production model.

# server/learning/evaluation/promotion.py

from dataclasses import dataclass


@dataclass(frozen=True)
class EvaluationResult:
    score: float
    baseline_score: float
    statistically_significant: bool
    drift_detected: bool


class PromotionEngine:

    def should_promote(
        self,
        result: EvaluationResult,
    ) -> bool:

        if result.drift_detected:
            return False

        if not result.statistically_significant:
            return False

        return result.score > result.baseline_score

16. Autonomous learning policy

This is where we enforce your safe/controlled boundary.

# server/learning/autonomous/policy.py

from enum import Enum


class OperationClass(Enum):

    AUTONOMOUS = "autonomous"
    CONTROLLED = "controlled"


AUTONOMOUS_OPERATIONS = {
    "collect_data",
    "clean_data",
    "create_dataset",
    "generate_features",
    "run_experiment",
    "train_model",
    "evaluate_model",
    "record_outcome",
    "identify_knowledge_gap",
}


CONTROLLED_OPERATIONS = {
    "promote_model",
    "change_decision_policy",
    "change_security_policy",
    "delete_knowledge",
    "modify_architecture",
    "change_permissions",
}


def classify(operation: str) -> OperationClass:

    if operation in AUTONOMOUS_OPERATIONS:
        return OperationClass.AUTONOMOUS

    if operation in CONTROLLED_OPERATIONS:
        return OperationClass.CONTROLLED

    raise ValueError(
        f"Unknown autonomous operation: {operation}"
    )

Aariya therefore can't just invent an operation called:

disable_security()

and execute it because it happens to be running inside the learning engine.

17. The autonomous orchestrator

This connects everything.

# server/learning/autonomous/orchestrator.py

from dataclasses import dataclass
from enum import Enum


class LearningStage(Enum):

    DISCOVER = "discover"
    COLLECT = "collect"
    VALIDATE = "validate"
    STORE = "store"
    PROCESS = "process"
    TRAIN = "train"
    EVALUATE = "evaluate"
    COMPARE = "compare"
    PROMOTE = "promote"
    OBSERVE = "observe"
    LEARN = "learn"


@dataclass
class LearningJob:

    job_id: str
    objective: str
    stage: LearningStage
    status: str = "pending"


class LearningOrchestrator:

    def __init__(
        self,
        collector,
        validator,
        trainer,
        evaluator,
        experience_memory,
        policy,
    ):
        self.collector = collector
        self.validator = validator
        self.trainer = trainer
        self.evaluator = evaluator
        self.experience_memory = experience_memory
        self.policy = policy

    async def execute(
        self,
        job: LearningJob,
    ):

        self._transition(
            job,
            LearningStage.DISCOVER,
        )

        await self._discover(job)

        self._transition(
            job,
            LearningStage.COLLECT,
        )

        data = await self.collector.collect()

        self._transition(
            job,
            LearningStage.VALIDATE,
        )

        validation = self.validator.validate(data)

        if not validation.valid:
            job.status = "failed"
            raise ValueError(
                validation.errors
            )

        self._transition(
            job,
            LearningStage.STORE,
        )

        dataset = await self._store(data)

        self._transition(
            job,
            LearningStage.PROCESS,
        )

        processed = await self._process(dataset)

        self._transition(
            job,
            LearningStage.TRAIN,
        )

        model = await self.trainer.train(
            processed
        )

        self._transition(
            job,
            LearningStage.EVALUATE,
        )

        evaluation = await self.evaluator.evaluate(
            model,
            processed,
        )

        self._transition(
            job,
            LearningStage.COMPARE,
        )

        approved = await self._compare(
            evaluation
        )

        if approved:
            self._transition(
                job,
                LearningStage.PROMOTE,
            )

            await self._promote(model)

        job.status = "completed"

        return {
            "job_id": job.job_id,
            "status": job.status,
            "promoted": approved,
        }

    def _transition(
        self,
        job: LearningJob,
        stage: LearningStage,
    ):
        job.stage = stage

    async def _discover(self, job):
        pass

    async def _store(self, data):
        return data

    async def _process(self, dataset):
        return dataset

    async def _compare(self, evaluation):
        return False

    async def _promote(self, model):
        raise PermissionError(
            "Production promotion requires controlled authorization"
        )

Notice the last part.

The learning engine can train a candidate.

It cannot simply decide:

"I trained a new model, therefore I am allowed to replace production."

That goes through the authorization system we already designed for Aariya.

18. Task generation

Eventually Aariya shouldn't just run the same training job forever.

It should identify what it needs to investigate.

# server/learning/autonomous/task_generator.py

from dataclasses import dataclass


@dataclass(frozen=True)
class LearningTask:
    objective: str
    reason: str
    priority: float


class LearningTaskGenerator:

    def generate(
        self,
        drift_detected: bool,
        prediction_error: float,
        data_quality: float,
    ) -> list[LearningTask]:

        tasks: list[LearningTask] = []

        if drift_detected:
            tasks.append(
                LearningTask(
                    objective="investigate_feature_drift",
                    reason="Feature distribution changed",
                    priority=1.0,
                )
            )

        if abs(prediction_error) > 0.10:
            tasks.append(
                LearningTask(
                    objective="investigate_prediction_error",
                    reason="Prediction error exceeded threshold",
                    priority=0.9,
                )
            )

        if data_quality < 0.95:
            tasks.append(
                LearningTask(
                    objective="investigate_data_quality",
                    reason="Data quality degraded",
                    priority=0.95,
                )
            )

        return sorted(
            tasks,
            key=lambda task: task.priority,
            reverse=True,
        )

This gives you the beginning of self-directed learning.

3. Rate limiter

Aariya shouldn't hammer APIs.

# server/learning/acquisition/rate_limit.py

import asyncio
import time


class RateLimiter:

    def __init__(
        self,
        requests_per_minute: int,
    ):
        if requests_per_minute <= 0:
            raise ValueError(
                "requests_per_minute must be positive"
            )

        self.interval = 60.0 / requests_per_minute
        self._lock = asyncio.Lock()
        self._last_request = 0.0

    async def acquire(self):

        async with self._lock:

            now = time.monotonic()

            wait = (
                self.interval
                - (now - self._last_request)
            )

            if wait > 0:
                await asyncio.sleep(wait)

            self._last_request = time.monotonic()

For production, we'll eventually move distributed rate limiting into Redis so multiple Aariya workers share the same limits.

4. Retry with HTTP awareness

We shouldn't retry every error.

# server/learning/acquisition/retry.py

import asyncio
import random

import httpx


RETRYABLE_STATUS_CODES = {
    408,
    425,
    429,
    500,
    502,
    503,
    504,
}


async def request_with_retry(
    client: httpx.AsyncClient,
    method: str,
    url: str,
    *,
    attempts: int = 5,
    **kwargs,
) -> httpx.Response:

    last_error = None

    for attempt in range(attempts):

        try:

            response = await client.request(
                method,
                url,
                **kwargs,
            )

            if (
                response.status_code
                not in RETRYABLE_STATUS_CODES
            ):
                return response

            last_error = RuntimeError(
                f"HTTP {response.status_code}"
            )

        except (
            httpx.TimeoutException,
            httpx.NetworkError,
        ) as exc:

            last_error = exc

        if attempt == attempts - 1:
            break

        delay = min(
            60.0,
            2 ** attempt + random.random(),
        )

        await asyncio.sleep(delay)

    raise RuntimeError(
        f"Request failed after {attempts} attempts"
    ) from last_error

This prevents a temporary outage from killing the learning cycle.

import hashlib

from datetime import datetime, timezone

from .collector import CollectedData
from .rate_limit import RateLimiter

class AcquisitionService:

        self,
        collector,

        self.validator = validator
        self.rate_limiter = rate_limiter

    async def acquire(
        self,

            resource=data.resource,
            payload=data.payload,
            content_type=data.content_type,

            retrieved_at=data.retrieved_at,

7. Immutable raw storage

Don't put every raw payload directly into PostgreSQL.

Use PostgreSQL for metadata and object storage/filesystem for large immutable objects.

# server/learning/datasets/raw_store.py

import hashlib
from pathlib import Path


class RawObjectStore:

    def __init__(self, root: Path):

        self.root = root

        self.root.mkdir(
            parents=True,
            exist_ok=True,
        )

    def put(
        self,
        payload: bytes,
    ) -> tuple[str, Path]:

        checksum = hashlib.sha256(
            payload
        ).hexdigest()

        directory = self.root / checksum[:2]

        directory.mkdir(
            parents=True,
            exist_ok=True,
        )

        path = directory / checksum

        if path.exists():

            return checksum, path

        temporary = path.with_suffix(".tmp")

        temporary.write_bytes(payload)

        temporary.replace(path)

        return checksum, path

That gives us content-addressed storage:

├── 01/
│   └── 01af...
├── 23/
│   └── 23d8...
├── 7a/
│   └── 7a92...
└── ...

Identical data automatically resolves to the same object.

8. Scheduled collection

Aariya needs to wake itself up.

# server/learning/acquisition/scheduler.py

import asyncio
import logging

logger = logging.getLogger(__name__)


class CollectionScheduler:

    def __init__(self):

        self._jobs: list[asyncio.Task] = []
        self._running = False

    async def start(self):

        self._running = True

        logger.info(
            "Learning collection scheduler started"
        )

    async def stop(self):

        self._running = False

        for task in self._jobs:
            task.cancel()

        self._jobs.clear()

    async def every(
        self,
        interval_seconds: int,
        operation,
    ):

        while self._running:

            try:
                await operation()

            except asyncio.CancelledError:
                raise

            except Exception:
                logger.exception(
                    "Scheduled collection failed"
                )

            await asyncio.sleep(
                interval_seconds
            )

For the actual production version, I recommend using your existing infrastructure with a persistent scheduler/worker rather than relying solely on an in-process loop. That prevents a server restart from losing scheduled jobs.

Aariya shouldn't just say "data received."

# server/learning/datasets/quality.py

from dataclasses import dataclass


@dataclass(frozen=True)
class QualityReport:

    row_count: int

    missing_values: int

    duplicate_rows: int

    invalid_rows: int

    earliest_timestamp: str | None

    latest_timestamp: str | None

    quality_score: float


def calculate_quality(
    row_count: int,
    missing_values: int,
    duplicate_rows: int,
    invalid_rows: int,
) -> QualityReport:

    if row_count <= 0:
        return QualityReport(
            row_count=0,
            missing_values=0,
            duplicate_rows=0,
            invalid_rows=0,
            earliest_timestamp=None,
            latest_timestamp=None,
            quality_score=0.0,
        )

    problems = (
        missing_values
        + duplicate_rows
        + invalid_rows
    )

    quality = max(
        0.0,
        1.0 - (problems / row_count),
    )

    return QualityReport(
        row_count=row_count,
        missing_values=missing_values,
        duplicate_rows=duplicate_rows,
        invalid_rows=invalid_rows,
        earliest_timestamp=None,
        latest_timestamp=None,
        quality_score=quality,
    )

For financial data we'll make the quality engine domain-specific rather than using only a generic score.

# server/learning/datasets/financial_validation.py

from dataclasses import dataclass


@dataclass(frozen=True)
class OHLCV:
    timestamp: int
    open: float
    high: float
    low: float
    close: float
    volume: float


def validate_ohlcv(row: OHLCV) -> list[str]:

    errors = []

    values = (
        row.open,
        row.high,
        row.low,
        row.close,
        row.volume,
    )

    if any(
        value != value
        for value in values
    ):
        errors.append("NaN detected")

    if row.volume < 0:
        errors.append(
            "Negative volume"
        )

    if row.high < max(
        row.open,
        row.close,
        row.low,
    ):
        errors.append(
            "Invalid high price"
        )

    if row.low > min(
        row.open,
        row.close,
        row.high,
    ):
        errors.append(
            "Invalid low price"
        )

    if row.open <= 0:
        errors.append(
            "Invalid open price"
        )

    if row.high <= 0:
        errors.append(
            "Invalid high price"
        )

    if row.low <= 0:
        errors.append(
            "Invalid low price"
        )

    if row.close <= 0:
        errors.append(
            "Invalid close price"
        )

    return errors

# server/learning/acquisition/providers/http_market_data.py

from datetime import datetime
from decimal import Decimal

import httpx

from ..provider import (
    MarketDataProvider,
    PriceBar,
)


class HTTPMarketDataProvider(MarketDataProvider):

    def __init__(
        self,
        base_url: str,
        api_key: str,
    ):
        self.base_url = base_url.rstrip("/")
        self.api_key = api_key

    async def historical_prices(
        self,
        symbol: str,
        start: datetime,
        end: datetime,
    ) -> list[PriceBar]:

        params = {
            "symbol": symbol,
            "start": start.isoformat(),
            "end": end.isoformat(),
        }

        headers = {
            "Authorization":
                f"Bearer {self.api_key}"
        }

        async with httpx.AsyncClient(
            timeout=30
        ) as client:

            response = await client.get(
                self.base_url,
                params=params,
                headers=headers,
            )

            response.raise_for_status()

            payload = response.json()

        return self._parse(payload, symbol)

    def _parse(
        self,
        payload: dict,
        symbol: str,
    ) -> list[PriceBar]:

        records = payload["data"]

        return [
            PriceBar(
                symbol=symbol,
                timestamp=datetime.fromisoformat(
                    item["timestamp"]
                ),
                open=Decimal(str(item["open"])),
                high=Decimal(str(item["high"])),
                low=Decimal(str(item["low"])),
                close=Decimal(str(item["close"])),
                volume=Decimal(str(item["volume"])),
            )
            for item in records
        ]

# server/learning/datasets/financial_validation.py

from decimal import Decimal

from ..acquisition.provider import PriceBar


def validate_price_bar(
    bar: PriceBar,
) -> list[str]:

    errors: list[str] = []

    prices = {
        "open": bar.open,
        "high": bar.high,
        "low": bar.low,
        "close": bar.close,
    }

    for name, value in prices.items():

        if value <= Decimal("0"):
            errors.append(
                f"{name} must be positive"
            )

    if bar.volume < Decimal("0"):
        errors.append(
            "volume cannot be negative"
        )

    if bar.high < max(
        bar.open,
        bar.close,
        bar.low,
    ):
        errors.append(
            "high is inconsistent"
        )

    if bar.low > min(
        bar.open,
        bar.close,
        bar.high,
    ):
        errors.append(
            "low is inconsistent"
        )

    return errors
Batch validation
# server/learning/datasets/financial_validator.py

from dataclasses import dataclass

from ..acquisition.provider import PriceBar
from .financial_validation import validate_price_bar


@dataclass(frozen=True)
class ValidationReport:

    total: int
    valid: int
    invalid: int
    errors: tuple[str, ...]


def validate_price_bars(
    bars: list[PriceBar],
) -> ValidationReport:

    errors = []
    valid = 0

    for index, bar in enumerate(bars):

        row_errors = validate_price_bar(bar)

        if row_errors:

            errors.extend(
                f"row={index}: {error}"
                for error in row_errors
            )

        else:
            valid += 1

    return ValidationReport(
        total=len(bars),
        valid=valid,
        invalid=len(bars) - valid,
        errors=tuple(errors),
    )
Deduplication

Financial feeds can return overlapping periods.

We therefore identify records deterministically.

# server/learning/datasets/deduplication.py

import hashlib

from ..acquisition.provider import PriceBar


def price_bar_key(
    bar: PriceBar,
) -> str:

    raw = "|".join(
        [
            bar.symbol,
            bar.timestamp.isoformat(),
            str(bar.open),
            str(bar.high),
            str(bar.low),
            str(bar.close),
            str(bar.volume),
        ]
    )

    return hashlib.sha256(
        raw.encode("utf-8")
    ).hexdigest()


def deduplicate(
    bars: list[PriceBar],
) -> list[PriceBar]:

    seen: set[str] = set()
    result: list[PriceBar] = []

    for bar in bars:

        key = price_bar_key(bar)

        if key in seen:
            continue

        seen.add(key)
        result.append(bar)

    return result

import hashlib
import json

from datetime import datetime

from ..acquisition.provider import PriceBar

class DatasetBuilder:

        self,
        bars: list[PriceBar],

            "symbol",
            "timestamp",
            "open",
            "high",
            "low",
            "close",
            "volume",

            json.dumps(

                sort_keys=True,

def returns(
    prices: list[Decimal],
) -> list[Decimal | None]:

    result: list[Decimal | None] = [None]

    for previous, current in zip(
        prices,
        prices[1:],

        if previous == 0:
            result.append(None)
            continue

        result.append(

@dataclass(frozen=True)

        name: str,

For the first production implementation, I would start with a gradient-boosting model rather than immediately throwing a Transformer at the problem.

Why?

# server/learning/training/gradient_boosting.py

from sklearn.ensemble import HistGradientBoostingRegressor

from .trainer import Trainer, TrainingResult


class GradientBoostingTrainer(Trainer):

    def train(
        self,
        X_train,
        y_train,
        X_validation,
        y_validation,
        parameters: dict,
    ) -> TrainingResult:

        model = HistGradientBoostingRegressor(
            **parameters
        )

        model.fit(
            X_train,
            y_train,
        )

        train_score = model.score(
            X_train,
            y_train,
        )

        validation_score = model.score(
            X_validation,
            y_validation,
        )

        return TrainingResult(
            model=model,
            metrics={
                "train_r2": float(train_score),
                "validation_r2": float(
                    validation_score
                ),
            },
            parameters=parameters,
            training_rows=len(X_train),
            validation_rows=len(
                X_validation
            ),
        )

9. Don't select based on training score

This is important.

Never do:

best = max(
    results,
    key=lambda x: x.metrics["train_r2"]
)

That encourages overfitting.

Use validation performance:

best = max(
    results,
    key=lambda x:
    x.metrics["validation_r2"]
)

And even that isn't sufficient for financial models.

The final selection must happen using walk-forward out-of-sample performance.

# server/learning/training/artifact.py

import hashlib
from pathlib import Path

import joblib


class ModelArtifactStore:

    def __init__(
        self,
        root: Path,
    ):

        self.root = root

        self.root.mkdir(
            parents=True,
            exist_ok=True,
        )

    def save(
        self,
        model,
        model_name: str,
        version: str,
    ):

        path = (
            self.root
            / model_name
            / version
        )

        path.mkdir(
            parents=True,
            exist_ok=True,
        )

        artifact = path / "model.joblib"

        joblib.dump(
            model,
            artifact,
        )

        checksum = hashlib.sha256(
            artifact.read_bytes()
        ).hexdigest()

        return {
            "path": str(artifact),
            "checksum": checksum,
        }

3. Walk-forward splitter
# server/learning/evaluation/splitter.py

from dataclasses import dataclass


@dataclass(frozen=True)
class TimeWindow:

    train_start: int
    train_end: int

    test_start: int
    test_end: int


class WalkForwardSplitter:

    def __init__(
        self,
        train_size: int,
        test_size: int,
        step_size: int,
    ):
        if min(
            train_size,
            test_size,
            step_size,
        ) <= 0:
            raise ValueError(
                "Window sizes must be positive"
            )

        self.train_size = train_size
        self.test_size = test_size
        self.step_size = step_size

    def split(
        self,
        total_rows: int,
    ):

        start = 0

        while (
            start
            + self.train_size
            + self.test_size
            <= total_rows
        ):

            train_start = start
            train_end = (
                start + self.train_size
            )

            test_start = train_end
            test_end = (
                test_start + self.test_size
            )

            yield TimeWindow(
                train_start=train_start,
                train_end=train_end,
                test_start=test_start,
                test_end=test_end,
            )

            start += self.step_size
4. Walk-forward execution
# server/learning/evaluation/walk_forward.py

from dataclasses import dataclass


@dataclass(frozen=True)
class FoldResult:

    fold: int

    train_rows: int
    test_rows: int

    metrics: dict[str, float]


class WalkForwardEvaluator:

    def __init__(
        self,
        splitter,
        trainer,
        predictor,
        metric_engine,
    ):
        self.splitter = splitter
        self.trainer = trainer
        self.predictor = predictor
        self.metric_engine = metric_engine

    def evaluate(
        self,
        X,
        y,
    ):

        results = []

        for fold_number, window in enumerate(
            self.splitter.split(len(X))
        ):

            X_train = X[
                window.train_start:
                window.train_end
            ]

            y_train = y[
                window.train_start:
                window.train_end
            ]

            X_test = X[
                window.test_start:
                window.test_end
            ]

            y_test = y[
                window.test_start:
                window.test_end
            ]

            model = self.trainer(
                X_train,
                y_train,
            )

            predictions = self.predictor(
                model,
                X_test,
            )

            metrics = (
                self.metric_engine.evaluate(
                    y_test,
                    predictions,
                )
            )

            results.append(
                FoldResult(
                    fold=fold_number,
                    train_rows=len(X_train),
                    test_rows=len(X_test),
                    metrics=metrics,
                )
            )

        return results

Now every result comes from genuinely unseen future data.

6. Backtesting engine
# server/learning/evaluation/backtest.py

from dataclasses import dataclass


@dataclass(frozen=True)
class Trade:

    timestamp: object

    position: float

    asset_return: float

    transaction_cost: float

    pnl: float


class BacktestEngine:

    def run(
        self,
        predictions,
        returns,
        timestamps,
        transaction_cost=0.0,
    ):

        trades = []

        previous_position = 0.0

        for timestamp, prediction, asset_return in zip(
            timestamps,
            predictions,
            returns,
        ):

            position = float(prediction)

            turnover = abs(
                position - previous_position
            )

            cost = (
                turnover
                * transaction_cost
            )

            pnl = (
                position
                * asset_return
                - cost
            )

            trades.append(
                Trade(
                    timestamp=timestamp,
                    position=position,
                    asset_return=asset_return,
                    transaction_cost=cost,
                    pnl=pnl,
                )
            )

            previous_position = position

        return trades

7. Portfolio metrics
# server/learning/evaluation/portfolio.py

import numpy as np


def cumulative_return(
    returns,
) -> float:

    values = np.asarray(
        returns,
        dtype=float,
    )

    return float(
        np.prod(1.0 + values) - 1.0
    )


def maximum_drawdown(
    returns,
) -> float:

    values = np.asarray(
        returns,
        dtype=float,
    )

    equity = np.cumprod(
        1.0 + values
    )

    peak = np.maximum.accumulate(
        equity
    )

    drawdown = (
        equity / peak
    ) - 1.0

    return float(
        drawdown.min()
    )


def sharpe_ratio(
    returns,
    periods_per_year: int,
) -> float:

    values = np.asarray(
        returns,
        dtype=float,
    )

    if len(values) < 2:
        return 0.0

    volatility = values.std(
        ddof=1
    )

    if volatility == 0:
        return 0.0

    return float(
        values.mean()
        / volatility
        * np.sqrt(periods_per_year)
    )

9. Statistical significance

This is another critical layer.

Suppose:

Model A:
+12.1%

Model B:
+10.4%

That doesn't automatically mean A is better.

The difference could simply be noise.

We need statistical testing.

# server/learning/evaluation/statistics.py

import numpy as np


def bootstrap_difference(
    candidate_returns,
    baseline_returns,
    iterations=5000,
    seed=42,
):

    rng = np.random.default_rng(seed)

    candidate = np.asarray(
        candidate_returns,
        dtype=float,
    )

    baseline = np.asarray(
        baseline_returns,
        dtype=float,
    )

    if len(candidate) != len(baseline):
        raise ValueError(
            "Return series must have equal length"
        )

    differences = candidate - baseline

    samples = rng.choice(
        differences,
        size=(
            iterations,
            len(differences),
        ),
        replace=True,
    )

    means = samples.mean(
        axis=1
    )

    lower = float(
        np.percentile(means, 2.5)
    )

    upper = float(
        np.percentile(means, 97.5)
    )

    return {
        "observed_difference":
            float(differences.mean()),
        "confidence_interval":
            (lower, upper),
        "significant":
            lower > 0 or upper < 0,
    }

For production, we can add more appropriate time-series tests and multiple-comparison corrections.

# server/learning/evaluation/drift.py

import numpy as np


class DistributionDriftDetector:

    def __init__(
        self,
        threshold: float,
    ):
        self.threshold = threshold

    def detect(
        self,
        reference,
        current,
    ):

        reference = np.asarray(
            reference,
            dtype=float,
        )

        current = np.asarray(
            current,
            dtype=float,
        )

        reference_mean = reference.mean()
        current_mean = current.mean()

        reference_std = (
            reference.std()
        )

        if reference_std == 0:
            return {
                "drift": False,
                "score": 0.0,
            }

        score = abs(
            current_mean
            - reference_mean
        ) / reference_std

        return {
            "drift": score > self.threshold,
            "score": float(score),
        }

Later we'll use more rigorous distribution tests.

# server/learning/evaluation/decision.py

from dataclasses import dataclass


@dataclass(frozen=True)
class PromotionPolicy:

    minimum_folds: int

    minimum_sharpe: float

    maximum_drawdown: float

    minimum_baseline_improvement: float

    require_statistical_significance: bool


class PromotionEngine:

    def evaluate(
        self,
        report,
        policy: PromotionPolicy,
    ):

        reasons = []

        if len(report.folds) < (
            policy.minimum_folds
        ):
            reasons.append(
                "Insufficient evaluation folds"
            )

        sharpe = report.metrics.get(
            "sharpe",
            float("-inf"),
        )

        if sharpe < policy.minimum_sharpe:
            reasons.append(
                "Sharpe ratio below threshold"
            )

        drawdown = report.metrics.get(
            "max_drawdown",
            float("-inf"),
        )

        if drawdown < policy.maximum_drawdown:
            reasons.append(
                "Maximum drawdown exceeded"
            )

        if policy.require_statistical_significance:

            if not report.statistics.get(
                "significant",
                False,
            ):
                reasons.append(
                    "Performance difference "
                    "not statistically significant"
                )

        return {
            "promote": len(reasons) == 0,
            "reasons": reasons,
        }

# server/learning/experience/error_analysis.py


class ErrorAnalyzer:

    def classify(
        self,
        prediction,
        outcome,
    ):

        if prediction.prediction > 0:
            predicted_direction = 1
        elif prediction.prediction < 0:
            predicted_direction = -1
        else:
            predicted_direction = 0

        if outcome.actual_value > 0:
            actual_direction = 1
        elif outcome.actual_value < 0:
            actual_direction = -1
        else:
            actual_direction = 0

        if (
            predicted_direction
            != actual_direction
        ):
            return "directional_error"

        if (
            prediction.confidence > 0.8
            and abs(outcome.error) > 0.05
        ):
            return "overconfidence"

        if (
            prediction.confidence < 0.3
            and abs(outcome.error) < 0.01
        ):
            return "underconfidence"

        return "normal_error"

class PatternDetector:

    def recurring_error_types(
        self,
        experiences,
    ):

        errors = [
            experience.error_type
            for experience in experiences
            if experience.error_type

18. Add confidence calibration

Aariya also needs to learn whether its confidence is trustworthy.

Suppose it repeatedly says:

confidence = 0.90

but only gets the prediction right 60% of the time.

That's a calibration problem.

Create:

# server/learning/experience/calibration.py

import numpy as np


def calibration_error(
    confidence,
    correct,
    bins=10,
):

    confidence = np.asarray(
        confidence,
        dtype=float,
    )

    correct = np.asarray(
        correct,
        dtype=float,
    )

    boundaries = np.linspace(
        0.0,
        1.0,
        bins + 1,
    )

    total_error = 0.0

    for lower, upper in zip(
        boundaries[:-1],
        boundaries[1:],
    ):

        mask = (
            (confidence >= lower)
            & (confidence < upper)
        )

        if not mask.any():
            continue

        predicted = confidence[mask].mean()
        actual = correct[mask].mean()

        weight = mask.mean()

        total_error += (
            abs(predicted - actual)
            * weight
        )

    return float(total_error)

Now Aariya can learn:

"I am systematically overconfident
in this market regime."

That's a legitimate learning signal.

# server/learning/orchestrator/state_machine.py

from .state import (
    ALLOWED_TRANSITIONS,
    LearningState,
)


class InvalidTransition(Exception):
    pass


class LearningStateMachine:

    def __init__(
        self,
        state: LearningState,
    ):
        self.state = state

    def transition(
        self,
        new_state: LearningState,
    ):

        allowed = ALLOWED_TRANSITIONS.get(
            self.state,
            set(),
        )

        if new_state not in allowed:

            raise InvalidTransition(
                f"{self.state} -> "
                f"{new_state} is not allowed"
            )

        self.state = new_state

        return self.state

def calculate_priority(
    evidence,
    expected_improvement,
    model_importance,
    confidence,
    estimated_cost,
):

    if estimated_cost <= 0:
        raise ValueError(
            "Estimated cost must be positive"

    return (
        evidence

        * model_importance

10. The executor

This component executes one learning task.

# server/learning/orchestrator/executor.py

class LearningExecutor:

    def __init__(
        self,
        acquisition,
        dataset_pipeline,
        trainer,
        evaluator,
        experience,
        task_generator,
    ):

        self.acquisition = acquisition
        self.dataset_pipeline = dataset_pipeline
        self.trainer = trainer
        self.evaluator = evaluator
        self.experience = experience
        self.task_generator = task_generator

    async def execute(
        self,
        task,
    ):

        data = await self.acquisition.execute(
            task
        )

        dataset = await (
            self.dataset_pipeline.build(
                data
            )
        )

        candidates = await (
            self.trainer.train(
                dataset
            )
        )

        evaluations = []

        for candidate in candidates:

            result = await (
                self.evaluator.evaluate(
                    candidate,
                    dataset,
                )
            )

            evaluations.append(result)

        return evaluations

This is intentionally an orchestration layer.

The individual systems remain independent.

11. Main orchestrator
# server/learning/orchestrator/orchestrator.py

class LearningOrchestrator:

    def __init__(
        self,
        planner,
        executor,
        repository,
        concurrency,
    ):

        self.planner = planner
        self.executor = executor
        self.repository = repository
        self.concurrency = concurrency

    async def run_once(self):

        tasks = await (
            self.repository
            .get_pending_tasks()
        )

        if not tasks:
            return None

        task = self.planner.select(
            tasks
        )

        await self.repository.mark_running(
            task.id
        )

        try:

            async with self.concurrency:

                result = await (
                    self.executor.execute(
                        task
                    )
                )

            await self.repository.complete(
                task.id,
                result,
            )

            return result

        except Exception as exc:

            await self.repository.fail(
                task.id,
                str(exc),
            )

            raise

class CircuitBreaker:

    def __init__(
        self,
        failure_threshold: int = 5,
    ):

        self.failure_threshold = (
            failure_threshold
        )

        self.failures = 0
        self.open = False

    def success(self):

        self.failures = 0
        self.open = False

    def failure(self):

        self.failures += 1

        if (
            self.failures
            >= self.failure_threshold
        ):
            self.open = True

    def allow(self):

        return not self.open

14. Claim verification

Now cross-check claims.

# server/research/claims/verifier.py


class ClaimVerifier:

    async def verify(
        self,
        claim,
        supporting_claims,
    ):

        if not supporting_claims:

            return {
                "status": "unverified",
                "confidence": 0.0,
            }

        agreeing = 0
        contradicting = 0

        for other in supporting_claims:

            if (
                other.predicate
                == claim.predicate
                and other.object
                == claim.object
            ):
                agreeing += 1

            else:
                contradicting += 1

        total = (
            agreeing
            + contradicting
        )

        confidence = (
            agreeing / total
            if total
            else 0.0
        )

        return {
            "status":
                "verified"
                if confidence >= 0.8
                else "uncertain",
            "confidence": confidence,
        }

This is only a baseline.

For serious research we'll need entity resolution, semantic equivalence, temporal reasoning, and source weighting.

22. Research orchestrator
# server/research/orchestration/researcher.py


class AutonomousResearcher:

    def __init__(
        self,
        decomposer,
        source_registry,
        source_ranker,
        fetcher,
        validator,
        extractor,
        verifier,
        knowledge_graph,
        task_generator,
    ):

        self.decomposer = decomposer
        self.sources = source_registry
        self.ranker = source_ranker
        self.fetcher = fetcher
        self.validator = validator
        self.extractor = extractor
        self.verifier = verifier
        self.graph = knowledge_graph
        self.tasks = task_generator

    async def research(
        self,
        objective,
    ):

        subtasks = (
            self.decomposer.decompose(
                objective
            )
        )

        results = []

        for subtask in subtasks:

            sources = await self.sources.find()

            sources = self.ranker.rank(
                sources
            )

            for source in sources[
                :objective.max_sources
            ]:

                document = await self.fetcher.fetch(
                    source.uri
                )

                self.validator.validate(
                    document
                )

                claims = await (
                    self.extractor.extract(
                        document
                    )
                )

                for claim in claims:

                    verification = await (
                        self.verifier.verify(
                            claim,
                            [],
                        )
                    )

                    claim.confidence = (
                        verification[
                            "confidence"
                        ]
                    )

                    await self.graph.add_entity(
                        claim
                    )

                    results.append(
                        claim
                    )

        return results

This is the skeleton. The production implementation should add durable task state, retries, provenance persistence, deduplication, source quotas, and event emission around every operation.

def calculate_market_cap(
    price: float,
    diluted_shares: float,
) -> float:

    if price < 0:
        raise ValueError(
            "Price cannot be negative"
        )

    if diluted_shares <= 0:
        raise ValueError(
            "Shares must be positive"
        )

    return (
        price
        * diluted_shares
    )

export function ReportListener({
    socket,
    onReport,
}: {
    socket: WebSocket;
    onReport: (reportId: string) => void;
}) {

    useEffect(() => {

        const handler = (event: MessageEvent) => {

            const message =
                JSON.parse(event.data);

            if (
                message.type ===
                "INTELLIGENCE_REPORT_READY"
            ) {

                onReport(
                    message.report_id
                );
            }
        };

        socket.addEventListener(
            "message",
            handler
        );

        return () => {
            socket.removeEventListener(
                "message",
                handler
            );
        };

    }, [socket, onReport]);

    return null;
}

# server/reports/pdf.py

from io import BytesIO

from reportlab.lib.pagesizes import A4
from reportlab.platypus import (
    SimpleDocTemplate,
    Paragraph,
    Spacer,
    Table,
)
from reportlab.lib.styles import (
    getSampleStyleSheet,
)


def generate_report_pdf(
    report,
):

    buffer = BytesIO()

    document = SimpleDocTemplate(
        buffer,
        pagesize=A4,
    )

    styles = getSampleStyleSheet()

    story = []

    story.append(
        Paragraph(
            report.title,
            styles["Title"],
        )
    )

    story.append(
        Spacer(1, 12)
    )

    story.append(
        Paragraph(
            f"Company: {report.company}",
            styles["BodyText"],
        )
    )

    story.append(
        Paragraph(
            f"Ticker: {report.ticker or 'Not publicly listed'}",
            styles["BodyText"],
        )
    )

    story.append(
        Paragraph(
            f"Confidence: "
            f"{report.confidence:.1%}",
            styles["BodyText"],
        )
    )

    story.append(
        Spacer(1, 12)
    )

    table_data = [
        [
            "Scenario",
            "Probability",
            "Price",
            "Market Cap",
        ]
    ]

    for scenario in report.scenarios:

        table_data.append([
            scenario.name,
            f"{scenario.probability:.1%}",
            (
                str(scenario.price)
                if scenario.price is not None
                else "N/A"
            ),
            (
                str(scenario.market_cap)
                if scenario.market_cap is not None
                else "N/A"
            ),
        ])

    story.append(
        Table(table_data)
    )

    story.append(
        Spacer(1, 12)
    )

    story.append(
        Paragraph(
            "Thesis",
            styles["Heading2"],
        )
    )

    for item in report.thesis:

        story.append(
            Paragraph(
                f"• {item}",
                styles["BodyText"],
            )
        )

    story.append(
        Paragraph(
            "Risks",
            styles["Heading2"],
        )
    )

    for item in report.risks:

        story.append(
            Paragraph(
                f"• {item}",
                styles["BodyText"],
            )
        )

    document.build(story)

    buffer.seek(0)

    return buffer

28. FCM sender
# server/mobile/fcm.py

import httpx


class FCMClient:

    def __init__(
        self,
        project_id,
        access_token_provider,
    ):

        self.project_id = project_id
        self.access_token_provider = (
            access_token_provider
        )

    async def send(
        self,
        token,
        title,
        body,
        data,
    ):

        access_token = (
            await self.access_token_provider()
        )

        url = (
            "https://fcm.googleapis.com/v1/"
            f"{self.project_id}/messages:send"
        )

        payload = {
            "message": {
                "token": token,

                "notification": {
                    "title": title,
                    "body": body,
                },

                "data": {
                    key: str(value)
                    for key, value
                    in data.items()
                },
            }
        }

        async with httpx.AsyncClient() as client:

            response = await client.post(
                url,
                json=payload,
                headers={
                    "Authorization":
                        f"Bearer {access_token}",
                    "Content-Type":
                        "application/json",
                },
            )

            response.raise_for_status()

This uses the current FCM HTTP v1 model rather than the older legacy API.

For scientific publications, use the actual Crossref API rather than inventing a research dataset. Crossref's /works endpoint supports querying and cursor-based pagination.

# server/discovery/sources/crossref.py

import httpx


class CrossrefAdapter:

    BASE_URL = (
        "https://api.crossref.org/v1/works"
    )

    def __init__(self, mailto: str):

        self.mailto = mailto

    async def collect(
        self,
        cursor="*",
        rows=100,
    ):

        params = {
            "query":
                "technology invention breakthrough",
            "filter":
                "from-pub-date:"
                "2026-01-01",
            "rows": rows,
            "cursor": cursor,
            "mailto": self.mailto,
        }

        async with httpx.AsyncClient(
            timeout=30
        ) as client:

            response = await client.get(
                self.BASE_URL,
                params=params,
                headers={
                    "User-Agent":
                        "AariyaResearch/1.0"
                },

            response.raise_for_status()

            return response.json()

20. Collection worker
# server/discovery/worker.py


class DiscoveryWorker:

    async def execute(
        self,
        source_id,
    ):

        source = await self.repository.get(
            source_id
        )

        adapter = (
            self.adapter_factory
            .for_source(source)
        )

        checkpoint = (
            await self.checkpoints.get(
                source_id
            )
        )

        result = await adapter.collect(
            cursor=checkpoint.cursor
            if checkpoint
            else "*"
        )

        for item in result.items:

            normalized = (
                await adapter.normalize(
                    item
                )
            )

            if await self.is_duplicate(
                normalized
            ):
                continue

            await self.store(
                normalized
            )

        await self.checkpoints.update(
            source_id=source_id,
            cursor=result.next_cursor,
        )
21. Failure handling

Aariya should never stop because one source is unavailable.

SOURCE A ── success
SOURCE B ── timeout ── retry
SOURCE C ── 429 ────── backoff
SOURCE D ── 500 ────── retry
SOURCE E ── success

Implement exponential backoff:

def retry_delay(
    attempt: int,
    base: float = 2.0,
    maximum: float = 300.0,
):

    return min(
        maximum,
        base ** attempt,
    )

And respect source-provided rate limits.

Crossref, for example, explicitly documents rate/concurrency limits and recommends backing off when limits are exceeded.

16. DCF engine

Now the actual valuation.

# valuation/dcf.py


def enterprise_value(
    free_cash_flows,
    discount_rate,
    terminal_growth,
):

    if discount_rate <= terminal_growth:
        raise ValueError(
            "Discount rate must exceed "
            "terminal growth"
        )

    pv = 0.0

    for year, fcf in enumerate(
        free_cash_flows,
        start=1,
    ):

        pv += (
            fcf
            / (
                1 + discount_rate
            ) ** year
        )

    terminal_fcf = (
        free_cash_flows[-1]
        * (1 + terminal_growth)
    )

    terminal_value = (
        terminal_fcf
        / (
            discount_rate
            - terminal_growth
        )
    )

    terminal_pv = (
        terminal_value
        / (
            1 + discount_rate
        ) ** len(free_cash_flows)
    )

    return pv + terminal_pv

import random


def simulate(
    base_revenue,
    growth_distribution,
    margin_distribution,
    iterations=10000,
):

    results = []

    for _ in range(iterations):

        growth = random.choice(
            growth_distribution
        )

        margin = random.choice(
            margin_distribution
        )

        revenue = (
            base_revenue
            * (1 + growth)
        )

        results.append(
            revenue * margin
        )

    return results

But don't use random.choice as the production implementation. This illustrates the interface only; the production Monte Carlo engine should use properly fitted distributions from real historical/validated data.

def calculate_confidence(
    data_quality,
    entity_confidence,
    model_accuracy,
    comparable_strength,
    uncertainty,
):

    confidence = (
        data_quality * 0.20
        + entity_confidence * 0.20
        + model_accuracy * 0.25
        + comparable_strength * 0.15
        + (1 - uncertainty) * 0.20

    return max(
        0.0,

23. Forecast-change detection

Don't notify the user for meaningless changes.

def materially_changed(
    old_value,
    new_value,
    threshold=0.05,
):

    if old_value == 0:
        return True

    change = abs(
        new_value - old_value
    ) / abs(old_value)

    return change >= threshold

Later this threshold can become model-specific.

6. Calculate prediction error
def percentage_error(
    predicted,
    actual,
):

    if actual is None:
        return None

    if actual == 0:
        return None

    return abs(
        predicted - actual
    ) / abs(actual)

Directional accuracy:

def direction_correct(
    predicted_return,
    actual_return,
):

    if (
        predicted_return is None
        or actual_return is None
    ):
        return None

    return (
        predicted_return >= 0
        and actual_return >= 0
    ) or (
        predicted_return < 0
        and actual_return < 0
    )

class ErrorAnalyzer:

    async def analyze(
        self,
        forecast,
        outcome,
    ):

        errors = []

        if outcome.actual_return is None:
            return errors

        if (
            forecast.expected_return > 0

25. Evaluation interface
class WalkForwardEvaluator:

    async def evaluate(
        self,
        model,
        dataset,
        windows,
    ):

        results = []

        for window in windows:

            trained = model.fit(
                window.train
            )

            predictions = trained.predict(
                window.test
            )

            results.append(
                self.score(
                    predictions,
                    window.test,
                )
            )

        return results
