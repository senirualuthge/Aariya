import uuid

class FaceTracker:
    def __init__(self):
        self.tracks = {}

    def assign_id(self, face_box):
        # find nearest existing track by IoU, or create new track
        best_id = None
        best_iou = 0.0
        for tid, tbox in self.tracks.items():
            iou = self._box_iou(face_box, tbox)
            if iou > best_iou:
                best_iou = iou
                best_id = tid
        if best_iou > 0.3:
            self.tracks[best_id] = face_box
            return best_id
        face_id = str(uuid.uuid4())
        self.tracks[face_id] = face_box
        return face_id

    def _box_iou(self, a, b):
        ax, ay, aw, ah = a
        bx, by, bw, bh = b
        xi1, yi1 = max(ax, bx), max(ay, by)
        xi2, yi2 = min(ax + aw, bx + bw), min(ay + ah, by + bh)
        inter = max(0, xi2 - xi1) * max(0, yi2 - yi1)
        union = aw * ah + bw * bh - inter
        return inter / union if union > 0 else 0

class LearningRouter:
    def __init__(self, lr=0.01, gamma=0.95):
        self.q_table = {}
        self.lr = lr
        self.gamma = gamma

    def update(self, action, reward):
        state_key = f"state_{action}"
        old_q = self.q_table.get(state_key, 0.0)
        self.q_table[state_key] = old_q + self.lr * (reward + self.gamma * old_q - old_q)

        "bbox": f.bbox.astype(int).tolist(),
        "confidence": float(f.det_score),
        "embedding": f.embedding.astype(np.float32)

known_db = []  # populated from DB on startup
def load_known_people():
    from db import get_all_identities
    known_db.extend(get_all_identities())

THRESHOLD = 0.6

def get_embedding(face_img):
    rep = DeepFace.represent(face_img, model_name="Facenet", enforce_detection=False)
    return rep[0]["embedding"]

def match_face(embedding):
    best = None
    best_score = 999

    for person in known_db:
        dist = np.linalg.norm(np.array(embedding) - np.array(person["embedding"]))
        if dist < best_score:
            best_score = dist
            best = person

    if best_score < THRESHOLD:
        return best, best_score

    return best, 1.0  # no match above threshold; return best candidate with max distance
🎥 Camera Loop
def run():
    cap = cv2.VideoCapture(0)

    while True:
        ret, frame = cap.read()
        if not ret:
            continue

        detections = detect_faces(frame)

        tracks = tracker.update_tracks(detections, frame=frame)

        for track in tracks:
            if not track.is_confirmed():
                continue

            track_id = track.track_id
            l, t, r, b = map(int, track.to_ltrb())

            face = frame[t:b, l:r]

            embedding = get_embedding(face)
            person, score = match_face(embedding)

            if person:
                broadcast_event({
                    "type": "known",
                    "id": person["id"],
                    "name": person["name"],
                    "track_id": track_id
                })
            else:
                broadcast_event({
                    "type": "unknown",
                    "track_id": track_id,
                    "embedding": embedding
                })

        cv2.imshow("cam", frame)
        if cv2.waitKey(1) == 27:
            break

import numpy as np

class Bandit:
    def __init__(self):

            context["novelty"],
            context["confidence"],
            context["duration"],
            context["role_weight"]

        return np.argmax(scores)

    def update(self, action, reward):

from dataclasses import dataclass, field
from typing import List, Optional
import asyncio, json, time

@dataclass
class ConfidencePacket:
    track_id: str
    embedding: List[float]
    detection_conf: float
    embedding_quality: float
    identity_id: Optional[str] = None
    match_score: Optional[float] = None
    timestamp: float = field(default_factory=time.time)

class EventPipeline:
    def __init__(self):
        self.queue = asyncio.Queue(maxsize=500)
        self.confidence_bands = {"high": 0.8, "medium": 0.5, "low": 0.0}

    async def push(self, packet: ConfidencePacket):
        await self.queue.put(packet)

    async def process(self):
        while True:
            packet = await self.queue.get()
            band = "low"
            for name, threshold in sorted(self.confidence_bands.items(), key=lambda x: -x[1]):
                if packet.detection_conf >= threshold:
                    band = name
                    break
            packet.embedding_quality = band
            await self._route(packet)

        # routing logic: forward to identity matcher, RL decision, and dashboard broadcast
        identity_event = {
            "type": "identity_event",

            "match_score": packet.match_score,
            "identity_id": packet.identity_id,
            "timestamp": packet.timestamp
        }

import numpy as np
from collections import defaultdict

class CausalAnalyzer:
    def __init__(self):
        self.history = []
        self.causal_links = defaultdict(list)

    def log_decision(self, state, action, outcome, reward, track_id):
        record = {
            "track_id": track_id,
            "state": state,
            "action": action,
            "outcome": outcome,
            "reward": reward,
            "timestamp": time.time()
        }
        self.history.append(record)
        # build causal chain: which state factors drove which actions
        if outcome in ("false_match", "false_positive"):
            causes = [k for k, v in state.items() if v < 0.5]
            self.causal_links["false_positive"].append({"causes": causes, "state": state})

    def get_causal_chain(self, track_id):
        chain = [r for r in self.history if r["track_id"] == track_id]
        return chain

    def drift_insight(self):
        if len(self.history) < 10:
            return "insufficient data"
        rewards = [r["reward"] for r in self.history[-50:]]
        return {"mean_reward": np.mean(rewards), "trend": "improving" if np.mean(rewards[-10:]) > np.mean(rewards[:10]) else "degrading"}

import numpy as np

class LinUCB:
    def __init__(self, n_actions, d, alpha=1.0):
        self.n_actions = n_actions
        self.alpha = alpha
        self.A = [np.identity(d) for _ in range(n_actions)]
        self.b = [np.zeros((d, 1)) for _ in range(n_actions)]

    def select_action(self, x):
        x = x.reshape(-1, 1)
        p = []

        for a in range(self.n_actions):
            A_inv = np.linalg.inv(self.A[a])
            theta = A_inv @ self.b[a]
            p_t = (theta.T @ x + self.alpha * np.sqrt(x.T @ A_inv @ x))
            p.append(p_t.item())

        return int(np.argmax(p))

    def update(self, action, x, reward):
        x = x.reshape(-1, 1)
        self.A[action] += x @ x.T
        self.b[action] += reward * x

import numpy as np

class Bandit:
    def __init__(self):
        self.model = [np.identity(5) for _ in range(4)]
        self.b = [np.zeros((5,1)) for _ in range(4)]

    def select(self, x):
        scores = []
        for i in range(4):
            A_inv = np.linalg.inv(self.model[i])
            theta = A_inv @ self.b[i]
            score = theta.T @ x + np.sqrt(x.T @ A_inv @ x)
            scores.append(score.item())
        return int(np.argmax(scores))

import numpy as np
import random

import cv2
from deepface import DeepFace

def _get_embedding(face_img):
    rep = DeepFace.represent(face_img, model_name="Facenet", enforce_detection=False)
    return rep[0]["embedding"]

def _cosine_similarity(a, b):
    a, b = np.array(a), np.array(b)
    return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b) + 1e-12))

def process_face(face_img=None, known_embeddings=None):
    if face_img is None:
        return np.random.rand(512).tolist(), random.random()
    embedding = _get_embedding(face_img)
    best_sim = 0.0
    for ke in (known_embeddings or []):
        sim = _cosine_similarity(embedding, ke)
        if sim > best_sim:
            best_sim = sim
    return embedding, best_sim

from collections import Counter

class DelayedIdentity:
    def __init__(self, window=5, min_confirmations=3):
        self.votes = {}
        self.window = window
        self.threshold = min_confirmations

    def vote(self, track_id, candidate_id):
        if track_id not in self.votes:
            self.votes[track_id] = []
        self.votes[track_id].append(candidate_id)
        if len(self.votes[track_id]) >= self.window:
            counter = Counter(self.votes[track_id])
            top, count = counter.most_common(1)[0]
            if count >= self.threshold:
                del self.votes[track_id]
                return {"status": "confirmed", "identity": top}
            self.votes[track_id].pop(0)
        return {"status": "pending", "votes_so_far": len(self.votes.get(track_id, [])), "needed": self.threshold}

import torch.optim as optim

model = DQN()
optimizer = optim.Adam(model.parameters(), lr=1e-3)
gamma = 0.95

def train(batch):
    states, actions, rewards, next_states = batch

    q_values = model(states)
    next_q = model(next_states).detach()

    target = q_values.clone()

    for i in range(len(states)):
        target[i][actions[i]] = rewards[i] + gamma * torch.max(next_q[i])

    loss = ((q_values - target)**2).mean()

    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

🧠 Replay Buffer (memory/replay_buffer.py)
import random
from collections import deque

class ReplayBuffer:
    def __init__(self, capacity=100000):
        self.buffer = deque(maxlen=capacity)

    def push(self, state, action, reward, next_state):
        self.buffer.append((state, action, reward, next_state))

    def sample(self, batch_size):
        batch = random.sample(self.buffer, batch_size)
        states, actions, rewards, next_states = zip(*batch)
        return states, actions, rewards, next_states

    def __len__(self):
        return len(self.buffer)

    states, actions, rewards, next_states = buffer.sample(BATCH_SIZE)

    states = torch.tensor(states, dtype=torch.float32)
    next_states = torch.tensor(next_states, dtype=torch.float32)
    rewards = torch.tensor(rewards, dtype=torch.float32)
    actions = torch.tensor(actions)

    q_values = model(states)
    next_q = model(next_states).detach()

    target = q_values.clone()

    for i in range(BATCH_SIZE):
        target[i][actions[i]] = rewards[i] + GAMMA * torch.max(next_q[i])

    loss = ((q_values - target)**2).mean()

    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

    return loss.item()

    loss = train_step()

epsilon = 1.0

import random

def select_action(state):
    global epsilon

    if random.random() < epsilon:
        return random.randint(0, 3)

    with torch.no_grad():
        q = model(torch.tensor(state, dtype=torch.float32))
        return int(torch.argmax(q))

def decay_epsilon():
    global epsilon
    epsilon = max(epsilon_min, epsilon * epsilon_decay)

# memory/persistent_buffer.py

import pickle
import os
from datetime import datetime

class PersistentReplayBuffer:
    def __init__(self, capacity=100000, path="buffer_store"):
        self.buffer = []
        self.capacity = capacity
        self.path = path
        os.makedirs(path, exist_ok=True)

    def push(self, experience):
        if len(self.buffer) >= self.capacity:
            self.buffer.pop(0)
        self.buffer.append(experience)

    def save(self):
        filename = f"{self.path}/buffer_{datetime.now().strftime('%Y%m%d_%H%M%S')}.pkl"
        with open(filename, "wb") as f:
            pickle.dump(self.buffer, f)

    def load_latest(self):
        files = sorted(os.listdir(self.path))
        if not files:
            return
        with open(f"{self.path}/{files[-1]}", "rb") as f:
            self.buffer = pickle.load(f)

class TemperatureScaler(nn.Module):
    def __init__(self):
        super().__init__()

🎯 Fit Temperature
def calibrate(logits, labels):
    scaler = TemperatureScaler()
    optimizer = torch.optim.LBFGS([scaler.temperature], lr=0.01)

    def loss_fn():
        optimizer.zero_grad()
        loss = nn.CrossEntropyLoss()(scaler(logits), labels)
        loss.backward()
        return loss

    optimizer.step(loss_fn)
    return scaler

import os
import shutil

MODEL_DIR = "storage/models"
ACTIVE_LINK = os.path.join(MODEL_DIR, "active")

class VersionManager:

    def list_versions(self):
        return sorted(os.listdir(MODEL_DIR))

    def get_active_version(self):
        return os.readlink(ACTIVE_LINK)

    def switch_version(self, version: str):
        target_path = os.path.join(MODEL_DIR, version)

        if not os.path.exists(target_path):
            raise ValueError("Model version not found")

        if os.path.islink(ACTIVE_LINK):
            os.remove(ACTIVE_LINK)

        os.symlink(target_path, ACTIVE_LINK)

        return {"status": "switched", "active": version}

def rollback(reason: str):

    current = vm.get_active_version()
    versions = vm.list_versions()

    if len(versions) < 2:

def compute_reward(action, metrics):

    stability = metrics["accuracy"]
    drift_penalty = metrics["drift"] * 2
    complaint_penalty = metrics["user_complaints"] * 3

    if action == 1:  # rollback
        return 1 if drift_penalty > 0.5 else -1

    if action == 0:  # keep
        return stability - drift_penalty - complaint_penalty

    if action == 2:  # extend shadow
        return stability * 0.5
🤖 RL Agent (Simple Policy)
agent.py
import random

class RollbackAgent:
    def __init__(self, state_dim=4, action_dim=3, lr=1e-3):
        import torch.nn as nn
        self.net = nn.Sequential(
            nn.Linear(state_dim, 64),
            nn.ReLU(),
            nn.Linear(64, action_dim)
        )
        self.optimizer = torch.optim.Adam(self.net.parameters(), lr=lr)
        self.epsilon = 0.2

    def act(self, state):
        if random.random() < self.epsilon:
            return random.choice([0, 1, 2])
        with torch.no_grad():
            q = self.net(torch.tensor(state, dtype=torch.float32))
            return int(torch.argmax(q))

    def train(self, state, action, reward, next_state):
        q = self.net(torch.tensor(state, dtype=torch.float32))
        next_q = self.net(torch.tensor(next_state, dtype=torch.float32)).detach()
        target = q.clone()
        target[action] = reward + 0.95 * torch.max(next_q)
        loss = ((q - target) ** 2).mean()
        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()

class EventStore:
    def __init__(self, redis_client=None):
        self.redis = redis_client
        self.stream_key = "model_registry_events"

    def append(self, event):
        import json, time
        payload = {**event, "timestamp": time.time()}
        if self.redis:
            self.redis.xadd(self.stream_key, {"data": json.dumps(payload)})
        with open("registry_events.log", "a") as f:
            f.write(json.dumps(payload) + "\n")

    def get_events(self):
        import json
        events = []
        try:
            with open("registry_events.log", "r") as f:
                for line in f:
                    events.append(json.loads(line.strip()))
        except FileNotFoundError:
            import logging
            logging.warning("registry_events.log not found — returning empty event list")
        return events
