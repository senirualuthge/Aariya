  Section 404: PREAMBLE (from NEWPersistent autonomous cognitive system t.txt)

The core ideas here are:

Background Knowledge Gaps — Aariya detects gaps mid-conversation, queues them, and researches asynchronously
Proactive Conflict Resolution — When internal contradictions or unknowns arise, she researches and surfaces insights on next session start
Persistent Memory + Learning Store — Research results are stored and indexed for future use

Here's the full architecture with code:

from autonomous.insight_store import get_unsurfaced_insights, mark_surfaced

def get_aariya_briefing() -> str:
    """
    Call this at the START of every conversation with Aariya.
    Returns a brief of everything she learned while you were away.
    """
    insights = get_unsurfaced_insights()

    if not insights:
        return ""

    lines = ["🌌 *Aariya has been learning while you were away...*\n"]

    for insight in insights:
        lines.append(f"━━━━━━━━━━━━━━━━━━━━━━━━")
        lines.append(f"📚 **{insight['topic']}**")
        lines.append(insight["insight"][:500] + ("..." if len(insight["insight"]) > 500 else ""))
        lines.append(f"*(Learned at {insight['learned_at'][:16]})*\n")
        mark_surfaced(insight["id"])

    return "\n".join(lines)

from neo4j import AsyncGraphDatabase
from typing import List, Dict, Any
import uuid

class KnowledgeGraph:
    """Neo4j knowledge graph operations"""
    
    def __init__(self, uri: str, user: str, password: str):
        self.uri = uri
        self.user = user
        self.password = password
        self.driver = None
    
    async def initialize(self):
        self.driver = AsyncGraphDatabase.driver(
            self.uri, 
            auth=(self.user, self.password)
        )
    
    async def create_entity(self, name: str, properties: Dict[str, Any]) -> str:
        """Create or merge an entity node"""
        async with self.driver.session() as session:
            result = await session.run("""
                MERGE (e:Entity {name: $name})
                SET e += $properties
                RETURN e.id AS id
            """, name=name, properties=properties)
            record = await result.single()
            return record["id"] if record else None
    
    async def create_relation(
        self,
        source: str,
        target: str,
        relation_type: str,
        properties: Dict[str, Any] = None
    ):
        """Create relationship between entities"""
        async with self.driver.session() as session:
            await session.run("""
                MATCH (a:Entity {name: $source})
                MATCH (b:Entity {name: $target})
                CREATE (a)-[r:RELATIONSHIP {type: $rel_type}]->(b)
                SET r += $properties
            """, source=source, target=target, rel_type=relation_type, 
                properties=properties or {})
    
    async def find_connections(self, entity_name: str, depth: int = 2) -> List[Dict]:
        """Find connected entities up to depth"""
        async with self.driver.session() as session:
            result = await session.run("""
                MATCH path = (e:Entity {name: $name})-[*1..$depth]-(connected)
                RETURN connected.name AS name, 
                       labels(connected) AS labels,
                       relationships(path) AS relationships
            """, name=entity_name, depth=depth)
            return [record.data() async for record in result]

async def search_web(query: str) -> str:
    """Search using Brave Search API (or swap for SearXNG)."""
    async with httpx.AsyncClient() as client:
        resp = await client.get(
            SEARCH_API,
            headers={"Accept": "application/json", "X-Subscription-Token": BRAVE_API_KEY},

            timeout=15.0
        )

    results = []
    for r in data.get("web", {}).get("results", []):
