import os
import subprocess
from pathlib import Path
from typing import List, Dict

class FileSystemAgent:
    def __init__(self):
        self.allowed_roots = self._detect_roots()

    def _detect_roots(self) -> List[str]:
        roots = []

        # Windows drives
        if os.name == "nt":
            for letter in "ABCDEFGHIJKLMNOPQRSTUVWXYZ":
                drive = f"{letter}:\\"
                if os.path.exists(drive):
                    roots.append(drive)

        # Linux/macOS
        else:
            roots.extend([
                "/",
                "/mnt",
                "/media",
                "/Volumes"
            ])

        return roots

    # ---------------------------------------------------
    # LIST FILES
    # ---------------------------------------------------
    def list_directory(self, path: str) -> List[Dict]:
        p = Path(path)

        if not p.exists():
            return []

        results = []

        for item in p.iterdir():
            results.append({
                "name": item.name,
                "path": str(item),
                "is_dir": item.is_dir(),
                "size": item.stat().st_size if item.is_file() else 0
            })

        return results

    # ---------------------------------------------------
    # READ FILE
    # ---------------------------------------------------
    def read_file(self, path: str) -> str:
        p = Path(path)

        if not p.exists():
            return "File not found."

        if p.is_dir():
            return "Cannot read directory."

        try:
            with open(p, "r", encoding="utf-8", errors="ignore") as f:
                return f.read()
        except Exception as e:
            return str(e)

    # ---------------------------------------------------
    # WRITE FILE
    # ---------------------------------------------------
    def write_file(self, path: str, content: str):
        p = Path(path)

        p.parent.mkdir(parents=True, exist_ok=True)

        with open(p, "w", encoding="utf-8") as f:
            f.write(content)

        return True

    # ---------------------------------------------------
    # APPEND FILE
    # ---------------------------------------------------
    def append_file(self, path: str, content: str):
        with open(path, "a", encoding="utf-8") as f:
            f.write(content)

    # ---------------------------------------------------
    # OPEN FILE / APP
    # ---------------------------------------------------
    def open_path(self, path: str):
        if os.name == "nt":
            os.startfile(path)

        elif os.name == "posix":
            subprocess.Popen(["xdg-open", path])

        elif os.name == "mac":
            subprocess.Popen(["open", path])

    # ---------------------------------------------------
    # SEARCH FILES
    # ---------------------------------------------------
    def search_files(self, root: str, query: str):
        matches = []

        for dirpath, _, filenames in os.walk(root):
            for file in filenames:
                if query.lower() in file.lower():
                    matches.append(os.path.join(dirpath, file))

        return matches

from sentence_transformers import SentenceTransformer
import chromadb
from pathlib import Path

class FileMemoryIndexer:
    def __init__(self):
        self.model = SentenceTransformer(
            "all-MiniLM-L6-v2"
        )

        self.client = chromadb.PersistentClient(
            path="./file_memory_db"
        )

        self.collection = self.client.get_or_create_collection(
            "filesystem_memory"
        )

    def index_text_file(self, path: str):
        p = Path(path)

        if not p.exists():
            return

        try:
            text = p.read_text(
                encoding="utf-8",
                errors="ignore"
            )

            embedding = self.model.encode(text).tolist()

            self.collection.add(
                documents=[text[:5000]],
                embeddings=[embedding],
                ids=[str(path)],
                metadatas=[{
                    "path": str(path),
                    "filename": p.name
                }]
            )

        except Exception as e:
            print(e)

    def semantic_search(self, query: str, top_k=5):
        emb = self.model.encode(query).tolist()

        results = self.collection.query(
            query_embeddings=[emb],
            n_results=top_k
        )

        return results

            return self.fs.read_file(task["path"])

                task["path"],

            )

                task["query"]
            )

            self.fs.open_path(task["path"])
            return "opened"

        return {"status": "not_found", "path": path}  # file not accessible

from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler
from pathlib import Path
import time

class LiveFileHandler(FileSystemEventHandler):

    def __init__(self, indexer):
        self.indexer = indexer

    def on_created(self, event):
        if not event.is_directory:
            self.process(event.src_path)

    def on_modified(self, event):
        if not event.is_directory:
            self.process(event.src_path)

    def process(self, path):
        ext = Path(path).suffix.lower()

        if ext in [
            ".txt", ".md", ".py", ".json",
            ".pdf", ".docx"
        ]:
            print(f"[WATCHER] Indexing {path}")
            self.indexer.index_file(path)

class LiveWatcher:

    def __init__(self, indexer):
        self.indexer = indexer
        self.observer = Observer()

    def watch(self, path):
        handler = LiveFileHandler(self.indexer)

        self.observer.schedule(
            handler,
            path,
            recursive=True
        )

        self.observer.start()

        print(f"[WATCHING] {path}")

        try:
            while True:
                time.sleep(1)
        except KeyboardInterrupt:
            self.observer.stop()

        self.observer.join()

import threading
from pathlib import Path

class AutoIndexer:

    def __init__(self, indexer):
        self.indexer = indexer

    def scan_directory(self, root):

        for path in Path(root).rglob("*"):

            if path.is_file():

                ext = path.suffix.lower()

                if ext in [
                    ".txt", ".md", ".pdf",
                    ".docx", ".py", ".json"
                ]:
                    try:
                        self.indexer.index_file(str(path))
                    except Exception as e:
                        print(e)

    def background_scan(self, root):

        thread = threading.Thread(
            target=self.scan_directory,
            args=(root,),
            daemon=True
        )

        thread.start()

import pytesseract
from PIL import Image
import fitz

class OCREngine:

    def image_to_text(self, path):

        image = Image.open(path)

        text = pytesseract.image_to_string(image)

        return text

    def pdf_to_text(self, path):

        doc = fitz.open(path)

        full_text = ""

        for page in doc:

            text = page.get_text()

            if not text.strip():

                pix = page.get_pixmap()

                img_path = "temp_page.png"

                pix.save(img_path)

                text += self.image_to_text(img_path)

            full_text += text

        return full_text

import subprocess
import tempfile
import os

class SandboxedExecutor:

    def execute_python(self, code: str):

        with tempfile.NamedTemporaryFile(
            suffix=".py",
            delete=False,
            mode="w",
            encoding="utf-8"
        ) as f:

            f.write(code)

            temp_path = f.name

        try:

            result = subprocess.run(
                ["python", temp_path],
                capture_output=True,
                text=True,
                timeout=10
            )

            return {
                "stdout": result.stdout,
                "stderr": result.stderr,
                "returncode": result.returncode
            }

        except subprocess.TimeoutExpired:

            return {
                "error": "Execution timeout."
            }

        finally:

            os.remove(temp_path)

    def merge_results(self, outputs):
        merged = {}
        for output in outputs:
            for key, value in output.items():
                if key in merged:
                    if isinstance(merged[key], list) and isinstance(value, list):
                        merged[key].extend(value)
                    elif isinstance(merged[key], dict) and isinstance(value, dict):
                        merged[key].update(value)

import asyncio

class ConsciousLoop:

    def __init__(self, cognition):

        self.cognition = cognition
        self.running = True

    async def run(self):

        while self.running:

            await self.observe()

            await self.reflect()

            await self.plan()

            await asyncio.sleep(1)

    async def observe(self):
        observations = {}
        for source in self.sources:
            observations[source] = await self._poll_source(source)
        self._observation_cache = observations
        return observations

    async def reflect(self):
        if not self._observation_cache:
            return {"insights": [], "summary": "no data"}
        insights = self._analyze_patterns(self._observation_cache)
        return {"insights": insights, "summary": self._summarize(insights)}

    async def plan(self):
        reflection = await self.reflect()
        if not reflection["insights"]:
            return {"actions": ["continue_monitoring"]}
        actions = [self._derive_action(i) for i in reflection["insights"]]
        return {"actions": actions, "based_on": reflection["summary"]}

This becomes the “always awake” layer.
