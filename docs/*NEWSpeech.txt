import time

_last_action_time = 0
_action_count = []

def should_act(decision, context):
    global _last_action_time, _action_count

    now = time.time()

    # Cooldown
    if now - _last_action_time < 60:
        return False

    # Max per hour
    _action_count = [t for t in _action_count if now - t < 3600]
    if len(_action_count) >= 3:
        return False

    # Don’t interrupt active user
    if now - context["last_user_time"] < 20:
        return False

    return decision.get("should_act", False)

from typing import List
from fastapi import WebSocket

class ConnectionManager:
    def __init__(self):
        self.active_connections: List[WebSocket] = []

    async def connect(self, websocket: WebSocket):
        await websocket.accept()
        self.active_connections.append(websocket)

    def disconnect(self, websocket: WebSocket):
        self.active_connections.remove(websocket)

    async def broadcast(self, message: dict):
        for connection in self.active_connections:
            await connection.send_json(message)

def train(batch):
    states = batch["states"]
    actions = batch["actions"]
    rewards = batch["rewards"]

    probs = policy(states)

    dist = torch.distributions.Categorical(probs)
    log_probs = dist.log_prob(actions)

    advantage = rewards - rewards.mean()

    loss = -(log_probs * advantage).mean()

    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

class Trainer:

    def train_batch(self, batch):
        loss_total = 0

        for s, a, s_next in batch:
            s = torch.tensor(s)
            a = torch.tensor(a)
            s_next = torch.tensor(s_next)

            pred = self.model(s, a)
            loss = ((pred - s_next) ** 2).mean()

            self.opt.zero_grad()
            loss.backward()
            self.opt.step()

            loss_total += loss.item()

        return loss_total

class MuZeroCore:
    def __init__(self, policy, dynamics, reward_model):
        self.policy = policy
        self.dynamics = dynamics
        self.reward = reward_model

    def predict(self, state):
        action_probs, value = self.policy(state)
        return action_probs, value

    def simulate(self, state, action):
        next_state = self.dynamics(state, action)
        reward = self.reward(state, action)
        return next_state, reward

            path.append((node, action))
            node = node.children[action]

        # EXPAND
        probs, value = model.predict(node.state)
        node.expand(probs)

        # SIMULATE
        next_state, reward = model.simulate(node.state, action)

        # BACKUP
        G = value + reward
        for parent, act in reversed(path):
            parent.update(act, G)

    return root.best_action()

import torch
import torch.nn as nn

class WorldModel(nn.Module):
    def __init__(self, dim, action_dim):
        super().__init__()

        self.dynamics = nn.Sequential(
            nn.Linear(dim + action_dim, 256),
            nn.ReLU(),
            nn.Linear(256, dim)
        )

        self.reward_head = nn.Sequential(
            nn.Linear(dim + action_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 1)
        )

        self.termination_head = nn.Sequential(
            nn.Linear(dim, 64),
            nn.ReLU(),
            nn.Linear(64, 1),
            nn.Sigmoid()
        )

    def step(self, state, action):
        next_state = self.dynamics(torch.cat([state, action], dim=-1))
        reward = self.reward_head(torch.cat([state, action], dim=-1))
        done = self.termination_head(next_state)
        return next_state, reward, done
