class IntentClassifier:
    def classify(self, text: str):
        text = text.lower()
        score = {"support": 0, "coding": 0, "study": 0, "planner": 0,
                 "conversation": 0.2, "joking": 0, "venting": 0, "testing": 0}

        support_kw = ["help", "sad", "tired", "lost", "depressed", "anxious", "lonely", "hurt"]
        coding_kw = ["code", "error", "bug", "function", "api", "syntax", "debug", "server"]
        study_kw = ["study", "explain", "learn", "teach", "tutorial", "understand", "concept"]
        planner_kw = ["plan", "schedule", "todo", "organize", "agenda", "deadline", "remind"]
        joking_kw = ["haha", "lol", "joke", "funny", "😂", "😄", "just kidding"]
        venting_kw = ["fucking", "hate", "stupid", "annoying", "fed up", "sick of"]
        testing_kw = ["are you", "can you", "test", "prove", "what are you"]

        kw_map = [
            (support_kw, "support", 0.4),
            (coding_kw, "coding", 0.5),
            (study_kw, "study", 0.4),
            (planner_kw, "planner", 0.4),
            (joking_kw, "joking", 0.5),
            (venting_kw, "venting", 0.4),
            (testing_kw, "testing", 0.3),
        ]

        for keywords, intent, weight in kw_map:
            for kw in keywords:
                if kw in text:
                    score[intent] += weight

        return max(score, key=score.get)

class ResponseValidator:
    def validate(self, simple, advanced):
        if len(advanced) > 2 * len(simple):
            return simple  # too complex

        if self.similarity(simple, advanced) < 0.5:
            return simple  # fallback

        return advanced

    def similarity(self, a, b):
        return len(set(a.split()) & set(b.split())) / max(len(a.split()), 1)

import asyncio

class StreamManager:
    def __init__(self):
        self.current_task = None

    async def start_stream(self, coro):
        if self.current_task:
            self.current_task.cancel()

        self.current_task = asyncio.create_task(coro)

        try:
            return await self.current_task
        except asyncio.CancelledError:
            return {"status": "cancelled", "partial": None}

    def interrupt(self):
        if self.current_task:
            self.current_task.cancel()
