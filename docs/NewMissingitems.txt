Cognitive Loop v1

Implement this before any new modules.

async def cognitive_cycle():

    perception = await perceive()

    attention = await attention_engine.focus(
        perception
    )

    memories = await memory_engine.retrieve(
        attention
    )

    goals = await goal_manager.get_active()

    emotions = await emotion_reasoner.analyze()

    thought = await thought_fusion.merge(
        attention,
        memories,
        goals,
        emotions
    )

    plan = await planner.plan(thought)

    response = await executor.execute(plan)

    reflection = await reflector.evaluate(
        response
    )

    await learner.learn(reflection)

This is the first real "mind loop."
