3️⃣ Latency-Aware Turn-Taking

Your problem:

🎯 Fix

Add latency buffer before AI speaks.

Server — turn-taking delay

In turn_taking.py:

def compute_speech_delay(latency_ms: float):
    base_delay = 120  # natural human pause
    network_buffer = latency_ms * 0.5
    return min(600, base_delay + network_buffer)

Before sending TTS:

await asyncio.sleep(compute_speech_delay(session.latency) / 1000)

Effect:

50 ms	~145 ms

using System;
using System.Collections;
using System.Text;
using UnityEngine;
using WebSocketSharp;

[Serializable]
public class NetworkMetrics
{
    public float rtt;
    public float jitter;
    public float packetLoss;
    public float latencyEMA;
    public string connectionType;
}

[Serializable]
public class DeviceInfo
{
    public string type;
    public string os;
    public string model;
    public string appVersion;
}

[Serializable]
public class SessionPayload
{
    public string sessionId;
    public DeviceInfo device;
    public NetworkMetrics network;
}

public class TelemetryClient : MonoBehaviour
{
    public string serverUrl = "ws://localhost:8000/ws";
    private WebSocket ws;

    private float lastPingTime;
    private float emaLatency = 0f;
    private const float alpha = 0.2f;
    private bool pongReceived = false;

    private string sessionId;

    void Start()
    {
        sessionId = Guid.NewGuid().ToString();

        ws = new WebSocket(serverUrl);

        ws.OnOpen += (s, e) =>
        {
            SendSessionRegister();
            StartCoroutine(PingLoop());
            StartCoroutine(HeartbeatLoop());
        };

        ws.OnMessage += (s, e) =>
        {
            if (e.Data.Contains("pong"))
            {
                float rtt = (Time.realtimeSinceStartup - lastPingTime) * 1000f;
                UpdateLatency(rtt);
                pongReceived = true;
            }
            else if (e.Data.Contains("qos"))
            {
                ApplyQoS(e.Data);
            }
        };

        ws.Connect();
    }

    void SendSessionRegister()
    {
        var payload = new SessionPayload
        {
            sessionId = sessionId,
            device = new DeviceInfo
            {
                type = Application.isMobilePlatform ? "mobile" : "desktop",
                os = SystemInfo.operatingSystem,
                model = SystemInfo.deviceModel,
                appVersion = Application.version
            },
            network = new NetworkMetrics
            {
                connectionType = GetConnectionType()
            }
        };

        ws.Send(JsonUtility.ToJson(payload));
    }

    IEnumerator PingLoop()
    {
        while (true)
        {
            lastPingTime = Time.realtimeSinceStartup;
            pongReceived = false;

            ws.Send("{\"type\":\"ping\"}");

            yield return new WaitUntil(() => pongReceived);
            yield return new WaitForSeconds(2f);
        }
    }

    IEnumerator HeartbeatLoop()
    {
        while (true)
        {
            var metrics = new NetworkMetrics
            {
                rtt = emaLatency,
                latencyEMA = emaLatency,
                connectionType = GetConnectionType()
            };

            ws.Send(JsonUtility.ToJson(new
            {
                type = "heartbeat",
                sessionId = sessionId,
                network = metrics
            }));

            yield return new WaitForSeconds(3f);
        }
    }

    void UpdateLatency(float rtt)
    {
        if (emaLatency == 0) emaLatency = rtt;
        else emaLatency = alpha * rtt + (1 - alpha) * emaLatency;
    }

    string GetConnectionType()
    {
        switch (Application.internetReachability)
        {
            case NetworkReachability.ReachableViaLocalAreaNetwork:
                return "wifi";
            case NetworkReachability.ReachableViaCarrierDataNetwork:
                return "cellular";
            default:
                return "offline";
        }
    }

    void ApplyQoS(string json)
    {
        Debug.Log("QoS Update: " + json);
        const chunkSize = config.ttsChunkSize || 1024;
      const smoothing = config.ttsSmoothing || 0.3;
      const turnDelay = config.ttsTurnDelay || 500;
      applyTtsParams(chunkSize, smoothing, turnDelay)
    }

def compute_qos(rtt_ema: float):
    if rtt_ema < 80:
        return {"tier": "LOW", "ttsChunkMs": 60, "turnDelayMs": 0, "emotionSmoothing": 0.6}
    elif rtt_ema < 180:
        return {"tier": "MEDIUM", "ttsChunkMs": 100, "turnDelayMs": 120, "emotionSmoothing": 0.75}
    else:
        return {"tier": "HIGH", "ttsChunkMs": 140, "turnDelayMs": 260, "emotionSmoothing": 0.9}

# ---------------- Broadcast ---------------- #

async def broadcast_dashboard():
    payload = json.dumps({
        "type": "sessions",
        "data": list(sessions.values())
    })
    for ws in dashboard_sockets:
        await ws.send_text(payload)

# ---------------- TTS Scheduler ---------------- #

def schedule_tts(session_id: str):
    session = sessions.get(session_id)
    if not session:
        return {"status": "no_data", "session": session_id}

    ema = session["network"].get("latencyEMA", 0)
    server_now = int(time.time() * 1000)

    return {
        "type": "tts_schedule",
        "ttsStartAt": server_now + int(ema),
        "visemeOffsetMs": int(ema)
    }

@app.websocket("/ws")
async def websocket_endpoint(ws: WebSocket):
    await ws.accept()

    role = None
    session_id = None

    try:
        while True:
            data = await ws.receive_text()
            msg = json.loads(data)

            # -------- DASHBOARD CONNECT -------- #
            if msg.get("type") == "dashboard_connect":
                role = "dashboard"
                dashboard_sockets.add(ws)
                await broadcast_dashboard()
                continue

            # -------- SESSION REGISTER -------- #
            if msg.get("type") == "session_register":
                role = "client"
                session_id = msg["sessionId"]

                client_sockets[session_id] = ws

                sessions[session_id] = {
                    "sessionId": session_id,
                    "device": msg.get("device", {}),
                    "network": msg.get("network", {}),
                    "lastSeen": time.time(),
                    "qos": "LOW"
                }

                await broadcast_dashboard()
                continue

            # -------- PING / PONG -------- #
            if msg.get("type") == "ping":
                await ws.send_text(json.dumps({"type": "pong"}))
                continue

            # -------- HEARTBEAT -------- #
            if msg.get("type") == "heartbeat":
                session_id = msg["sessionId"]
                net = msg.get("network", {})

                session = sessions.get(session_id)
                if not session:
                    continue

                # EMA update
                rtt = net.get("rtt", 0)
                prev_ema = session["network"].get("latencyEMA", rtt)
                ema = ALPHA * rtt + (1 - ALPHA) * prev_ema

                session["network"].update(net)
                session["network"]["latencyEMA"] = ema
                session["lastSeen"] = time.time()

                # QoS compute
                qos = compute_qos(ema)
                session["qos"] = qos["tier"]

                # Send QoS to client
                await ws.send_text(json.dumps({
                    "type": "qos",
                    **qos
                }))

                await broadcast_dashboard()
                continue

            # -------- TTS REQUEST (optional trigger) -------- #
            if msg.get("type") == "request_tts_schedule":
                schedule = schedule_tts(msg["sessionId"])
                if schedule:
                    await ws.send_text(json.dumps(schedule))

    except WebSocketDisconnect:
        if role == "dashboard":
            dashboard_sockets.discard(ws)
        elif session_id:
            client_sockets.pop(session_id, None)
            sessions.pop(session_id, None)
            await broadcast_dashboard()

2️⃣ Audio Thread Buffering

📄 AudioStreamPlayer.cs
using UnityEngine;
using System.Collections.Generic;

public class AudioStreamPlayer : MonoBehaviour
{
    public int sampleRate = 24000;
    private Queue<float> audioQueue = new Queue<float>();

    private AudioSource audioSource;
    private double dspStartTime;
    private bool started = false;

    void Start()
    {
        audioSource = gameObject.AddComponent<AudioSource>();
        audioSource.clip = AudioClip.Create(
            "TTSStream",
            sampleRate * 10,
            1,
            sampleRate,
            true,
            OnAudioRead
        );
        audioSource.loop = true;
        audioSource.Play();
    }

    public void EnqueueSamples(float[] samples)
    {
        lock (audioQueue)
        {
            foreach (var s in samples)
                audioQueue.Enqueue(s);
        }

        if (!started && audioQueue.Count > sampleRate * 0.15f) // 150 ms buffer
        {
            dspStartTime = AudioSettings.dspTime;
            started = true;
        }
    }

    void OnAudioRead(float[] data)
    {
        lock (audioQueue)
        {
            for (int i = 0; i < data.Length; i++)
            {
                if (audioQueue.Count > 0)
                    data[i] = audioQueue.Dequeue();
                else
                    data[i] = 0f; // PLC silence fallback
            }
        }
    }

    public double GetAudioTime()
    {
        if (!started) return 0;
        return AudioSettings.dspTime - dspStartTime;
    }
}

Now audio playback is:

buffered

DSP-timed

immune to frame drops

📄 VisemeJob.cs
using Unity.Burst;
using Unity.Collections;
using Unity.Jobs;
using Unity.Mathematics;

[BurstCompile]
public struct VisemeInterpolationJob : IJob
{
    [ReadOnly] public NativeArray<VisemeFrame> frames;
    public float audioTime;

    public NativeArray<float> outputWeights;

    public void Execute()
    {
        for (int i = 0; i < frames.Length - 1; i++)
        {
            if (audioTime >= frames[i].time && audioTime <= frames[i + 1].time)
            {
                float t = math.unlerp(frames[i].time, frames[i + 1].time, audioTime);

                for (int j = 0; j < outputWeights.Length; j++)
                {
                    outputWeights[j] = math.lerp(
                        frames[i].weights[j],
                        frames[i + 1].weights[j],
                        t
                    );
                }

[BurstCompile]
public struct PCAReconstructJob : IJob
{
    [ReadOnly] public NativeArray<float> pcaWeights;        // 10
    [ReadOnly] public NativeArray<float> pcaMatrix;         // 10 x 52
    public NativeArray<float> blendshapeWeights;            // 52

    public void Execute()
    {
        for (int i = 0; i < 52; i++)
        {
            float value = 0f;
            for (int j = 0; j < 10; j++)
            {
                value += pcaWeights[j] * pcaMatrix[j * 52 + i];
            }
            blendshapeWeights[i] = value;
        }
    }
}

You just cut CPU math by 5×–10×.

public class BlendshapeDriver : MonoBehaviour
{
    public SkinnedMeshRenderer smr;
    private float[] lastWeights;

    void Start()
    {
        lastWeights = new float[smr.sharedMesh.blendShapeCount];
    }

    public void ApplyWeights(NativeArray<float> weights)
    {
        for (int i = 0; i < weights.Length; i++)
        {
            float w = weights[i] * 100f;

            if (math.abs(w - lastWeights[i]) > 0.5f) // threshold
            {
                smr.SetBlendShapeWeight(i, w);
                lastWeights[i] = w;
            }
        }
    }
}

You avoid redundant GPU uploads.
