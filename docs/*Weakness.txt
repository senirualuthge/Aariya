Event bus:

class EventBus:

    def __init__(self):

        self.subscribers={}

    def subscribe(self,event,handler):

        self.subscribers.setdefault(
            event,
            []
        ).append(handler)

    def publish(
        self,
        event
    ):

        for handler in self.subscribers.get(
            type(event),
            []
        ):
            handler(event)

class AttentionController:

    def prioritize(

        self,
        state
    ):

        scores=[]

        for item in state:

            score=(
                item.importance*
                item.novelty*
                item.urgency*
                item.uncertainty
            )

            scores.append(
                (item,score)
            )

        return sorted(

            scores,

            reverse=True
        )
