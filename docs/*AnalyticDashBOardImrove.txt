function getLastTrust(sessions) {
  return sessions.at(-1)?.trust || 0;
}

function getMoodTrend(sessions) {
  const last = sessions.slice(-3);
  const avg = last.reduce((a, s) => a + s.emotionScore, 0) / last.length;

  if (avg > 0.6) return "Positive";
  if (avg < 0.4) return "Negative";
  return "Neutral";
}

function getDropoffRisk(sessions) {
  const last = sessions.at(-1);
  if (!last) return "LOW";

  if (last.trust < 0.4) return "HIGH";
  if (last.trust < 0.7) return "MEDIUM";
  return "LOW";
}

// src/insightEngine/signals.js

export function extractSignals(sessions) {
  const last5 = sessions.slice(-5);

  const avgTrust = avg(last5.map(s => s.trust));
  const trustTrend = trend(last5.map(s => s.trust));

  const avgEmotion = avg(last5.map(s => s.emotionScore));
  const emotionTrend = trend(last5.map(s => s.emotionScore));

  const contradictionRate = avg(last5.map(s => s.contradiction || 0));

  return {
    avgTrust,
    trustTrend,
    avgEmotion,
    emotionTrend,
    contradictionRate
  };
}

function avg(arr) {
  if (!arr.length) return 0;
  return arr.reduce((a, b) => a + b, 0) / arr.length;
}

function trend(arr) {
  if (arr.length < 2) return 0;
  return arr[arr.length - 1] - arr[0]; // simple slope
}

export function detectPatterns(signals) {
  const patterns = [];

  if (signals.trustTrend < -0.2) {
    patterns.push("TRUST_DECLINE");
  }

  if (signals.emotionTrend < -0.2) {
    patterns.push("EMOTIONAL_DROP");
  }

  if (signals.contradictionRate > 0.5) {
    patterns.push("HIGH_CONTRADICTION");
  }

  if (signals.avgTrust < 0.4 && signals.contradictionRate > 0.4) {
    patterns.push("TRUST_BREAK_RISK");
  }

  return patterns;
}

// src/cognitiveRouter/applyActions.js

export function applyActionsToState(state, actions) {
  const newState = { assertiveness: state.assertiveness, empathy: state.empathy, intent: state.intent, energy: state.energy, lastIntentChange: state.lastIntentChange };

  actions.forEach(action => {
    switch (action.type) {
      case "REDUCE_ASSERTIVENESS":
        newState.assertiveness *= (1 - action.value);
        break;

      case "INCREASE_EMPATHY":
        newState.empathy += action.value;
        break;

      case "SWITCH_TO_COMFORT_MODE":
        newState.intent = "COMFORT";
        break;

      case "ENABLE_CLARIFICATION_MODE":
        newState.intent = "CLARIFY";
        break;

      case "MAXIMIZE_SAFETY_RESPONSE":
        newState.intent = "SAFE";
        newState.empathy = 1;
        break;
    }
  });

  return newState;
}

export function predictNextState(signals) {
  const { trustTrend, emotionTrend, contradictionRate, avgTrust } = signals;

  let predictedTrust = avgTrust + trustTrend * 0.8 - contradictionRate * 0.3;

  let risk = "LOW";
  if (predictedTrust < 0.4) risk = "HIGH";
  else if (predictedTrust < 0.7) risk = "MEDIUM";

  let mood = "Neutral";
  if (emotionTrend > 0.1) mood = "Improving";
  else if (emotionTrend < -0.1) mood = "Declining";

  return {
    trust: predictedTrust.toFixed(2),
    mood,
    risk
  };
}

export function updateThresholds(profile, outcomes) {
  let { thresholds } = profile;

  outcomes.forEach(o => {
    const trustChange = o.trustAfter - o.trustBefore;

    // If trust dropped → thresholds too loose
    if (trustChange < -0.1) {
      thresholds.trustDrop += 0.05;
    }

    // If trust improved → thresholds working
    if (trustChange > 0.1) {
      thresholds.trustDrop -= 0.02;
    }
  });

  // Clamp
  thresholds.trustDrop = clamp(thresholds.trustDrop, -1, 0);

  return thresholds;
}

import {
  LineChart, Line, XAxis, YAxis, Tooltip, ResponsiveContainer

const MAX_POINTS = 60

export default function AnalyticsDashboard() {
  const socketRef = useRef(null)

  const [metrics, setMetrics] = useState([])
  const [brain, setBrain] = useState({})
  const [perf, setPerf] = useState({})
  const [logs, setLogs] = useState([])
  const [config, setConfig] = useState({
    emotion: true,
    memory: "high",
    drift: 0.02
  })

  // 🔌 WebSocket Connection
  useEffect(() => {
    const socket = new WebSocket("ws://localhost:8000/ws")
    socketRef.current = socket

    socket.onmessage = (event) => {
      const data = JSON.parse(event.data)

      switch (data.type) {
        case "analytics.delta":
          setMetrics(prev => {
            const kept = prev.slice(-MAX_POINTS);
            kept.push({ time: Date.now(), ECS: data.metrics.ECS, PSI: data.metrics.PSI, latency: data.metrics.latency });
            return kept;
          })
          break

        case "brain.state":
          setBrain(data.brain_state)
          break

        case "perf.delta":
          setPerf(data)
          break

        case "debug.log":
          setLogs(prev => {
            const updated = [{ time: new Date().toLocaleTimeString(), msg: data.message }];
            const head = prev.slice(0, 50);
            for (let i = 0; i < head.length; i++) updated.push(head[i]);
            return updated;
          })
          break

        default:
          break
      }
    }

    return () => socket.close()
  }, [])

  // 🎛️ Send config updates
  const updateConfig = (newConfig) => {
    const updated = {

    window.append(value)

    if len(window) > 30:
        window.pop(0)

    mean = np.mean(window)
    std = np.std(window)

# detector.py
import numpy as np

class AnomalyDetector:
    def __init__(self, window_size=50):
        self.windows = {}
        self.window_size = window_size

    def update(self, metric, value):
        if metric not in self.windows:
            self.windows[metric] = []

        w = self.windows[metric]
        w.append(value)

        if len(w) > self.window_size:
            w.pop(0)

        if len(w) < 10:
            return {"status": "insufficient_data", "samples": len(w)}

        mean = np.mean(w)
        std = np.std(w) or 1e-5

        z = (value - mean) / std

        if abs(z) > 3:
            return {
                "metric": metric,
                "value": value,
                "z_score": z,
                "severity": "high" if abs(z) > 5 else "medium",
                "status": "anomaly"
            }

        return {"status": "normal", "z_score": round(float(z), 2)}
