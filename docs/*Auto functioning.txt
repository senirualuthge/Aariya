base_adapter.py
class BaseAdapter:
    def open_app(self, name): pass
    def open_url(self, url): pass
    def get_active_window(self): pass
    def list_installed_apps(self): pass
mac_adapter.py
import subprocess

class MacAdapter(BaseAdapter):
    def open_app(self, name):
        subprocess.Popen(["open", "-a", name])
windows_adapter.py
import subprocess

class WindowsAdapter(BaseAdapter):
    def open_app(self, name):
        subprocess.Popen(["start", name], shell=True)
linux_adapter.py
import subprocess

class LinuxAdapter(BaseAdapter):
    def open_app(self, name):
        subprocess.Popen([name])
🔹 4️⃣ Platform Detection

On startup:

import platform

def get_adapter():
    os_name = platform.system()

    if os_name == "Darwin":
        return MacAdapter()
    elif os_name == "Windows":
        return WindowsAdapter()
    elif os_name == "Linux":
        return LinuxAdapter()
    else:
        raise Exception("Unsupported OS")

Now everything becomes modular.
