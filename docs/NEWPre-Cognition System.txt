def governance_evaluate(before, after):

    risk = 0

    if after["accuracy"] < before["accuracy"]:
        risk += 0.5

    if after["latency"] > before["latency"] * 1.2:
        risk += 0.3

    if after.get("instability", 0) > 0.2:
        risk += 0.4

    if risk > 0.5:
        return "REJECT"

    elif risk > 0.2:
        return "REVIEW"

    return "APPROVE"

class CognitiveScheduler:

        self.queue = []

        heapq.heappush(

            (-task.priority, task)

        if not self.queue:

        return heapq.heappop(self.queue)[1]
