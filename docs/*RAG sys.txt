🔹 Step 3 — Vector Store
For MVP: cosine similarity in Python.
vector_store.py


import numpy as np
import json

def cosine_similarity(a, b):
    a = np.array(a)
    b = np.array(b)
    return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))

class VectorStore:
    def search(self, query_embedding, memory_rows, top_k=5):
        scored = []

        for mem in memory_rows:
            embedding = json.loads(mem["embedding"])
            score = cosine_similarity(query_embedding, embedding)
            scored.append((score, mem))

        scored.sort(reverse=True, key=lambda x: x[0])
        return scored[:top_k]

import math
from datetime import datetime

def recency_score(timestamp):
    days = (datetime.utcnow() - timestamp).days
    return math.exp(-days / 30)  # 30-day half-life

def rerank(results):
    reranked = []

    for sim, mem in results:
        score = (
            0.5 * sim +
            0.2 * mem["importance"] +
            0.2 * recency_score(mem["timestamp"]) +
            0.1 * mem["emotional_weight"]
        )
        reranked.append((score, mem))

    reranked.sort(reverse=True, key=lambda x: x[0])
    return [m for _, m in reranked]


🔹 Step 5 — RAG Pipeline
rag_pipeline.py


class RAGPipeline:
    def __init__(self, memory_system):
        self.memory = memory_system
        self.embedder = EmbeddingManager()
        self.vector_store = VectorStore()

    def retrieve(self, query_text, trust_score):
        query_emb = self.embedder.embed(query_text)

        # Load candidate memories from DB
        memories = self.memory.load_all_embeddings()

        top_similar = self.vector_store.search(query_emb, memories, top_k=20)

        reranked = rerank(top_similar)

        # Trust-gated recall
        quota = self.calculate_quota(trust_score)

        return reranked[:quota]

    def calculate_quota(self, trust):
        if trust < 0.3:
            return 1
        elif trust < 0.6:
            return 3
        else:
            return 6

3️⃣ Hybrid Retrieval (Vector + Keyword)
SQL Query (pgvector + BM25)


WITH vector_search AS (
  SELECT id, content,
         1 - (embedding <=> %(query_embedding)s) AS v_score
  FROM memories
  WHERE user_id = %(user_id)s
  ORDER BY embedding <=> %(query_embedding)s
  LIMIT 20
),
keyword_search AS (
  SELECT id, content,
         ts_rank_cd(to_tsvector(content), plainto_tsquery(%(query)s)) AS k_score
  FROM memories
  WHERE user_id = %(user_id)s
    AND to_tsvector(content) @@ plainto_tsquery(%(query)s)
  LIMIT 20
)
SELECT m.*,
       COALESCE(v.v_score,0) * 0.6 +
       COALESCE(k.k_score,0) * 0.4 AS score
FROM memories m
LEFT JOIN vector_search v ON m.id = v.id
LEFT JOIN keyword_search k ON m.id = k.id
ORDER BY score DESC
LIMIT 20;

# server/systems/memory.py
from datetime import datetime
from .rag.embedding_manager import EmbeddingManager
from .rag.db import get_pg_conn

class MemorySystem:
    def __init__(self, user_id: str):
        self.user_id = user_id
        self.embedder = EmbeddingManager()
        self.db = get_pg_conn()

    def store(self, content, mem_type="turn",
              importance=0.5, emotion=0.5, confidence=1.0,
              parent_id=None):

        emb = self.embedder.embed(content)

        with self.db.cursor() as cur:
            cur.execute("""
              INSERT INTO memories
              (user_id,type,content,embedding,importance,
               emotional_weight,confidence)
              VALUES (%s,%s,%s,%s,%s,%s,%s)
              RETURNING id
            """, (
                self.user_id, mem_type, content,
                emb, importance, emotion, confidence
            ))
            mem_id = cur.fetchone()[0]

            if parent_id:
                cur.execute("""
                  INSERT INTO memory_edges
                  (from_id,to_id,relation)
                  VALUES (%s,%s,'next_turn')
                """, (parent_id, mem_id))

        self.db.commit()
        return mem_id

    def retrieve(self, query, trust):
        q_emb = self.embedder.embed(query)

        with self.db.cursor() as cur:
            cur.execute(HYBRID_QUERY, {
                "query_embedding": q_emb,
                "query": query,
                "user_id": self.user_id
            })
            rows = cur.fetchall()

        quota = 1 if trust < 0.3 else 3 if trust < 0.6 else 6
        return rows[:quota]
