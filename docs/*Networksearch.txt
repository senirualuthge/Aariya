import requests
from bs4 import BeautifulSoup

def fetch_page_text(url):
    headers = {"User-Agent": "Mozilla/5.0"}
    response = requests.get(url, headers=headers, timeout=10)
    soup = BeautifulSoup(response.text, "html.parser")

    # Remove scripts and styles
    for tag in soup(["script", "style"]):
        tag.decompose()

    return soup.get_text(separator=" ", strip=True)

def run_agent(question):

    messages = [
        {"role": "system", "content": "You are a research agent. Use tools when needed."},
        {"role": "user", "content": question}
    ]

    while True:

        response = local_llm.chat(messages, tools=TOOL_SCHEMA)

        if response.tool_call:
            tool_name = response.tool_call["name"]
            args = response.tool_call["arguments"]

            result = TOOLS[tool_name](**args)

            messages.append({
                "role": "tool",
                "name": tool_name,
                "content": result
            })

        else:
            return response.content

class CitationManager:
    def __init__(self):

def score_source(url, content):
    score = 0

    if ".gov" in url or ".edu" in url:
        score += 0.4

    if "wikipedia" in url:
        score += 0.2

    if len(content) > 2000:
        score += 0.1

    if "opinion" in url:
        score -= 0.2

    return max(min(score, 1.0), 0.0)

def compute_confidence(evidence_list, contradictions):

    avg_credibility = sum(
        e["credibility_score"] for e in evidence_list
    ) / len(evidence_list)

    contradiction_penalty = 0.3 if contradictions else 0.0

    source_diversity = len(set(e["source_url"] for e in evidence_list)) * 0.05

    confidence = avg_credibility + source_diversity - contradiction_penalty

    return max(min(confidence, 1.0), 0.0)

Step 2: Compare events

def detect_temporal_conflicts(claims):
    event_map = {}

    conflicts = []

    for c in claims:
        event = c["event"]
        date = c["date"]

        if event in event_map and event_map[event] != date:
            conflicts.append({
                "event": event,
                "dates": [event_map[event], date]
            })
        else:
            event_map[event] = date

    return conflicts


If conflict found:

def simulate_counterfactual(graph, cause, effect):
    if graph.has_edge(cause, effect):
        weight = graph[cause][effect]["weight"]

        # Remove causal link
        graph.remove_edge(cause, effect)

        # Recompute effect probability
        counterfactual_probability = 1 - weight

        return counterfactual_probability

    score += 0.2 if "wikipedia" in url else 0
