class EngineRegistry:

    def __init__(self):
        self.engines = {}

    def register(self, engine):
        self.engines[engine.name] = engine

    def get(self, name):
        return self.engines.get(name)

    def get_many(self, names):
        return [
            self.engines[n]
            for n in names
            if n in self.engines
        ]

    def available(self):
        return list(self.engines.keys())

class RuntimeOrchestrator:

    def __init__(
        self,
        executive,
        registry
    ):
        self.executive = executive
        self.registry = registry

    async def process(
        self,
        user_input,
        context
    ):

        decision = self.executive.decide(
            user_input
        )

        results = {}

        engines = self.registry.get_many(
            decision["engines"]
        )

        for engine in engines:

            output = await engine.run(
                context
            )

            results[
                engine.name
            ] = output

        return {
            "task":
                decision["task"],

            "results":
                results
        }
