  void connect(String url) {
    _channel = WebSocketChannel.connect(Uri.parse(url));
  }

  void sendMessage({
    String? text,
    Map<String, dynamic>? vision,
  }) {
    final payload = {
      "type": "input.multimodal",
      "timestamp": DateTime.now().millisecondsSinceEpoch,
      "text": text,
      "vision": vision,
    };

    _channel?.sink.add(jsonEncode(payload));
  }

  Stream<Map<String, dynamic>> get stream {
    return _channel!.stream.map((event) {
      return jsonDecode(event);
    });
  }

  void disconnect() {
    _channel?.sink.close();
  }
}

def store(self, text, emotion, trust_delta):
        embedding = self.embedder.encode(text).tolist()

        significance = (
            abs(emotion["valence"]) * 0.4
            + abs(trust_delta) * 0.4
        )

        if significance < 0.5:
            return  # skip weak memories

        self.collection.add(
            documents=[text],
            embeddings=[embedding],
            metadatas=[{
                "valence": emotion["valence"],
                "arousal": emotion["arousal"],
                "timestamp": time.time(),
                "id": str(uuid.uuid4())
            }]
        )

def retrieve(self, query, k=3):
        embedding = self.embedder.encode(query).tolist()

        results = self.collection.query(
            query_embeddings=[embedding],
            n_results=k
        )

        memories = []

        for i in range(len(results["documents"][0])):
            memories.append({
                "text": results["documents"][0][i],
                "valence": results["metadatas"][0][i]["valence"],
                "arousal": results["metadatas"][0][i]["arousal"]
            })

        return memories
