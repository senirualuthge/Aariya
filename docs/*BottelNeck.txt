🧠 1. Boot Profiler (TypeScript)
bootProfiler.ts
export type PerfTier = "LOW" | "MID" | "HIGH"

export interface BootProfile {
  cpuScore: number
  gpuFrameTime: number
  memoryGB: number
  cores: number
  tier: PerfTier
}

async function cpuBenchmark(): Promise<number> {
  const t0 = performance.now()
  let n = 0
  while (performance.now() - t0 < 50) {
    Math.sqrt(Math.random())
    n++
  }
  return n
}

async function gpuBenchmark(): Promise<number> {
  const canvas = document.createElement("canvas")
  const gl = canvas.getContext("webgl")
  if (!gl) return 999

  const frames = 60
  const t0 = performance.now()

  for (let i = 0; i < frames; i++) {
    gl.clear(gl.COLOR_BUFFER_BIT)
  }

  const t1 = performance.now()
  return (t1 - t0) / frames
}

function classifyTier(cpu: number, gpuMs: number, memory: number): PerfTier {
  if (cpu < 20000 || memory <= 4) return "LOW"
  if (gpuMs > 16) return "MID"
  return "HIGH"
}

export async function runBootProfiler(): Promise<BootProfile> {
  const [cpuScore, gpuFrameTime] = await Promise.all([
    cpuBenchmark(),
    gpuBenchmark(),
  ])

  const memoryGB = (navigator as any).deviceMemory || 4
  const cores = navigator.hardwareConcurrency || 4

  const tier = classifyTier(cpuScore, gpuFrameTime, memoryGB)

  return { cpuScore, gpuFrameTime, memoryGB, cores, tier }
}

let fps = 10

self.onmessage = async (e: MessageEvent) => {
  const { type, config } = e.data

  if (type === "INIT") {
    fps = config.faceTrackingHz

        expr.fearful,

    bitmap.close()
  }

  if (type === "START") running = true

Mapper Function
export function mapEmotionToBlend(
  emotion: EmotionVector,
  blendMap: BlendMap,
  prev: Record<string, number>,
  dt: number
) {
  const result: Record<string, number> = {}

  for (const key in blendMap) {
    const weights = blendMap[key]

    let target = 0
    for (let i = 0; i < emotion.length; i++) {
      target += emotion[i] * weights[i]
    }

    const current = prev[key] || 0

    // smoothing
    result[key] = current + (target - current) * Math.min(dt * 8, 1)
  }

  return result
}

    frameTimes.reduce((a, b) => a + b, 0) / (frameTimes.length || 1)

export function getMouthOpenness(analyser: AnalyserNode) {
  const data = new Uint8Array(analyser.fftSize)
  analyser.getByteTimeDomainData(data)

  let sum = 0
  for (let i = 0; i < data.length; i++) {
    const v = (data[i] - 128) / 128
    sum += v * v
  }

  return Math.sqrt(sum / data.length) // RMS
}

Viseme Solver
export function solveViseme(
  rms: number,
  phonemeHint: string | null,
  lod: "LOW" | "MID" | "HIGH",
  prev: Record<string, number>,
  dt: number
) {
  const visemes = VISEME_LOD[lod]
  const result: Record<string, number> = {}

  // base openness
  const open = Math.min(rms * 3, 1)

  for (const v of visemes) {
    let target = 0

    if (v === "sil") target = 1 - open
    else if (v === "open") target = open
    else if (phonemeHint === v) target = open
    else target = open * 0.3

    const current = prev[v] || 0
    result[v] = current + (target - current) * Math.min(dt * 12, 1)
  }

  return result
}

    int hz = GetMonitorHz();

    if (hz < 120)
    {
        Application.targetFrameRate = 60;
        yield break;
    }

    // Temporary unlock for test
    Application.targetFrameRate = -1;

    yield return new WaitForSeconds(2f); // warmup

    float avg = 0f;
    int samples = 120;

    for (int i = 0; i < samples; i++)
    {
        avg += Time.unscaledDeltaTime;
        yield return null;
    }

    avg /= samples;
    float fps = 1f / avg;

    if (fps >= 110f)
        Application.targetFrameRate = 120;
    else
        Application.targetFrameRate = 60;
}

🧠 1. CPU vs GPU Bound Detection

We compare:

CPU frame time

GPU frame time

Whichever is higher = bottleneck.

Add timing access
FrameTiming[] frameTimings = new FrameTiming[1];

bool GetTimings(out float cpuMs, out float gpuMs)
{
    FrameTimingManager.CaptureFrameTimings();
    uint count = FrameTimingManager.GetLatestTimings(1, frameTimings);

    if (count > 0)
    {
        cpuMs = frameTimings[0].cpuFrameTime;
        gpuMs = frameTimings[0].gpuFrameTime;
        return true;
    }

    cpuMs = gpuMs = 0f;
    return false;
}

Detect bound state
enum Bottleneck
{
    CPU,
    GPU,
    None
}

Bottleneck currentBottleneck = Bottleneck.None;

void DetectBottleneck()
{
    if (!GetTimings(out float cpuMs, out float gpuMs)) return;

    float diff = Mathf.Abs(cpuMs - gpuMs);

    if (diff < 1.0f)
        currentBottleneck = Bottleneck.None;
    else if (cpuMs > gpuMs)
        currentBottleneck = Bottleneck.CPU;
    else
        currentBottleneck = Bottleneck.GPU;
}


Call this every frame (cheap).

🎯 How We Use It

GPU bound → lower resolution / graphics

CPU bound → don’t touch resolution, reduce CPU-heavy systems instead (AI tick rate, physics rate, crowd density, etc.)

This prevents useless resolution drops when CPU is the problem.

    else
    {
        MemorySystem.ResumeWrites();

        TTS.UseLowLatencyVoice(false);
    }
}

bool thermalThrottling = false;

void DetectThermals(float cpuMs)
{
    if (cpuMs > 20f)
        thermalTimer += Time.unscaledDeltaTime;
    else
        thermalTimer = Mathf.Max(0f, thermalTimer - Time.unscaledDeltaTime);

    thermalThrottling = thermalTimer > 15f;
}

🖥️ 4. VRR Equivalent → Audio-Latency-Aware Frame Pacing

For a companion, audio latency matters more than visual FPS.

So we pace the render loop to the audio buffer, not the monitor.

Target frame time = audio buffer / 2

If audio buffer = 40 ms → render at ~20 ms (~50 FPS)

void SyncToAudioBuffer(float audioBufferMs)
{
    float targetFrameMs = audioBufferMs * 0.5f;
    int targetFPS = Mathf.Clamp((int)(1000f / targetFrameMs), 30, 60);

    Application.targetFrameRate = targetFPS;
}


This keeps:

✔ Lip sync stable
✔ No buffer underruns
✔ Lower power usage
