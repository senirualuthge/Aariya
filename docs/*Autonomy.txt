🧠 1. learning_engine.py (Complete)
import json
import os
import random
from collections import defaultdict

# -----------------------------
# Utilities
# -----------------------------
def clamp(x, a=0.0, b=1.0):
    return max(a, min(b, x))


# -----------------------------
# Learning Engine
# -----------------------------
class LearningEngine:
    def __init__(self, path="data/learning/"):
        self.path = path
        os.makedirs(self.path, exist_ok=True)

        self.weights = self._load("weights.json", self._default_weights())
        self.stats = self._load("bandit.json", defaultdict(lambda: {"reward": 0.0, "count": 1}))

        self.lr = 0.01
        self.exploration_rate = 0.1

    # -----------------------------
    # Defaults
    # -----------------------------
    def _default_weights(self):
        return {
            "memory": {
                "relevance": 0.35,
                "trust": 0.25,
                "intensity": 0.20,
                "session": 0.10,
                "question": 0.10
            }
        }

    # -----------------------------
    # Persistence
    # -----------------------------
    def _load(self, name, default):
        path = os.path.join(self.path, name)
        if os.path.exists(path):
            with open(path, "r") as f:
                return json.load(f)
        return default

    def _save(self, name, data):
        path = os.path.join(self.path, name)
        with open(path, "w") as f:
            json.dump(data, f, indent=2)

    def save_all(self):
        self._save("weights.json", self.weights)
        self._save("bandit.json", self.stats)

    # -----------------------------
    # Reward Function
    # -----------------------------
    def compute_reward(self, outcome):
        """
        outcome = {
            follow_up: 0/1
            sentiment_shift: -1 to +1
            engagement: text length
            interrupted: 0/1
        }
        """

        reward = (
            0.4 * outcome.get("follow_up", 0)
            + 0.3 * clamp(outcome.get("sentiment_shift", 0) + 0.5)
            + 0.2 * clamp(outcome.get("engagement", 0) / 50)
            - 0.2 * outcome.get("interrupted", 0)
        )

        return clamp(reward, 0, 1)

    # -----------------------------
    # Weight Update
    # -----------------------------
    def update_weights(self, subsystem, features, reward):
        """
        features = {
            "relevance": 0–1,
            "trust": 0–1,
            "intensity": 0–1,
            "session": 0–1,
            "question": 0–1
        }
        """

        w = self.weights[subsystem]

        for k in w:
            if k in features:
                w[k] += self.lr * reward * features[k]

        # normalize
        total = sum(w.values())
        for k in w:
            w[k] = clamp(w[k] / total, 0.05, 0.7)

    # -----------------------------
    # Bandit Selection
    # -----------------------------
    def choose(self, key_a, key_b):
        if random.random() < self.exploration_rate:
            return random.choice([key_a, key_b])

        a = self.stats[key_a]
        b = self.stats[key_b]

        avg_a = a["reward"] / a["count"]
        avg_b = b["reward"] / b["count"]

        return key_a if avg_a > avg_b else key_b

    def update_bandit(self, key, reward):
        self.stats[key]["reward"] += reward
        self.stats[key]["count"] += 1

    # -----------------------------
    # Exploration Adjustment
    # -----------------------------
    def explore(self, decision):
        """
        Randomly flip decisions for learning
        """

        if random.random() < self.exploration_rate:
            decision["use_memory"] = not decision.get("use_memory", False)

        return decision

    # -----------------------------
    # Main Learning Step
    # -----------------------------
    def learn(self, decision, features, outcome):
        reward = self.compute_reward(outcome)

        # update weights
        if decision.get("use_memory"):
            self.update_weights("memory", features.get("memory", {}), reward)

        # update bandit
        key = "memory_on" if decision.get("use_memory") else "memory_off"
        self.update_bandit(key, reward)

        self.save_all()

        return reward

    def _safe_parse(self, text: str) -> Dict[str, Any]:
        import json
        import re

        try:
            return json.loads(text)
        except:

                return json.loads(match.group(0))

class ExecutionEngine:
    def __init__(self, systems: Dict[str, Any]):
        self.systems = systems  # inject your real systems

    async def run(self, decision, input_data):
        results = {}

        if decision.strategy == "parallel":
            import asyncio
            tasks = [
                self.systems[name].run(input_data)
                for name in decision.systems
                if name in self.systems
            ]
            outputs = await asyncio.gather(*tasks)

            for name, output in zip(decision.systems, outputs):
                results[name] = output

        else:  # chain
            current_data = input_data

            for name in decision.systems:
                if name not in self.systems:
                    continue

                output = await self.systems[name].run(current_data)
                results[name] = output

                # pass forward
                current_data = output

        return results

class PolicyEngine:

    def evaluate(self, ctx: CognitiveContext) -> PolicyOutput:
        trust = ctx.trust_score
        contradiction = ctx.contradiction_ema
        latency = ctx.latency_pressure

        use_memory = trust > 0.4 and latency < 0.7

        use_contradiction = contradiction > 0.5

        response_mode = (
            "fast" if latency > 0.7 else
            "deep" if trust > 0.7 else
            "balanced"
        )

        emotion_intensity = min(1.0, ctx.emotion["arousal"] * (0.5 + trust))

        verbosity = 0.3 if latency > 0.7 else 0.7 if trust > 0.6 else 0.5

        use_voice = trust > 0.5 and latency < 0.8

        priority = self._compute_priority(ctx)

import random

def _select_best(self, intent, combos):
    epsilon = 0.1  # exploration

    if random.random() < epsilon:
        return random.choice(combos)

    scored = []
    for combo in combos:
        stat = self.stats.get(intent, combo)
        score = stat.avg_reward if stat else 0.5
        scored.append((combo, score))

    scored.sort(key=lambda x: x[1], reverse=True)

    return scored[0][0]

def execute_dag(nodes, ctx):

    completed = set()

    while len(completed) < len(nodes):

        for node in nodes.values():

            if node.name in completed:
                continue

            if all(dep in completed for dep in node.deps):
                inputs = {dep: nodes[dep].output for dep in node.deps}

                node.output = node.system.run(ctx, inputs)
                completed.add(node.name)

    return nodes["llm"].output

from concurrent.futures import ThreadPoolExecutor

def execute_parallel(nodes, ctx):
    executor = ThreadPoolExecutor()

    futures = {}

    while not all(n.ready for n in nodes.values()):

        for node in nodes.values():
            if node.ready:
                continue

            if all(nodes[d].ready for d in node.deps):
                futures[node.name] = executor.submit(node.system.run, ctx)
                node.ready = True
