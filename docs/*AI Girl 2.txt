class TraitEngine:
    def __init__(self, traits):
        self.traits = traits
        self.state = {
            "focus_level": 0.5,
            "urgency_level": 0.5,
            "confidence_level": 0.5,
            "empathy_level": 0.5,
            "cognitive_load": 0.5,
            "risk_tolerance": 0.5,
            "engagement_level": 0.5,
            "system_stability": 1.0
        }

    def update_state(self, ml_prediction):
        # Blend ML prediction with current state
        alpha = 0.7
        for key in self.state:
            self.state[key] = alpha * self.state[key] + (1 - alpha) * ml_prediction[key]

        # Apply trait nudges
        for trait in self.traits.active_traits():  # Max 3
            self.apply_trait(trait)

        # Clamp 0.0–1.0
        for key in self.state:
            self.state[key] = min(max(self.state[key], 0.0), 1.0)

        return self.state

    def apply_trait(self, trait):
        # Map trait → latent state nudges
        mapping = {
            "AssertiveEfficiency": {"confidence_level": 0.05, "focus_level": 0.03},
            "EmpatheticCalibration": {"empathy_level": 0.07, "cognitive_load": -0.02},
            "ExistentialAwareness": {"system_stability": -0.03, "cognitive_load": 0.03},
            # Add all 23 traits here
        }
        if trait in mapping:
            for k, v in mapping[trait].items():
                self.state[k] += v

using UnityEngine;

public class FaceController : MonoBehaviour
{
    public SkinnedMeshRenderer faceMesh;

    // Blendshape indices (set in Inspector)
    public int jawOpen;
    public int smile;
    public int frown;
    public int browUp;
    public int browDown;
    public int blinkLeft;
    public int blinkRight;

    void Start()
    {
        ResetFace();
        InvokeRepeating(nameof(RandomBlink), 2f, Random.Range(3f, 5f));
    }

    public void ResetFace()
    {
        SetAll(0);
    }

    void SetAll(float value)
    {
        for (int i = 0; i < faceMesh.sharedMesh.blendShapeCount; i++)
            faceMesh.SetBlendShapeWeight(i, value);
    }

    public void SetEmotion(string emotion)
    {
        ResetFace();

        switch (emotion)
        {
            case "happy":
                faceMesh.SetBlendShapeWeight(smile, 70);
                faceMesh.SetBlendShapeWeight(browUp, 30);
                break;

            case "sad":
                faceMesh.SetBlendShapeWeight(frown, 60);
                faceMesh.SetBlendShapeWeight(browDown, 40);
                break;

            case "surprised":
                faceMesh.SetBlendShapeWeight(jawOpen, 50);
                faceMesh.SetBlendShapeWeight(browUp, 80);
                break;

            case "focused":
                faceMesh.SetBlendShapeWeight(browDown, 50);
                break;
        }
    }

    void RandomBlink()
    {
        StartCoroutine(Blink());
    }

    System.Collections.IEnumerator Blink()
    {
        faceMesh.SetBlendShapeWeight(blinkLeft, 100);
        faceMesh.SetBlendShapeWeight(blinkRight, 100);
        yield return new WaitForSeconds(0.1f);
        faceMesh.SetBlendShapeWeight(blinkLeft, 0);
        faceMesh.SetBlendShapeWeight(blinkRight, 0);
    }
}


✔ This alone fixes 60% of stiffness.

    private VRMBlendShapeProxy blendShapeProxy;

    // Smoothed values
    private float valenceSmooth;
    private float arousalSmooth;

    void Start()
    {
        animator = GetComponent<Animator>();
        blendShapeProxy = GetComponent<VRMBlendShapeProxy>();

        valenceSmooth = valence;
        arousalSmooth = arousal;
    }

    void Update()
    {
        // Smooth emotion values
        valenceSmooth = Mathf.Lerp(valenceSmooth, valence, Time.deltaTime * emotionSmoothSpeed);
        arousalSmooth = Mathf.Lerp(arousalSmooth, arousal, Time.deltaTime * emotionSmoothSpeed);

        UpdateAnimator();
        UpdateFace();
        UpdateBodyEnergy();
    }

    void UpdateAnimator()
    {
        animator.SetBool("IsSpeaking", isSpeaking);
        animator.SetFloat("Arousal", arousalSmooth);
        animator.SetFloat("Valence", valenceSmooth);
    }

    void UpdateFace()
    {
        if (blendShapeProxy == null) return;

        // Reset all expressions first
        blendShapeProxy.ImmediatelySetValue(BlendShapeKey.CreateFromPreset(BlendShapePreset.Neutral), 1f);

        float joy = Mathf.Clamp01(valenceSmooth);
        float sad = Mathf.Clamp01(-valenceSmooth);
        float angry = Mathf.Clamp01(-valenceSmooth * arousalSmooth);

        blendShapeProxy.ImmediatelySetValue(BlendShapeKey.CreateFromPreset(BlendShapePreset.Joy), joy);
        blendShapeProxy.ImmediatelySetValue(BlendShapeKey.CreateFromPreset(BlendShapePreset.Sad), sad);
        blendShapeProxy.ImmediatelySetValue(BlendShapeKey.CreateFromPreset(BlendShapePreset.Angry), angry);

        // Arousal → eye openness
        float surprised = arousalSmooth * 0.5f;
        blendShapeProxy.ImmediatelySetValue(BlendShapeKey.CreateFromPreset(BlendShapePreset.Fun), surprised);
    }

    void UpdateBodyEnergy()
    {
        // Higher arousal → faster idle motion
        float animationSpeed = Mathf.Lerp(0.8f, 1.4f, arousalSmooth);
        animator.speed = animationSpeed;

        // Optional posture shift via animator float (if you add it)
        // animator.SetFloat("Posture", valenceSmooth);
    }

    // 🔌 Call this from your AI system
    public void SetEmotion(float newValence, float newArousal, bool speaking)
    {
        valence = Mathf.Clamp(newValence, -1f, 1f);
        arousal = Mathf.Clamp01(newArousal);
        isSpeaking = speaking;
    }
}

using UnityEngine;
using NativeWebSocket;
using System.Text;

public class AIWebSocketBridge : MonoBehaviour
{
    public string serverURL = "ws://localhost:8080";

    private WebSocket websocket;
    public EmotionAvatarController avatar;

    async void Start()
    {
        websocket = new WebSocket(serverURL);

        websocket.OnOpen += () =>
        {
            Debug.Log("WebSocket connected");
        };

        websocket.OnMessage += (bytes) =>
        {
            string json = Encoding.UTF8.GetString(bytes);
            ProcessMessage(json);
        };

        websocket.OnError += (e) =>
        {
            Debug.Log("WebSocket error: " + e);
        };

        websocket.OnClose += (e) =>
        {
            Debug.Log("WebSocket closed");
        };

        await websocket.Connect();
    }

    void ProcessMessage(string json)
    {
        // Expected JSON:
        // { "valence": 0.5, "arousal": 0.7, "isSpeaking": true }

        EmotionMessage msg = JsonUtility.FromJson<EmotionMessage>(json);

        if (avatar != null)
        {
            avatar.SetEmotion(msg.valence, msg.arousal, msg.isSpeaking);
        }
    }

    private async void OnApplicationQuit()
    {
        await websocket.Close();
    }
}

[System.Serializable]
public class EmotionMessage
{
    public float valence;
    public float arousal;
    public bool isSpeaking;
}

using UnityEngine;

public class BreathingController : MonoBehaviour
{
    public Transform chestBone;
    public EmotionAvatarController emotionSource;

    public float baseBreathSpeed = 1.2f;
    public float breathAmplitude = 0.01f;

    private Vector3 initialScale;
    private float breathTimer;

    void Start()
    {
        if (chestBone != null)
            initialScale = chestBone.localScale;
    }

    void Update()
    {
        if (chestBone == null || emotionSource == null) return;

        float arousal = emotionSource.arousal;

        float speed = baseBreathSpeed + arousal * 1.5f;
        float amplitude = breathAmplitude + arousal * 0.02f;

        breathTimer += Time.deltaTime * speed;

        float scaleOffset = Mathf.Sin(breathTimer) * amplitude;

        chestBone.localScale = initialScale + new Vector3(scaleOffset, scaleOffset, scaleOffset);
    }
}

using UnityEngine;
using WebSocketSharp;

[Serializable]
public class EmotionData
{
    public float valence;   // -1 to 1
    public float energy;    // 0 to 1
    public float anger;     // 0 to 1
}

public class EmotionWebSocket : MonoBehaviour
{
    public Animator animator;

    private WebSocket ws;

    // Smoothed values
    private float valence;
    private float energy;
    private float anger;

    public float smoothSpeed = 5f;

    void Start()
    {
        ws = new WebSocket("ws://127.0.0.1:8765");

        ws.OnMessage += (sender, e) =>
        {
            EmotionData data = JsonUtility.FromJson<EmotionData>(e.Data);

            valence = data.valence;
            energy  = data.energy;
            anger   = data.anger;
        };

        ws.Connect();
    }

    void Update()
    {
        if (animator == null) return;

        // Smooth animator values
        float mood = animator.GetFloat("MoodValence");
        float en   = animator.GetFloat("Energy");

        mood = Mathf.Lerp(mood, valence, Time.deltaTime * smoothSpeed);
        en   = Mathf.Lerp(en, energy,  Time.deltaTime * smoothSpeed);

        animator.SetFloat("MoodValence", mood);
        animator.SetFloat("Energy", en);
        animator.SetBool("IsAngry", anger > 0.6f);

        // Optional facial
        animator.SetFloat("Smile", Mathf.Clamp01((mood + 1f) / 2f));
        animator.SetFloat("BlinkRate", Mathf.Lerp(0.5f, 1.5f, en));
    }

    void OnApplicationQuit()
    {
        if (ws != null)
            ws.Close();
    }
}

import asyncio
import json
import random
import websockets

async def send_emotions(websocket):
    while True:
        data = {
            "valence": random.uniform(-1.0, 1.0),
            "energy": random.uniform(0.2, 0.9),
            "anger": random.uniform(0.0, 1.0)
        }

        await websocket.send(json.dumps(data))
        await asyncio.sleep(0.5)  # send twice per second

async def handler(websocket):
    await send_emotions(websocket)

async def main():
    async with websockets.serve(handler, "127.0.0.1", 8765):
        print("Emotion server running on port 8765")
        await asyncio.Future()  # run forever

asyncio.run(main())
