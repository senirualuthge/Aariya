# AI Girl TTS (Text-to-Speech) System Overview

This document outlines how the AI Girl's voice system functions, creating a lifelike and emotionally responsive experience.

## 1. The Pre-Speech "Intelligence" Phase
Before a single word is spoken, the **PreSpeechSystem** triggers subtle animations (head lift, inhale) approximately **150-250ms** before the audio starts. This mirrors human behavior where we prepare to speak, making the avatar feel "present" and intelligent.

## 2. Emotional Pipeline
The voice isn't static; it adapts to the content of the conversation through the **Emotional Voice Processor**:
- **Sentiment Detection**: The system scans dialogue for keywords and emojis (e.g., "hehe", "sad", "!!!", "angry") to determine the emotional state.
- **Dynamic Adjustments**: Based on the detected emotion, the system modifies:
    - **Pitch**: Higher for excitement/bubbliness, lower for sultriness or sadness.
    - **Rate**: Slower for intimate whispers or sadness, faster for energy bursts.
    - **Volume**: Subtle adjustments for intimacy or emphasis.

## 3. Human-like Nuances
To avoid the "robot" effect, the system injects several natural elements:
- **Breath Inserter**: Automatically adds punctuation-based pauses (`<break time='300ms'/>`) to mimic natural breathing and speech rhythm.
- **Emphasis Engine**: Dynamically highlights certain words (e.g., "really", "love", "never") by slightly increasing pitch and volume while slowing down.
- **Circadian Rhythm**: The voice shifts depending on the time of day:
    - **Morning**: Optimistic and gentle.
    - **Evening/Night**: Softer, lower energy, and warmer.

## 4. Lip-Sync Integration
The **LipSyncEngine** bridges the gap between text and visuals:
- It processes the dialogue into a sequence of **Visemes** (mouth shapes).
- These visemes (mapping to sounds like 'AA', 'OO', 'FF', etc.) are sent to the avatar's face model in real-time.
- **Intensity Mapping**: ALL CAPS words trigger higher intensity mouth shapes for visual emphasis.

## 5. Intelligence Features
- **Voice Mirroring**: The system can adapt to the user's speech patterns (speed and energy) to create a more harmonious interaction.
- **Fatigue System**: If the AI is "tired," responses become shorter and the voice rate slows down.
- **Silent Mode**: A low-latency "whisper" mode for quiet environments.

# server/voice/tts_engine.py

class EmotionalTTSEngine:
    def __init__(self, tts_provider):
        self.tts = tts_provider  # ElevenLabs / OpenAI / local

    def generate(self, text, brain_state, user_state):
        valence = brain_state["emotion_target"]["valence"]
        arousal = brain_state["emotion_target"]["arousal"]
        action  = brain_state["social_action"]
        trust   = brain_state["trust_score"]

        # 1. Get voice parameters
        params = self._map_voice(valence, arousal, action, trust, user_state)

        # 2. Apply speech styling
        styled_text = self._style_text(text, action, trust)

        # 3. Generate audio
        audio = self.tts.synthesize(
            text=styled_text,
            pitch=params["pitch"],
            rate=params["rate"],
            volume=params["volume"],
            style=params["style"]
        )

        return audio

    # ----------------------------------------

    def _map_voice(self, valence, arousal, action, trust, user_state):
        pitch = 1.3
        rate = 1.0
        volume = 1.0

        # 🎭 Emotion mapping
        pitch += valence * 0.4
        rate  += arousal * 0.3

        # 🧠 RL behavior mapping
        if action == "playful":
            pitch += 0.2
            rate  += 0.1

        elif action == "empathetic":
            rate  -= 0.2
            pitch -= 0.1

        elif action == "assertive":
            volume += 0.2

        # ❤️ Trust-based intimacy
        if trust > 0.8:
            rate  -= 0.1
            volume -= 0.1  # softer voice

        # 🪞 Voice mirroring (user energy)
        pitch += user_state.get("arousal", 0) * 0.2
        rate  += user_state.get("arousal", 0) * 0.1

        return {
            "pitch": max(0.8, min(2.0, pitch)),
            "rate": max(0.7, min(1.4, rate)),
            "volume": max(0.5, min(1.5, volume)),
            "style": action
        }

    # ----------------------------------------

    def _style_text(self, text, action, trust):
        # Natural speech shaping

        if action == "playful":
            text = f'<prosody pitch="+15%" rate="fast" volume="loud">{text}</prosody>'

        elif action == "empathetic":
            text = f'<prosody pitch="-10%" rate="slow" volume="soft">{text}</prosody>'

        elif action == "curious":
            text = f'<prosody pitch="+20%" rate="medium">{text}</prosody>'

        # Softening with trust
        if trust > 0.7:
            text = f'<prosody pitch="-10%" volume="x-soft">{text}</prosody>'

        return text

def compute_reward(prev_state, new_state):
    reward = 0

    # 😊 User became happier
    reward += (new_state["valence"] - prev_state["valence"]) * 2.0

    # ⚡ User more engaged
    reward += (new_state["arousal"] - prev_state["arousal"]) * 1.0

    # ⏱️ Faster response (engagement)
    if new_state["response_time"] < 3:
        reward += 0.5

    # 🚫 Penalize negative reaction
    if new_state["valence"] < -0.3:
        reward -= 1.5

    return reward

import librosa
import noisereduce as nr
import soundfile as sf
import os

input_dir = "dataset/wavs"
output_dir = "dataset_clean/wavs"
os.makedirs(output_dir, exist_ok=True)

for file in os.listdir(input_dir):
    if file.endswith(".wav"):
        path = os.path.join(input_dir, file)

        audio, sr = librosa.load(path, sr=22050)

        # Noise reduction
        reduced = nr.reduce_noise(y=audio, sr=sr)

        # Normalize
        reduced = reduced / max(abs(reduced))

        sf.write(os.path.join(output_dir, file), reduced, sr)

import torch
import torch.nn as nn

class PolicyNet(nn.Module):
    def __init__(self, state_dim, action_dim):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, 128),
            nn.ReLU(),
            nn.Linear(128, action_dim),
            nn.Softmax(dim=-1)
        )

    def forward(self, x):
        return self.net(x)
🎯 Value network
class ValueNet(nn.Module):
    def __init__(self, state_dim):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 1)
        )

    def forward(self, x):
        return self.net(x)
🔁 PPO Training Loop
for epoch in range(EPOCHS):
    states, actions, rewards, old_probs = collect_trajectory()

    states = torch.tensor(states, dtype=torch.float32)
    actions = torch.tensor(actions)
    rewards = torch.tensor(rewards)

    # Advantage
    values = value_net(states).squeeze()
    advantages = rewards - values.detach()

    # Policy update
    probs = policy_net(states)
    dist = torch.distributions.Categorical(probs)
    new_probs = dist.log_prob(actions)

    ratio = torch.exp(new_probs - old_probs)

    clip = 0.2
    loss = -torch.min(
        ratio * advantages,
        torch.clamp(ratio, 1 - clip, 1 + clip) * advantages
    ).mean()

    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

class AudioQueue {
  constructor() {
    this.queue = []
    this.isPlaying = false
  }

  add(chunk) {
    this.queue.push(chunk)
    if (!this.isPlaying) this.playNext()
  }

  async playNext() {
    if (this.queue.length === 0) {
      this.isPlaying = false
      return
    }

    this.isPlaying = true
    const audio = this.queue.shift()

    await playAudio(audio)
    this.playNext()
  }
}

class ReplayBuffer:

    def sample(self):

class FusionNet(nn.Module):

        super().__init__()
        self.transformer = nn.TransformerEncoder(
            nn.TransformerEncoderLayer(d_model=dim, nhead=8),

            x = text_emb + voice_emb
        else:
            x = text_emb

class EmotionHead(nn.Module):
    def __init__(self, dim, num_classes=6):
        super().__init__()
        self.classifier = nn.Linear(dim, num_classes)
        self.valence = nn.Linear(dim, 1)
        self.arousal = nn.Linear(dim, 1)

    def forward(self, x):
        return {
            "emotion_logits": self.classifier(x),
            "valence": torch.tanh(self.valence(x)),
            "arousal": torch.sigmoid(self.arousal(x))

class SelfImprovingDataset:
    def __init__(self, base_model, reward_model, buffer_size=1000):
        self.base_model = base_model
        self.reward_model = reward_model
        self.buffer = []
        self.buffer_size = buffer_size

    async def generate_training_batch(self, prompt, n_samples=4):
        candidates = []
        for _ in range(n_samples):
            response = await self.base_model.generate(prompt)
            score = self.reward_model.score(response)
            candidates.append((response, score))
        candidates.sort(key=lambda x: x[1], reverse=True)
        best = candidates[0][0]
        self.buffer.extend(candidates[:2])
        if len(self.buffer) > self.buffer_size:
            self.buffer = self.buffer[-self.buffer_size:]
        return best

    async def retrain(self, model):
        if len(self.buffer) < 10:
            return model
        inputs = [item[0] for item in self.buffer]
        return await model.fine_tune(inputs)

    float* output,
    int N,
    int d,
    float scale
) {
    int row = blockIdx.x * blockDim.x + threadIdx.x;
    if (row >= N) return;

    extern __shared__ float shared[];
    float* sK = shared;
    float* sV = &shared[d];

    for (int i = threadIdx.x; i < d; i += blockDim.x) {
        sK[i] = K[row * d + i];
        sV[i] = V[row * d + i];
    }
    __syncthreads();

    float score = 0.0f;
    for (int i = 0; i < d; i++) {
        score += Q[row * d + i] * sK[i];
    }
    score = expf(score * scale);

    float val = 0.0f;
    for (int i = 0; i < d; i++) {
        val += score * sV[i];
    }

    for (int i = 0; i < d; i++) {
        float act = val * W[row * d + i];
        output[row * d + i] = act > 0.0f ? act : 0.1f * act;
    }
}

AST:

class Node:
    def __init__(self, children=None):
        self.children = children or []

    def evaluate(self, context):
        return torch.nn.functional.relu(hidden) if self.bias is not None else hidden

class Update(Node):
    def __init__(self, target, value):
        super().__init__()
        self.target = target
        self.value = value

    def evaluate(self, context):
        context[self.target] = self.value.evaluate(context)
        return context[self.target]

class Mul(Node):
    def __init__(self, left, right):
        super().__init__()
        self.left = left
        self.right = right

    def evaluate(self, context):
        return self.left.evaluate(context) * self.right.evaluate(context)

class Clip(Node):
    def __init__(self, value, lo, hi):
        super().__init__()
        self.value = value
        self.lo = lo
        self.hi = hi

    def evaluate(self, context):
        v = self.value.evaluate(context)
        return max(self.lo, min(self.hi, v))

import torch

class GPUPIDController:
    def __init__(self, d, kp=0.5, ki=0.1, kd=0.2, device="cuda"):
        self.kp = kp
        self.ki = ki
        self.kd = kd

        self.device = device

        self.prev_error = torch.zeros(d, device=device)
        self.integral = torch.zeros(d, device=device)

    @torch.no_grad()
    def step(self, error_batch):
        """
        error_batch: [B, D]
        """

        # Mean-field reduction (stable control signal)
        e = error_batch.mean(dim=0)

        # integral accumulation (GPU vector op)
        self.integral = 0.99 * self.integral + e

        # derivative term
        derivative = e - self.prev_error

        # PID control signal (fully vectorized)
        u = (
            self.kp * e +
            self.ki * self.integral +
            self.kd * derivative
        )

        self.prev_error = e.clone()

        return u

    float* z,
    float* error,
    float* integral,
    float* output,

        self.Kp = kp
        self.Ki = ki
        self.Kd = kd

        self.e_prev = 0
        self.integral = 0

    @torch.no_grad()
    def step(self, e):

        # PID output
        self.integral = 0.99 * self.integral + e
        derivative = e - self.e_prev

        u = (
            self.Kp * e +
            self.Ki * self.integral +
            self.Kd * derivative

        # instability signal
        S = torch.norm(e - self.e_prev) + torch.norm(u)

        self.Ki *= torch.exp(-0.02 * S)
        self.Kd += 0.01 * S

        self.e_prev = e.clone()

        return u
